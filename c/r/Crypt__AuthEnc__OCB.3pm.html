Crypt::AuthEnc::OCB(3) Documentación de Perl aportada por el usuario Crypt::AuthEnc::OCB(3)

NOMBRE

Crypt::AuthEnc::OCB - Cifrado autenticado en modo OCBv3

SINOPSIS

### Interfaz OO
use Crypt::AuthEnc::OCB;
# cifrar y autenticar
my $ae = Crypt::AuthEnc::OCB->new("AES", $key, $nonce, $tag_len);
$ae->adata_add('datos_adicionales_autenticados1');
$ae->adata_add('datos_adicionales_autenticados2');
my $ct = $ae->encrypt_add('datos1');
$ct .= $ae->encrypt_add('datos2');
$ct .= $ae->encrypt_add('datos3');
$ct .= $ae->encrypt_last('resto de los datos');
my $tag = $ae->encrypt_done();
# descifrar y verificar
my $ae = Crypt::AuthEnc::OCB->new("AES", $key, $nonce, $tag_len);
$ae->adata_add('datos_adicionales_autenticados1');
$ae->adata_add('datos_adicionales_autenticados2');
my $pt = $ae->decrypt_add('cifrado1');
$pt .= $ae->decrypt_add('cifrado2');
$pt .= $ae->decrypt_add('cifrado3');
$pt .= $ae->decrypt_last('resto de los datos');
my $tag = $ae->decrypt_done();
die "descifrado fallido" unless $tag eq $expected_tag;
#o
my $result = $ae->decrypt_done($expected_tag); # 0 o 1
### Interfaz funcional
use Crypt::AuthEnc::OCB qw(ocb_encrypt_authenticate ocb_decrypt_verify);
my ($ciphertext, $tag) = ocb_encrypt_authenticate('AES', $key, $nonce, $adata, $tag_len, $plaintext);
my $plaintext = ocb_decrypt_verify('AES', $key, $nonce, $adata, $ciphertext, $tag);

DESCRIPCIÓN

Este módulo implementa OCB v3 de acuerdo con https://tools.ietf.org/html/rfc7253

EXPORTAR

Nada se exporta por defecto.

Puedes exportar funciones seleccionadas:

use Crypt::AuthEnc::OCB qw(ocb_encrypt_authenticate ocb_decrypt_verify);

FUNCIONES

ocb_encrypt_authenticate

my ($ciphertext, $tag) = ocb_encrypt_authenticate($cipher, $key, $nonce, $adata, $tag_len, $plaintext);
# $cipher .. 'AES' o el nombre de cualquier otro cifrador con longitud de bloque de 16 bytes
# $key ..... clave AES de la longitud adecuada (128/192/256 bits)
# $nonce ... nonce/sal única (no es necesario mantenerlo en secreto)
# $adata ... datos adicionales autenticados
# $tag_len . longitud requerida de la etiqueta de salida

ocb_decrypt_verify

my $plaintext = ocb_decrypt_verify($cipher, $key, $nonce, $adata, $ciphertext, $tag);
# en caso de error devuelve undef

MÉTODOS

new

my $ae = Crypt::AuthEnc::OCB->new($cipher, $key, $nonce, $tag_len);
# $cipher .. 'AES' o el nombre de cualquier otro cifrador con longitud de bloque de 16 bytes
# $key ..... clave AES de la longitud adecuada (128/192/256 bits)
# $nonce ... nonce/sal única (no es necesario mantenerlo en secreto)
# $tag_len . longitud requerida de la etiqueta de salida

adata_add

$ae->adata_add($adata);                        # se puede llamar varias veces

encrypt_add

$ciphertext = $ae->encrypt_add($data);         # se puede llamar varias veces
# ¡CUIDADO!: el tamaño de $data debe ser múltiplo de la longitud del bloque (16 para AES)

encrypt_last

$ciphertext = $ae->encrypt_last($data);

encrypt_done

$tag = $ae->encrypt_done();                    # devuelve el valor de $tag

decrypt_add

$plaintext = $ae->decrypt_add($ciphertext);    # se puede llamar varias veces
# ¡CUIDADO!: el tamaño de $ciphertext debe ser múltiplo de la longitud del bloque (16 para AES)

decrypt_last

$plaintext = $ae->decrypt_last($data);

decrypt_done

my $tag = $ae->decrypt_done;           # devuelve el valor de $tag
#o
my $result = $ae->decrypt_done($tag);  # devuelve 1 (éxito) o 0 (fallo)

clone

my $ae_new = $ae->clone;
