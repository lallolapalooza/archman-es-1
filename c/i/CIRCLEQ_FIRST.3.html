struct entry {
    int data;
    CIRCLEQ_ENTRY(entry) entries;           /* Cola */
};
CIRCLEQ_HEAD(circlehead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct circlehead head;                 /* Cabecera de la cola */
    int i;
    CIRCLEQ_INIT(&head);                    /* Inicializar la cola */
    n1 = malloc(sizeof(struct entry));      /* Insertar en la cabecera */
    CIRCLEQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* Insertar en la cola */
    CIRCLEQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* Insertar después */
    CIRCLEQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* Insertar antes */
    CIRCLEQ_INSERT_BEFORE(&head, n2, n3, entries);
    CIRCLEQ_REMOVE(&head, n2, entries);     /* Eliminación */
    free(n2);
                                            /* Recorrido hacia adelante */
    i = 0;
    CIRCLEQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* Recorrido hacia atrás */
    CIRCLEQ_FOREACH_REVERSE(np, &head, entries)
        printf("%i\n", np->data);
                                            /* Eliminación de la cola */
    n1 = CIRCLEQ_FIRST(&head);
    while (n1 != (void *)&head) {
        n2 = CIRCLEQ_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    CIRCLEQ_INIT(&head);
    exit(EXIT_SUCCESS);
}
