subroutine cget24 (logical comp, integer jtype, real thresh, integer, dimension( 4 ) iseed, integer nounit, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( lda, * ) h, complex, dimension( lda, * ) ht, complex, dimension( * ) w, complex, dimension( * ) wt, complex, dimension( * ) wtmp, complex, dimension( ldvs, * ) vs, integer ldvs, complex, dimension( ldvs, * ) vs1, real rcdein, real rcdvin, integer nslct, integer, dimension( * ) islct, integer isrt, real, dimension( 17 ) result, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, logical, dimension( * ) bwork, integer info)

CGET24

Propósito:

!>
!>    CGET24 comprueba el controlador experto del problema de valores propios no simétricos (forma de Schur) CGEESX.
!>
!>    Si COMP = .FALSE., se realizarán las primeras 13 de las siguientes pruebas en la matriz de entrada A, y también las pruebas 14 y 15
!>    si LWORK es suficientemente grande.
!>    Si COMP = .TRUE., se realizarán las 17 pruebas.
!>
!>    (1)     0 si T está en forma de Schur, 1/ulp en caso contrario
!>           (sin ordenación de valores propios)
!>
!>    (2)     | A - VS T VS' | / ( n |A| ulp )
!>
!>      Aquí VS es la matriz de vectores propios de Schur, y T está en forma de Schur
!>      (sin ordenación de valores propios).
!>
!>    (3)     | I - VS VS' | / ( n ulp ) (sin ordenación de valores propios).
!>
!>    (4)     0     si W son valores propios de T
!>            1/ulp en caso contrario
!>            (sin ordenación de valores propios)
!>
!>    (5)     0     si T (con VS) = T (sin VS),
!>            1/ulp en caso contrario
!>            (sin ordenación de valores propios)
!>
!>    (6)     0     si valores_propios(con VS) = valores_propios(sin VS),
!>            1/ulp en caso contrario
!>            (sin ordenación de valores propios)
!>
!>    (7)     0 si T está en forma de Schur, 1/ulp en caso contrario
!>            (con ordenación de valores propios)
!>
!>    (8)     | A - VS T VS' | / ( n |A| ulp )
!>
!>      Aquí VS es la matriz de vectores propios de Schur, y T está en forma de Schur
!>      (con ordenación de valores propios).
!>
!>    (9)     | I - VS VS' | / ( n ulp ) (con ordenación de valores propios).
!>
!>    (10)    0     si W son valores propios de T
!>            1/ulp en caso contrario
!>            Si el espacio de trabajo es suficiente, también se compara W con y
!>            sin números de condición recíprocos
!>            (con ordenación de valores propios)
!>
!>    (11)    0     si T(con VS) = T(sin VS),
!>            1/ulp en caso contrario
!>            Si el espacio de trabajo es suficiente, también se compara T con y sin
!>            números de condición recíprocos
!>            (con ordenación de valores propios)
!>
!>    (12)    0     si valores_propios(con VS) = valores_propios(sin VS),
!>            1/ulp en caso contrario
!>            Si el espacio de trabajo es suficiente, también se compara VS con y sin
!>            números de condición recíprocos
!>            (con ordenación de valores propios)
!>
!>    (13)    si la ordenación funcionó y SDIM es el número de
!>            valores propios que fueron SELECCIONADOS
!>            Si el espacio de trabajo es suficiente, también se compara SDIM con y
!>            sin números de condición recíprocos
!>
!>    (14)    si RCONDE es el mismo sin importar si VS y/o RCONDV se calcularon
!>
!>    (15)    si RCONDV es el mismo sin importar si VS y/o RCONDE se calcularon
!>
!>    (16)  |RCONDE - RCDEIN| / cond(RCONDE)
!>
!>       RCONDE es el número de condición promedio recíproco de valores propios
!>       calculado por CGEESX y RCDEIN (el valor verdadero precalculado)
!!>       se proporciona como entrada. cond(RCONDE) es el número de condición
!>       de RCONDE, y tiene en cuenta los errores al calcular RCONDE,
!>       de modo que la cantidad resultante debería ser O(ULP). cond(RCONDE)
!>       se da esencialmente por norm(A)/RCONDV.
!>
!>    (17)  |RCONDV - RCDVIN| / cond(RCONDV)
!>
!>       RCONDV es el número de condición recíproco del subespacio invariante derecho
!>       calculado por CGEESX y RCDVIN (el valor verdadero precalculado)
!>       se proporciona como entrada. cond(RCONDV) es el número de condición
!>       de RCONDV, y tiene en cuenta los errores al calcular RCONDV,
!>       de modo que la cantidad resultante debería ser O(ULP).
!>       cond(RCONDV) se da esencialmente por norm(A)/RCONDE.
!>

Parámetros

COMP 
!>          COMP es LOGICAL
!>          COMP describe qué pruebas de entrada realizar:
!>            = .FALSE. si los números de condición calculados no deben
!>                      ser probados contra RCDVIN y RCDEIN
!>            = .TRUE.  si deben ser comparados
!>

JTYPE

!>          JTYPE es INTEGER
!>          Tipo de matriz de entrada. Se utiliza para etiquetar la salida si ocurre un error.
!>

ISEED

!>          ISEED es un array INTEGER, dimension (4)
!>          Si COMP = .FALSE., la semilla del generador de números aleatorios
!>          utilizada para producir la matriz.
!>          Si COMP = .TRUE., ISEED(1) = el número del ejemplo.
!>          Se utiliza para etiquetar la salida si ocurre un error.
!>

THRESH

!>          THRESH es REAL
!>          Una prueba contará como si el , calculado como
!>          se describe anteriormente, excede THRESH. Tenga en cuenta que el error
!>          se escala para ser O(1), por lo que THRESH debería ser un múltiplo
!>          razonablemente pequeño de 1, por ejemplo, 10 o 100. En particular,
!>          no debe depender de la precisión (simple o doble)
!>          o el tamaño de la matriz. Debe ser al menos cero.
!>

NOUNIT

!>          NOUNIT es INTEGER
!>          El número de unidad FORTRAN para imprimir mensajes de error
!>          (por ejemplo, si una rutina devuelve INFO distinto de 0.)
!>

N

!>          N es INTEGER
!>          La dimensión de A. N debe ser al menos 0.
!>

A

!>          A es un array COMPLEX, dimension (LDA, N)
!>          Se utiliza para contener la matriz cuyos valores propios se deben
!>          calcular.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de A, y H. LDA debe ser al
!>          menos 1 y al menos N.
!>

H

!>          H es un array COMPLEX, dimension (LDA, N)
!>          Otra copia de la matriz de prueba A, modificada por CGEESX.
!>

HT

!>          HT es un array COMPLEX, dimension (LDA, N)
!>          Otra copia más de la matriz de prueba A, modificada por CGEESX.
!>

W

!>          W es un array COMPLEX, dimension (N)
!>          Los valores propios calculados de A.
!>

WT

!>          WT es un array COMPLEX, dimension (N)
!>          Al igual que W, este array contiene los valores propios de A,
!>          pero los calculados cuando CGEESX solo calcula una descomposición
!>          de valores propios parcial, es decir, sin vectores de Schur.
!>

WTMP

!>          WTMP es un array COMPLEX, dimension (N)
!>          Al igual que W, este array contiene los valores propios de A,
!>          pero ordenados por parte real o imaginaria creciente.
!>

VS

!>          VS es un array COMPLEX, dimension (LDVS, N)
!>          VS contiene los vectores de Schur calculados.
!>

NAME

cget24

SYNOPSIS

This subroutine computes the 24 residual norm estimates defined in this chapter.

SUBROUTINE CGET24( N, COMP, LO, HI, RCDEIN, RCDVIN, NSLCT, ISLCT, ISRT,
                   RESULT, WORK, LWORK, RWORK, BWORK, INFO )

The subroutine calls the following LAPACK routines:

   LAPACK routines: None
   Other routines: None

Function/Subroutine Documentation

N

!>          N is INTEGER
!>          The number of eigenvalues, and the size of the matrix A.
!>

COMP

!>          COMP is LOGICAL
!>          If COMP = .TRUE., the selected eigenvalues and Schur vectors
!>          are computed.
!>          If COMP = .FALSE., the subroutine expects the precomputed
!>          reciprocal condition numbers RCDEIN and RCDVIN.
!>

LO

!>          LO is INTEGER
!>          The lower bound of the interval of interest.
!>

HI

!>          HI is INTEGER
!>          The upper bound of the interval of interest.
!>

RCDEIN

!>          RCDEIN is REAL
!>          When COMP = .TRUE. RCDEIN holds the precomputed reciprocal
!>          condition number for the average of selected eigenvalues.
!>

RCDVIN

!>          RCDVIN is REAL
!>          When COMP = .TRUE. RCDVIN holds the precomputed reciprocal
!>          condition number for the selected right invariant subspace.
!>

NSLCT

!>          NSLCT is INTEGER
!>          When COMP = .TRUE. the number of selected eigenvalues
!>          corresponding to the precomputed values RCDEIN and RCDVIN.
!>

ISLCT

!>          ISLCT is INTEGER array, dimension (NSLCT)
!>          When COMP = .TRUE. ISLCT selects the eigenvalues of the
!>          input matrix corresponding to the precomputed values RCDEIN
!>          and RCDVIN. For I=1, ... ,NSLCT, if ISLCT(I) = J, then the
!>          eigenvalue with the J-th largest real or imaginary part is
!>          selected. The real part is used if ISRT = 0, and the
!>          imaginary part if ISRT = 1.
!>          Not referenced if COMP = .FALSE.
!>

ISRT

!>          ISRT is INTEGER
!>          When COMP = .TRUE., ISRT describes how ISLCT is used to
!>          choose a subset of the spectrum.
!>          Not referenced if COMP = .FALSE.
!>

RESULT

!>          RESULT is REAL array, dimension (17)
!>          The values computed by the 17 tests described above.
!>          The values are currently limited to 1/ulp, to avoid
!>          overflow.
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N*N)
!>

LWORK

!>          LWORK is INTEGER
!>          The number of entries in WORK to be passed to CGEESX. This
!>          must be at least 2*N, and N*(N+1)/2 if tests 14--16 are to
!>          be performed.
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

BWORK

!>          BWORK is LOGICAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          If 0,  successful exit.
!>          If <0, input parameter -INFO had an incorrect value.
!>          If >0, CGEESX returned an error code, the absolute
!>                 value of which is returned.
!>
