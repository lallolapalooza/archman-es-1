subroutine cget02 (character trans, integer m, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

CGET02

Propósito

!>
!> CGET02 calcula el residuo para una solución de un sistema de ecuaciones
!> lineales op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> donde op(A) = A, A**T, o A**H, dependiendo de TRANS, y EPS es el
!> epsilon de la máquina.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones:
!>          = 'N':  A    * X = B  (Sin transposición)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada)
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B, la matriz de segundos miembros.
!>          NRHS >= 0.
!>

A

!>          A es array COMPLEX, dimensión (LDA,N)
!>          La matriz original A de M x N.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

X

!>          X es array COMPLEX, dimensión (LDX,NRHS)
!>          Los vectores de solución calculados para el sistema de
!>          ecuaciones lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  Si TRANS = 'N',
!>          LDX >= max(1,N); si TRANS = 'T' o 'C', LDX >= max(1,M).
!>

B

!>          B es array COMPLEX, dimensión (LDB,NRHS)
!>          En la entrada, los vectores del segundo miembro para el sistema
!>          de ecuaciones lineales.
!>          En la salida, B se sobrescribe con la diferencia B - op(A)*X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  Si TRANS = 'N',
!>          LDB >= max(1,M); si TRANS = 'T' o 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK es array REAL, dimensión (M)
!>

RESID

!>          RESID es REAL
!>          El máximo sobre el número de segundos miembros de
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
