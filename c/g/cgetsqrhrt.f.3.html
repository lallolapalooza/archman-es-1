subroutine cgetsqrhrt (integer m, integer n, integer mb1, integer nb1, integer nb2, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer lwork, integer info)

CGETSQRHRT

Propósito:

!>
!> CGETSQRHRT calcula una factorización QR bloqueada de tamaño NB2
!> de una matriz compleja M por N, A, con M >= N.
!>
!>    A = Q * R.
!>
!> La rutina utiliza internamente una factorización TSQR bloqueada de tamaño NB1
!> en columnas y de tamaño MB1 en filas, y realiza la reconstrucción
!> de los vectores de Householder a partir de la salida de TSQR. La rutina también
!> convierte el factor R_tsqr de la salida de la factorización TSQR en
!> el factor R que corresponde a la factorización QR de Householder,
!>
!>    A = Q_tsqr * R_tsqr = Q * R.
!>
!> Los factores de salida Q y R se almacenan en el mismo formato que en CGEQRT
!> (Q está en representación WY compacta y bloqueada). Consulte la documentación
!> de CGEQRT para obtener más detalles sobre el formato.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A. M >= N >= 0.
!>

MB1

!>          MB1 es INTEGER
!>          El tamaño del bloque de filas que se utilizará en la TSQR bloqueada.
!>          MB1 > N.
!>

NB1

!>          NB1 es INTEGER
!>          El tamaño del bloque de columnas que se utilizará en la TSQR bloqueada.
!>          N >= NB1 >= 1.
!>

NB2

!>          NB2 es INTEGER
!>          El tamaño del bloque que se utilizará en la QR bloqueada de salida.
!>          NB2 >= 1.
!>

A

!>          A es array COMPLEX*16, dimensión (LDA,N)
!>
!>          En la entrada: una matriz A de M por N.
!>
!>          En la salida:
!>           a) los elementos en y por encima de la diagonal
!>              de la matriz contienen la matriz triangular superior
!>              N por N, R, correspondiente a la QR de Householder;
!>           b) los elementos debajo de la diagonal representan Q mediante
!>              las columnas de V bloqueada (representación WY compacta).
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de la matriz A. LDA >= max(1,M).
!>

T

!>          T es array COMPLEX, dimensión (LDT,N))
!>          Los reflectores de bloque triangulares superiores almacenados en forma compacta
!>          como una secuencia de bloques triangulares superiores.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal de la matriz T. LDT >= NB2.
!>

WORK

!>          (espacio de trabajo) array COMPLEX, dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión de la matriz WORK.
!>          Si MIN(M,N) = 0, LWORK >= 1, de lo contrario
!>          LWORK >= MAX( 1, LWT + LW1, MAX( LWT+N*N+LW2, LWT+N*N+N ) ),
!>          donde
!>             NUM_ALL_ROW_BLOCKS = CEIL((M-N)/(MB1-N)),
!>             NB1LOCAL = MIN(NB1,N).
!>             LWT = NUM_ALL_ROW_BLOCKS * N * NB1LOCAL,
!>             LW1 = NB1LOCAL * N,
!>             LW2 = NB1LOCAL * MAX( NB1LOCAL, ( N - NB1LOCAL ) ).
!>
!>          Si LWORK = -1, se asume una consulta de espacio de trabajo.
!>          La rutina solo calcula el tamaño óptimo de la matriz WORK,
!>          devuelve este valor como la primera entrada de la matriz WORK,
!>          y XERBLA no emite ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
