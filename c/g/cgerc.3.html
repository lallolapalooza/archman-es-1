Documentación de función/subrutina

subroutine cgerc (integer m, integer n, complex alpha, complex, dimension(*) x, integer incx, complex, dimension(*) y, integer incy, complex, dimension(lda,*) a, integer lda)

CGERC

Propósito:

!>
!> CGERC  realiza la operación de rango 1
!>
!>    A := alpha*x*y**H + A,
!>
!> donde alpha es un escalar, x es un vector de m elementos, y es un vector de n elementos
!> y A es una matriz de m por n.
!>

Parámetros

M 
!>          M es INTEGER
!>           Al entrar, M especifica el número de filas de la matriz A.
!>           M debe ser al menos cero.
!>

N

!>          N es INTEGER
!>           Al entrar, N especifica el número de columnas de la matriz A.
!>           N debe ser al menos cero.
!>

ALPHA

!>          ALPHA es COMPLEX
!>           Al entrar, ALPHA especifica el escalar alpha.
!>

X

!>          X es un array COMPLEX, de dimensión al menos
!>           ( 1 + ( m - 1 )*abs( INCX ) ).
!>           Antes de la entrada, el array X incrementado debe contener el vector x
!>           de m elementos.
!>

INCX

!>          INCX es INTEGER
!>           Al entrar, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>

Y

!>          Y es un array COMPLEX, de dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Antes de la entrada, el array Y incrementado debe contener el vector y
!>           de n elementos.
!>

INCY

!>          INCY es INTEGER
!>           Al entrar, INCY especifica el incremento para los elementos de
!>           Y. INCY no debe ser cero.
!>

A

!>          A es un array COMPLEX, de dimensión ( LDA, N )
!>           Antes de la entrada, la parte principal de m por n del array A debe
!>           contener la matriz de coeficientes. A la salida, A es
!>           sobrescrita por la matriz actualizada.
!>

LDA

!>          LDA es INTEGER
!>           Al entrar, LDA especifica la primera dimensión de A tal como está declarada
!>           en el programa (sub)llamador. LDA debe ser al menos
!>           max( 1, m ).
!>
