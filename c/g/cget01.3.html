DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine cget01 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( * ) rwork, real resid)

CGET01

Propósito:

!>
!> CGET01 reconstruye una matriz A a partir de su factorización L*U y
!> calcula el residuo
!>    norm(L*U - A) / ( N * norm(A) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX, de dimensión (LDA,N)
!>          La matriz original M x N, A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

AFAC

!>          AFAC es un array COMPLEX, de dimensión (LDAFAC,N)
!>          La forma factorizada de la matriz A. AFAC contiene los factores
!>          L y U de la factorización L*U calculada por CGETRF.
!>          Se sobrescribe con la matriz reconstruida y luego con la
!>          diferencia L*U - A.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC. LDAFAC >= max(1,M).
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>          Los índices de pivote de CGETRF.
!>

RWORK

!>          RWORK es un array REAL, de dimensión (M)
!>

RESID

!>          RESID es REAL
!>          norm(L*U - A) / ( N * norm(A) * EPS )
!>
