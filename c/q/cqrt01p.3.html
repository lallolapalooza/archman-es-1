subroutine cqrt01p (integer m, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) r, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CQRT01P

Propósito:

!>
!> CQRT01P prueba CGEQRFP, que calcula la factorización QR de una matriz m-por-n
!> A, y prueba parcialmente CUNGQR que forma la matriz ortogonal m-por-m
!> Q.
!>
!> CQRT01P compara R con Q'*A y verifica que Q es ortogonal.
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es ENTERO
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es una matriz COMPLEJA, de dimensión (LDA,N)
!>          La matriz A de m-por-n.
!>

AF

!>          AF es una matriz COMPLEJA, de dimensión (LDA,N)
!>          Detalles de la factorización QR de A, tal como la devuelve CGEQRFP.
!>          Ver CGEQRFP para más detalles.
!>

Q

!>          Q es una matriz COMPLEJA, de dimensión (LDA,M)
!>          La matriz ortogonal m-por-m Q.
!>

R

!>          R es una matriz COMPLEJA, de dimensión (LDA,max(M,N))
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal de las matrices A, AF, Q y R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU es una matriz COMPLEJA, de dimensión (min(M,N))
!>          Los factores escalares de los reflectores elementales, tal como los
!>          devuelve CGEQRFP.
!>

WORK

!>          WORK es una matriz COMPLEJA, de dimensión (LWORK)
!>

LWORK

!>          LWORK es ENTERO
!>          La dimensión de la matriz WORK.
!>

RWORK

!>          RWORK es una matriz REAL, de dimensión (M)
!>

RESULT

!>          RESULT es una matriz REAL, de dimensión (2)
!>          Las razones de prueba:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
