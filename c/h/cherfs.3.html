subroutine cherfs (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHERFS

Propósito:

!>
!> CHERFS mejora la solución calculada de un sistema de ecuaciones lineales
!> cuando la matriz de coeficientes es Hermítica indefinida, y
!> proporciona cotas de error y estimaciones de error hacia atrás para la solución.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices B y X.  NRHS >= 0.
!>

A

!>          A es un array COMPLEX, dimensión (LDA,N)
!>          La matriz Hermítica A.  Si UPLO = 'U', la parte superior
!>          triangular N-por-N de A contiene la parte superior
!>          triangular de la matriz A, y la parte triangular estrictamente
!>          inferior de A no se referencia. Si UPLO = 'L', la parte inferior
!>          triangular N-por-N de A contiene la parte inferior
!>          triangular de la matriz A, y la parte triangular estrictamente
!>          superior de A no se referencia.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

AF

!>          AF es un array COMPLEX, dimensión (LDAF,N)
!>          La forma factorizada de la matriz A.  AF contiene la matriz
!>          diagonal por bloques D y los multiplicadores utilizados para obtener
!>          U o L de la factorización A = U*D*U**H o
!>          A = L*D*L**H calculada por CHETRF.
!>

LDAF

!>          LDAF es INTEGER
!>          La dimensión principal del array AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          tal como fueron determinados por CHETRF.
!>

B

!>          B es un array COMPLEX, dimensión (LDB,NRHS)
!>          La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es un array COMPLEX, dimensión (LDX,NRHS)
!>          En la entrada, la matriz solución X, tal como fue calculada por CHETRS.
!>          En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

FERR

!>          FERR es un array REAL, dimensión (NRHS)
!>          La cota de error hacia adelante estimada para cada vector solución
!>          X(j) (la j-ésima columna de la matriz solución X).
!>          Si XTRUE es la solución verdadera correspondiente a X(j), FERR(j)
!>          es una cota superior estimada para la magnitud del elemento más grande
!>          en (X(j) - XTRUE) dividido por la magnitud del elemento más grande
!>          en X(j). La estimación es tan confiable como
!>          la estimación para RCOND, y es casi siempre una ligera
!>          sobreestimación del error verdadero.
!>

BERR

!>          BERR es un array REAL, dimensión (NRHS)
!>          El error hacia atrás relativo componente a componente de cada vector solución
!>          X(j) (es decir, el cambio relativo más pequeño en
!>          cualquier elemento de A o B que hace que X(j) sea una solución exacta).
!>

WORK

!>          WORK es un array COMPLEX, dimensión (2*N)
!>

RWORK

!>          RWORK es un array REAL, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Parámetros Internos:

!>  ITMAX es el número máximo de pasos de refinamiento iterativo.
!>
