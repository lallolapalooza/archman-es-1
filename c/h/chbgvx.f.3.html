subroutine chbgvx (character jobz, character range, character uplo, integer n, integer ka, integer kb, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldbb, * ) bb, integer ldbb, complex, dimension( ldq, * ) q, integer ldq, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

CHBGVX

Propósito:

!>
!> CHBGVX calcula todos los valores propios, y opcionalmente, los vectores propios
!> de un problema de autovalores hermitiano-definido general complejo y
banda,
!> de la forma A*x=(lambda)*B*x. Aquí A y B se asumen Hermitianas
!> y de banda, y B es también positiva definida. Los valores propios y
!> los vectores propios pueden ser seleccionados especificando todos los
!> valores propios, un rango de valores o un rango de índices para los
!> valores propios deseados.
!>

Parámetros

JOBZ 
!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcula solo los valores propios;
!>          = 'V':  Calcula los valores propios y vectores propios.
!>

RANGE

!>          RANGE es CHARACTER*1
!>          = 'A': se encontrarán todos los valores propios;
!>          = 'V': se encontrarán todos los valores propios en el intervalo semiabierto (VL,VU]
!>                 que se desea;
!>          = 'I': se encontrarán los valores propios desde el IL-ésimo hasta el IU-ésimo.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacenan los triángulos superiores de A y B;
!>          = 'L':  Se almacenan los triángulos inferiores de A y B.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B.  N >= 0.
!>

KA

!>          KA es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'. KA >= 0.
!>

KB

!>          KB es INTEGER
!>          El número de superdiagonales de la matriz B si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'. KB >= 0.
!>

AB

!>          AB es un array COMPLEX de dimensión (LDAB, N)
!>          En la entrada, el triángulo superior o inferior de la matriz hermitiana
!>          de banda A, almacenado en las primeras ka+1 filas del array. El
!>          j-ésima columna de A se almacena en la j-ésima columna del array AB
!>          de la siguiente manera:
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) para max(1,j-ka)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+ka).
!>
!>          En la salida, el contenido de AB se destruye.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KA+1.
!>

BB

!>          BB es un array COMPLEX de dimensión (LDBB, N)
!>          En la entrada, el triángulo superior o inferior de la matriz hermitiana
!>          de banda B, almacenado en las primeras kb+1 filas del array. El
!>          j-ésima columna de B se almacena en la j-ésima columna del array BB
!>          de la siguiente manera:
!>          si UPLO = 'U', BB(kb+1+i-j,j) = B(i,j) para max(1,j-kb)<=i<=j;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) para j<=i<=min(n,j+kb).
!>
!>          En la salida, el factor S de la factorización Cholesky dividida
!>          B = S**H*S, tal como devuelve CPBSTF.
!>

LDBB

!>          LDBB es INTEGER
!>          La dimensión principal del array BB.  LDBB >= KB+1.
!>

Q

!>          Q es un array COMPLEX de dimensión (LDQ, N)
!>          Si JOBZ = 'V', la matriz n x n utilizada en la reducción de
!>          A*x = (lambda)*B*x a la forma estándar, es decir, C*x = (lambda)*x,
!>          y consecuentemente C a forma tridiagonal.
!>          Si JOBZ = 'N', el array Q no se utiliza.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q.  Si JOBZ = 'N',
!>          LDQ >= 1. Si JOBZ = 'V', LDQ >= max(1,N).
!>

VL

!>          VL es REAL
!>
!>          Si RANGE='V', el límite inferior del intervalo a
!>          buscar para los valores propios. VL < VU.
!>          No se utiliza si RANGE = 'A' o 'I'.
!>

VU

!>          VU es REAL
!>
!>          Si RANGE='V', el límite superior del intervalo a
!>          buscar para los valores propios. VL < VU.
!>          No se utiliza si RANGE = 'A' o 'I'.
!>

IL

!>          IL es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más pequeño a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No se utiliza si RANGE = 'A' o 'V'.
!>

IU

!>          IU es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más grande a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No se utiliza si RANGE = 'A' o 'V'.
!>

ABSTOL

!>          ABSTOL es REAL
!>          La tolerancia de error absoluta para los valores propios.
!>          Un valor propio aproximado se acepta como convergido
!>          cuando se determina que yace en un intervalo [a,b]
!>          de anchura menor o igual a
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          donde EPS es la precisión de la máquina. Si ABSTOL es menor o
!>          igual a cero, entonces EPS*|T| se usará en su lugar,
!>          donde |T| es la norma 1 de la matriz tridiagonal obtenida
!>          al reducir AP a forma tridiagonal.
!>
!>          Los valores propios se calcularán con mayor precisión cuando ABSTOL
!>          se establezca en el doble del umbral de subdesbordamiento 2*SLAMCH('S'),
!!>          no en cero. Si esta rutina regresa con INFO>0, indicando que algunos
!>          vectores propios no convergieron, intente establecer ABSTOL en
!>          2*SLAMCH('S').
!>

M

!>          M es INTEGER
!>          El número total de valores propios encontrados.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, y si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W es un array REAL de dimensión (N)
!>          Si INFO = 0, los valores propios en orden ascendente.
!>

Z

!>          Z es un array COMPLEX de dimensión (LDZ, N)
!>          Si JOBZ = 'V', entonces si INFO = 0, Z contiene la matriz Z de
!>          vectores propios, con la i-ésima columna de Z conteniendo el
!>          vector propio asociado con W(i). Los vectores propios están
!>          normalizados de tal forma que Z**H*B*Z = I.
!>          Si JOBZ = 'N', entonces Z no se utiliza.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= N.
!>

WORK

!>          WORK es un array COMPLEX de dimensión (N)
!>

RWORK

!>          RWORK es un array REAL de dimensión (7*N)
!>

IWORK

!>          IWORK es un array INTEGER de dimensión (5*N)
!>

IFAIl

!>          IFAIL es un array INTEGER de dimensión (N)
!>          En la salida, el k-ésimo elemento de IFAIL es
!>          un índice de error relativo al k-ésimo vector propio
!>          y valor propio. Si IFAIL(k) = 0, el vector propio
!>          correspondiente a W(k) convergió. Si IFAIL(k) > 0, el vector
!>          propio correspondiente a W(k) no convergió, y el
!>          resultado devuelto es un vector de ceros.
!>          Si IFAIL(k) < 0, el valor propio se encontró pero el vector
!>          propio no se calculó.
!>

INFO

!>          INFO es INTEGER
!>          INFO = 0:  éxito en la ejecución.
!>          INFO < 0:  si INFO = -i, la i-ésima argumento hacia abajo
!>                    en la llamada fue inválido.
!>          INFO > 0:  si INFO = i, entonces i
!>                    vectores propios no convergieron. Los resultados
!>                    no son fiables.
!>

IFAIL

!>          IFAIL es un arreglo de tipo INTEGER, dimensión (N)
!>          Si JOBZ = 'V', entonces si INFO = 0, los primeros M elementos de
!>          IFAIL son cero.  Si INFO > 0, entonces IFAIL contiene los
!>          índices de los autovectores que no lograron converger.
!>          Si JOBZ = 'N', entonces IFAIL no es referenciado.
!>

INFO

!>          INFO es de tipo INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, y i es:
!>             <= N:  entonces i autovectores no lograron converger. Sus
!>                    índices se almacenan en el arreglo IFAIL.
!>             > N:   si INFO = N + i, para 1 <= i <= N, entonces CPBSTF
!>                    devolvió INFO = i: B no es definida positiva.
!>                    La factorización de B no pudo completarse y
!>                    no se calcularon autovalores ni autovectores.
!>
