subroutine chetrs_aa (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CHETRS_AA

Propósito:

!>
!> CHETRS_AA resuelve un sistema de ecuaciones lineales A*X = B con una matriz
!> compleja hermitiana A usando la factorización A = U**H*T*U o
!> A = L*T*L**H calculada por CHETRF_AA.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, la forma es A = U**H*T*U;
!>          = 'L':  Triangular inferior, la forma es A = L*T*L**H.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es un array COMPLEX, de dimensión (LDA,N)
!>          Detalles de los factores calculados por CHETRF_AA.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>          Detalles de los intercambios calculados por CHETRF_AA.
!>

B

!>          B es un array COMPLEX, de dimensión (LDB,NRHS)
!>          En la entrada, la matriz del lado derecho B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

WORK

!>          WORK es un array COMPLEX, de dimensión (MAX(1,LWORK))
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          Si MIN(N,NRHS) = 0, LWORK >= 1, de lo contrario LWORK >= 3*N-2.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo;
!>          la rutina solo calcula el tamaño mínimo del array WORK,
!>          devuelve este valor como la primera entrada del array WORK y
!>          XERBLA no emite ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el argumento i tenía un valor ilegal
!>
