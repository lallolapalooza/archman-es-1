subroutine chetri2 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CHETRI2

Propósito:

!>
!> CHETRI2 calcula la inversa de una matriz hermítica indefinida COMPLEJA
!> A usando la factorización A = U*D*U**T o A = L*D*L**T calculada por
!> CHETRF. CHETRI2 establece la DIMENSIÓN PRINCIPAL del espacio de trabajo
!> antes de llamar a CHETRI2X que realmente calcula la inversa.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, forma es A = U*D*U**T;
!>          = 'L':  Triangular inferior, forma es A = L*D*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEJO, dimensión (LDA,N)
!>          En la entrada, la matriz diagonal por bloques D y los multiplicadores
!>          utilizados para obtener el factor U o L calculados por CHETRF.
!>
!>          En la salida, si INFO = 0, la inversa (simétrica) de la matriz original.
!>          Si UPLO = 'U', se forma la parte triangular superior de la inversa
!>          y la parte de A debajo de la diagonal no se referencia; si UPLO = 'L'
!>          se forma la parte triangular inferior de la inversa y la parte de A
!>          encima de la diagonal no se referencia.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          tal como los determina CHETRF.
!>

WORK

!>          WORK es un array COMPLEJO, dimensión (MAX(1,LWORK))
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          Si N = 0, LWORK >= 1, de lo contrario LWORK >= (N+NB+1)*(NB+3).
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          calcula:
!>              - el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK,
!>              - y no se emite ningún mensaje de error relacionado con LWORK por XERBLA.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0: si INFO = i, D(i,i) = 0; la matriz es singular y su
!>               inversa no pudo calcularse.
!>
