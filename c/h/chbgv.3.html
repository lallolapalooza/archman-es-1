subroutine chbgv (character jobz, character uplo, integer n, integer ka, integer kb, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldbb, * ) bb, integer ldbb, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHBGV

Propósito:

!>
!> CHBGV calcula todos los valores propios, y opcionalmente, los vectores propios
!> de un problema de valores propios generalizados Hermítico-definidos complejo
!> de banda, de la forma A*x=(lambda)*B*x. Aquí A y B se suponen Hermíticas
!> y de banda, y B es además positiva definida.
!>

Parámetros

JOBZ 
!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcular sólo los valores propios;
!>          = 'V':  Calcular valores propios y vectores propios.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacenan las partes triangulares superior de A y B;
!>          = 'L':  Se almacenan las partes triangulares inferior de A y B.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B.  N >= 0.
!>

KA

!>          KA es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'. KA >= 0.
!>

KB

!>          KB es INTEGER
!>          El número de superdiagonales de la matriz B si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'. KB >= 0.
!>

AB

!>          AB es un array de tipo COMPLEX, de dimensión (LDAB, N)
!>          En la entrada, la parte triangular superior o inferior de la matriz
!>          de banda Hermítica A, almacenada en las primeras ka+1 filas del
!>          array. La j-ésima columna de A se almacena en la j-ésima columna
!>          del array AB de la siguiente manera:
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) para max(1,j-ka) <= i <= j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j <= i <= min(n,j+ka).
!>
!>          En la salida, el contenido de AB se destruye.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KA+1.
!>

BB

!>          BB es un array de tipo COMPLEX, de dimensión (LDBB, N)
!>          En la entrada, la parte triangular superior o inferior de la matriz
!>          de banda Hermítica B, almacenada en las primeras kb+1 filas del
!>          array. La j-ésima columna de B se almacena en la j-ésima columna
!>          del array BB de la siguiente manera:
!>          si UPLO = 'U', BB(kb+1+i-j,j) = B(i,j) para max(1,j-kb) <= i <= j;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) para j <= i <= min(n,j+kb).
!>
!>          En la salida, el factor S de la factorización Cholesky dividida
!>          B = S**H*S, tal como devuelve CPBSTF.
!>

LDBB

!>          LDBB es INTEGER
!>          La dimensión principal del array BB.  LDBB >= KB+1.
!>

W

!>          W es un array de tipo REAL, de dimensión (N)
!>          Si INFO = 0, los valores propios en orden ascendente.
!>

Z

!>          Z es un array de tipo COMPLEX, de dimensión (LDZ, N)
!>          Si JOBZ = 'V', entonces si INFO = 0, Z contiene la matriz Z de
!>          vectores propios, con la i-ésima columna de Z conteniendo el
!>          vector propio asociado a W(i). Los vectores propios están normalizados
!>          tal que Z**H*B*Z = I.
!>          Si JOBZ = 'N', entonces Z no se referencia.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= N.
!>

WORK

!>          WORK es un array de tipo COMPLEX, de dimensión (N)
!>

RWORK

!>          RWORK es un array de tipo REAL, de dimensión (3*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, y i es:
!>             <= N:  el algoritmo falló en converger:
!>                    i elementos fuera de la diagonal de una forma
!>                    tridiagonal intermedia no convergieron a cero;
!>             > N:   si INFO = N + i, para 1 <= i <= N, entonces CPBSTF
!>                    devolvió INFO = i: B no es positiva definida.
!>                    La factorización de B no pudo completarse y
!>                    no se calcularon valores ni vectores propios.
!>
