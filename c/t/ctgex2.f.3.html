subroutine ctgex2 (logical wantq, logical wantz, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( ldz, * ) z, integer ldz, integer j1, integer info)

CTGEX2 intercambia bloques diagonales adyacentes en un par de matrices (quasi)triangulares superiores mediante una transformación de equivalencia unitaria.

Propósito:

!>
!> CTGEX2 intercambia bloques diagonales adyacentes de 1x1 (A11,B11) y (A22,B22)
!> en un par de matrices triangulares superiores (A, B) mediante una transformación de equivalencia unitaria.
!>
!> (A, B) deben estar en forma canónica de Schur generalizada, es decir, A y
!> B son ambas triangulares superiores.
!>
!> Opcionalmente, se actualizan las matrices Q y Z de vectores de Schur generalizados.
!>
!>        Q(in) * A(in) * Z(in)**H = Q(out) * A(out) * Z(out)**H
!>        Q(in) * B(in) * Z(in)**H = Q(out) * B(out) * Z(out)**H
!>
!>

Parámetros

WANTQ 
!>          WANTQ es LOGICAL
!>          .TRUE. : actualiza la matriz de transformación izquierda Q;
!>          .FALSE.: no actualiza Q.
!>

WANTZ

!>          WANTZ es LOGICAL
!>          .TRUE. : actualiza la matriz de transformación derecha Z;
!>          .FALSE.: no actualiza Z.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B. N >= 0.
!>

A

!>          A es un array COMPLEX, de dimensión (LDA,N)
!>          En la entrada, la matriz A en el par (A, B).
!>          En la salida, la matriz A actualizada.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,N).
!>

B

!>          B es un array COMPLEX, de dimensión (LDB,N)
!>          En la entrada, la matriz B en el par (A, B).
!>          En la salida, la matriz B actualizada.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,N).
!>

Q

!>          Q es un array COMPLEX, de dimensión (LDQ,N)
!>          Si WANTQ = .TRUE, en la entrada, la matriz unitaria Q. En la salida,
!>          la matriz Q actualizada.
!>          No se hace referencia si WANTQ = .FALSE..
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q. LDQ >= 1;
!>          Si WANTQ = .TRUE., LDQ >= N.
!>

Z

!>          Z es un array COMPLEX, de dimensión (LDZ,N)
!>          Si WANTZ = .TRUE, en la entrada, la matriz unitaria Z. En la salida,
!>          la matriz Z actualizada.
!>          No se hace referencia si WANTZ = .FALSE..
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z. LDZ >= 1;
!>          Si WANTZ = .TRUE., LDZ >= N.
!>

J1

!>          J1 es INTEGER
!>          El índice del primer bloque (A11, B11).
!>

INFO

!>          INFO es INTEGER
!>           =0:  Salida exitosa.
!>           =1:  El par de matrices transformado (A, B) estaría demasiado alejado
!>                de la forma de Schur generalizada; el problema está mal
!>                condicionado.
!>
