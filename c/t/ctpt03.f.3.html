subroutine ctpt03 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) ap, real scale, real, dimension( * ) cnorm, real tscal, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real resid)

CTPT03

Propósito:

!>
!> CTPT03 calcula el residuo de la solución a un sistema triangular escalado
!> de ecuaciones A*x = s*b,  A**T *x = s*b,  o  A**H *x = s*b,
!> cuando la matriz triangular A se almacena en formato empaquetado. Aquí A**T
!> denota la transpuesta de A, A**H denota la transpuesta conjugada de
!> A, s es un escalar, y x e b son matrices N por NRHS. La razón de prueba
!> es el máximo sobre el número de lados derechos de
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> donde op(A) denota A, A**T, o A**H, y EPS es el épsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica la operación aplicada a A.
!>          = 'N':  A *x = s*b     (Sin transpuesta)
!>          = 'T':  A**T *x = s*b  (Transpuesta)
!>          = 'C':  A**H *x = s*b  (Transpuesta conjugada)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es triangular unitaria o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices X y B.  NRHS >= 0.
!>

AP

!>          AP es array COMPLEX, dimensión (N*(N+1)/2)
!>          La matriz triangular superior o inferior A, empaquetada por
!>          columnas en un arreglo lineal. La j-ésima columna de A se
!>          almacena en el arreglo AP de la siguiente manera:
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) para j<=i<=n.
!>

SCALE

!>          SCALE es REAL
!>          El factor de escala s utilizado al resolver el sistema
!>          triangular.
!>

CNORM

!>          CNORM es array REAL, dimensión (N)
!>          Las normas 1 de las columnas de A, sin contar la diagonal.
!>

TSCAL

!>          TSCAL es REAL
!>          El factor de escala utilizado al calcular las normas 1 en CNORM.
!>          CNORM contiene en realidad las normas de columna de TSCAL*A.
!>

X

!>          X es array COMPLEX, dimensión (LDX,NRHS)
!>          Los vectores solución calculados para el sistema de ecuaciones
!>          lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del arreglo X.  LDX >= max(1,N).
!>

B

!>          B es array COMPLEX, dimensión (LDB,NRHS)
!>          Los vectores del lado derecho para el sistema de ecuaciones
!>          lineales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del arreglo B.  LDB >= max(1,N).
!>

WORK

!>          WORK es array COMPLEX, dimensión (N)
!>

RESID

!>          RESID es REAL
!>          El máximo sobre el número de lados derechos de
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
