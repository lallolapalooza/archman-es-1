Documentación de función/subrutina

subroutine ctrt05 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

CTRT05

Propósito:

!>
!> CTRT05 prueba los límites de error del refinamiento iterativo para la
!> solución calculada de un sistema de ecuaciones A*X = B, donde A es una
!> matriz triangular n por n.
!>
!> RESLTS(1) = prueba del límite de error
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> Se devuelve un valor grande si esta relación no es menor que uno.
!>
!> RESLTS(2) = residuo de la rutina de refinamiento iterativo
!>           = el máximo de BERR / ( (n+1)*EPS + (*) ), donde
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones.
!>          = 'N':  A * X = B  (Sin transposición)
!>          = 'T':  A'* X = B  (Transpuesta)
!>          = 'C':  A'* X = B  (Transpuesta conjugada = Transpuesta)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es triangular unitaria o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El número de filas de las matrices X, B y XACT, y el orden
!>          de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de las matrices X, B y XACT.
!>          NRHS >= 0.
!>

A

!>          A es un array COMPLEX, dimension (LDA,N)
!>          La matriz triangular A.  Si UPLO = 'U', la parte superior
!>          triangular principal de n por n del array A contiene la
!>          matriz triangular superior, y la parte estrictamente
!>          triangular inferior de A no se referencia.  Si UPLO = 'L',
!>          la parte inferior triangular principal de n por n del array
!>          A contiene la matriz triangular inferior, y la parte
!>          estrictamente triangular superior de A no se referencia.  Si
!>          DIAG = 'U', los elementos diagonales de A tampoco se
!>          referencian y se asumen que son 1.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

B

!>          B es un array COMPLEX, dimension (LDB,NRHS)
!>          Los vectores del lado derecho para el sistema de ecuaciones
!>          lineales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es un array COMPLEX, dimension (LDX,NRHS)
!>          Los vectores de solución calculados. Cada vector se almacena
!>          como una columna de la matriz X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

XACT

!>          XACT es un array COMPLEX, dimension (LDXACT,NRHS)
!>          Los vectores de solución exactos. Cada vector se almacena como
!>          una columna de la matriz XACT.
!>

LDXACT

!>          LDXACT es INTEGER
!>          La dimensión principal del array XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR es un array REAL, dimension (NRHS)
!>          Los límites de error de avance estimados para cada vector de
!>          solución X. Si XTRUE es la solución verdadera, FERR limita la
!>          magnitud de la entrada más grande en (X - XTRUE) dividida
!>          por la magnitud de la entrada más grande en X.
!>

BERR

!>          BERR es un array REAL, dimension (NRHS)
!>          El error residual componente a componente de cada vector de
!>          solución (es decir, el cambio relativo más pequeño en cualquier
!>          entrada de A o B que hace que X sea una solución exacta).
!>

RESLTS

!>          RESLTS es un array REAL, dimension (2)
!>          El máximo sobre los vectores de solución NRHS de las relaciones:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
