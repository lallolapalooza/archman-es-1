subroutine ctbsv (character uplo, character trans, character diag, integer n, integer k, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx)

CTBSV

Propósito:

!>
!> CTBSV resuelve uno de los sistemas de ecuaciones
!>
!>    A*x = b,   o   A**T*x = b,   o   A**H*x = b,
!>
!> donde b y x son vectores de n elementos y A es una matriz de banda
!> triangular superior o inferior unitaria o no unitaria de n por n,
!> con ( k + 1 ) diagonales.
!>
!> Esta rutina no incluye pruebas de singularidad o casi singularidad.
!> Dichas pruebas deben realizarse antes de llamar a esta rutina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>           En la entrada, UPLO especifica si la matriz es superior o
!>           inferior de la siguiente manera:
!>
!>              UPLO = 'U' o 'u'   A es una matriz triangular superior.
!>
!>              UPLO = 'L' o 'l'   A es una matriz triangular inferior.
!>

TRANS

!>          TRANS es CHARACTER*1
!>           En la entrada, TRANS especifica las ecuaciones a resolver de la
!>           siguiente manera:
!>
!>              TRANS = 'N' o 'n'   A*x = b.
!>
!>              TRANS = 'T' o 't'   A**T*x = b.
!>
!>              TRANS = 'C' o 'c'   A**H*x = b.
!>

DIAG

!>          DIAG es CHARACTER*1
!>           En la entrada, DIAG especifica si A es unitaria o no
!>           triangular de la siguiente manera:
!>
!>              DIAG = 'U' o 'u'   Se asume que A es unitaria triangular.
!>
!>              DIAG = 'N' o 'n'   No se asume que A es unitaria
!>                                  triangular.
!>

N

!>          N es INTEGER
!>           En la entrada, N especifica el orden de la matriz A.
!>           N debe ser al menos cero.
!>

K

!>          K es INTEGER
!>           En la entrada con UPLO = 'U' o 'u', K especifica el número de
!>           superdiagonales de la matriz A.
!>           En la entrada con UPLO = 'L' o 'l', K especifica el número de
!>           subdiagonales de la matriz A.
!>           K debe satisfacer  0 .le. K.
!>

A

!>          A es un array COMPLEX, dimensión ( LDA, N )
!>           Antes de la entrada con UPLO = 'U' o 'u', la parte principal
!>           ( k + 1 ) por n del array A debe contener la parte de banda
!>           triangular superior de la matriz de coeficientes, suministrada
!>           columna por columna, con la diagonal principal de la matriz en
!>           la fila ( k + 1 ) del array, la primera superdiagonal comenzando
!>           en la posición 2 en la fila k, y así sucesivamente. El triángulo
!>           superior izquierdo k por k del array A no se referencia.
!>           El siguiente segmento de programa transferirá una matriz de
!>           banda triangular superior desde el almacenamiento convencional de
!>           matrices completas a almacenamiento de banda:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Antes de la entrada con UPLO = 'L' o 'l', la parte principal
!>           ( k + 1 ) por n del array A debe contener la parte de banda
!>           triangular inferior de la matriz de coeficientes, suministrada
!>           columna por columna, con la diagonal principal de la matriz en
!>           la fila 1 del array, la primera subdiagonal comenzando en la
!>           posición 1 en la fila 2, y así sucesivamente. El triángulo
!>           inferior derecho k por k del array A no se referencia.
!>           El siguiente segmento de programa transferirá una matriz de
!>           banda triangular inferior desde el almacenamiento convencional de
!>           matrices completas a almacenamiento de banda:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Tenga en cuenta que cuando DIAG = 'U' o 'u', los elementos del
!>           array A correspondientes a los elementos diagonales de la matriz
!>           no se referencian, pero se asumen como unidad.
!>

LDA

!>          LDA es INTEGER
!>           En la entrada, LDA especifica la primera dimensión de A tal
!>           como se declara en el programa (sub)llamador. LDA debe ser al
!>           menos ( k + 1 ).
!>

X

!>          X es un array COMPLEX, dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Antes de la entrada, el array X incrementado debe contener el
!>           vector b del lado derecho de n elementos. En la salida, X se
!>           sobrescribe con el vector solución x.
!>

INCX

!>          INCX es INTEGER
!>           En la entrada, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>
