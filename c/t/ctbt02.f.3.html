subroutine ctbt02 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CTBT02

Propósito:

!>
!> CTBT02 calcula el residuo para la solución calculada de un sistema
!> de ecuaciones lineales triangulares op(A)*X = B, cuando A es una
!> matriz triangular de banda. La relación de prueba es el máximo sobre
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS ),
!> donde op(A) = A, A**T, o A**H, b es la columna de B, x es el
!> vector solución, y EPS es el épsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica la operación aplicada a A.
!>          = 'N':  A    * X = B  (Sin transposición)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitriangular o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales o subdiagonales de la
!>          matriz triangular de banda A.  KD >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices X e B.  NRHS >= 0.
!>

AB

!>          AB es un array COMPLEX, dimension (LDA,N)
!>          La matriz triangular de banda superior o inferior A, almacenada
!>          en las primeras kd+1 filas del array. La columna j de A se
!>          almacena en la columna j de la matriz AB de la siguiente
!>          manera:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= max(1,KD+1).
!>

X

!>          X es un array COMPLEX, dimension (LDX,NRHS)
!>          Los vectores solución calculados para el sistema de ecuaciones
!>          lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

B

!>          B es un array COMPLEX, dimension (LDB,NRHS)
!>          Los vectores del lado derecho para el sistema de ecuaciones
!>          lineales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

WORK

!>          WORK es un array COMPLEX, dimension (N)
!>

RWORK

!>          RWORK es un array REAL, dimension (N)
!>

RESID

!>          RESID es REAL
!>          El máximo sobre el número de lados derechos de
!>          norm(op(A)*x - b) / ( norm(op(A)) * norm(x) * EPS ).
!>
