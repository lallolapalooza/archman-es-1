subroutine ctprfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldwork, * ) work, integer ldwork)

CTPRFB aplica un reflector de bloque complejo "triangular-pentagonal" a una matriz compleja, que se compone de dos bloques.

Propósito:

!>
!> CTPRFB aplica un reflector de bloque complejo H o su
!> transpuesta conjugada H**H a una matriz compleja C, que se compone de dos
!> bloques A y B, ya sea desde la izquierda o desde la derecha.
!>
!>

Parámetros

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': aplicar H o H**H desde la Izquierda
!>          = 'R': aplicar H o H**H desde la Derecha
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N': aplicar H (Sin transponer)
!>          = 'C': aplicar H**H (Transpuesta conjugada)
!>

DIRECT

!>          DIRECT is CHARACTER*1
!>          Indica cómo se forma H a partir de un producto de reflectores
!>          elementales
!>          = 'F': H = H(1) H(2) . . . H(k) (Adelante)
!>          = 'B': H = H(k) . . . H(2) H(1) (Atrás)
!>

STOREV

!>          STOREV is CHARACTER*1
!>          Indica cómo se almacenan los vectores que definen los
!>          reflectores elementales:
!>          = 'C': Columnas
!>          = 'R': Filas
!>

M

!>          M is INTEGER
!>          El número de filas de la matriz B.
!>          M >= 0.
!>

N

!>          N is INTEGER
!>          El número de columnas de la matriz B.
!>          N >= 0.
!>

K

!>          K is INTEGER
!>          El orden de la matriz T, es decir, el número de reflectores
!>          elementales cuyo producto define el reflector de bloque.
!>          K >= 0.
!>

L

!>          L is INTEGER
!>          El orden de la parte trapezoidal de V.
!>          K >= L >= 0. Ver Detalles Adicionales.
!>

V

!>          V is COMPLEX array, dimension
!>                                (LDV,K) if STOREV = 'C'
!>                                (LDV,M) if STOREV = 'R' and SIDE = 'L'
!>                                (LDV,N) if STOREV = 'R' and SIDE = 'R'
!>          La matriz pentagonal V, que contiene los reflectores elementales
!>          H(1), H(2), ..., H(K). Ver Detalles Adicionales.
!>

LDV

!>          LDV is INTEGER
!>          La dimensión principal del array V.
!>          If STOREV = 'C' and SIDE = 'L', LDV >= max(1,M);
!>          if STOREV = 'C' and SIDE = 'R', LDV >= max(1,N);
!>          if STOREV = 'R', LDV >= K.
!>

T

!>          T is COMPLEX array, dimension (LDT,K)
!>          La matriz triangular K por K, T, en la representación del
!>          reflector de bloque.
!>

LDT

!>          LDT is INTEGER
!>          La dimensión principal del array T.
!>          LDT >= K.
!>

A

!>          A is COMPLEX array, dimension
!>          (LDA,N) if SIDE = 'L' or (LDA,K) if SIDE = 'R'
!>          En la entrada, la matriz A de K por N o M por K.
!>          En la salida, A se sobrescribe con el bloque correspondiente de
!>          H*C o H**H*C o C*H o C*H**H. Ver Detalles Adicionales.
!>

LDA

!>          LDA is INTEGER
!>          La dimensión principal del array A.
!>          If SIDE = 'L', LDA >= max(1,K);
!>          If SIDE = 'R', LDA >= max(1,M).
!>

B

!>          B is COMPLEX array, dimension (LDB,N)
!>          En la entrada, la matriz B de M por N.
!>          En la salida, B se sobrescribe con el bloque correspondiente de
!>          H*C o H**H*C o C*H o C*H**H. Ver Detalles Adicionales.
!>

LDB

!>          LDB is INTEGER
!>          La dimensión principal del array B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK is COMPLEX array, dimension
!>          (LDWORK,N) if SIDE = 'L',
!>          (LDWORK,K) if SIDE = 'R'.
!>

LDWORK

!>          LDWORK is INTEGER
!>          La dimensión principal del array WORK.
!>          If SIDE = 'L', LDWORK >= K;
!>          if SIDE = 'R', LDWORK >= M.
!>

Detalles Adicionales:

!>
!>  La matriz C es una matriz compuesta formada a partir de los bloques A y B.
!>  El bloque B es de tamaño M por N; si SIDE = 'R', A es de tamaño M por K,
!>  y si SIDE = 'L', A es de tamaño K por N.
!>
!>  If SIDE = 'R' and DIRECT = 'F', C = [A B].
!>
!>  If SIDE = 'L' and DIRECT = 'F', C = [A]
!>                                      [B].
!>
!>  If SIDE = 'R' and DIRECT = 'B', C = [B A].
!>
!>  If SIDE = 'L' and DIRECT = 'B', C = [B]
!>                                      [A].
!>
!>  La matriz pentagonal V se compone de un bloque rectangular V1 y un
!>  bloque trapezoidal V2. El tamaño del bloque trapezoidal está determinado por
!>  el parámetro L, donde 0<=L<=K. Si L=K, el bloque V2 de V es triangular;
!>  si L=0, no hay bloque trapezoidal, por lo tanto V = V1 es rectangular.
!>
!>  If DIRECT = 'F' and STOREV = 'C':  V = [V1]
!>                                         [V2]
!>     - V2 es superior trapezoidal (primeras L filas de la triangular superior K-por-K)
!>
!>  If DIRECT = 'F' and STOREV = 'R':  V = [V1 V2]
!>
!>     - V2 es inferior trapezoidal (primeras L columnas de la triangular inferior K-por-K)
!>
!>  If DIRECT = 'B' and STOREV = 'C':  V = [V2]
!>                                         [V1]
!>     - V2 es inferior trapezoidal (últimas L filas de la triangular inferior K-por-K)
!>
!>  If DIRECT = 'B' and STOREV = 'R':  V = [V2 V1]
!>
!>     - V2 es superior trapezoidal (últimas L columnas de la triangular superior K-por-K)
!>
!>  If STOREV = 'C' and SIDE = 'L', V es M-por-K con V2 L-por-K.
!>
!>  If STOREV = 'C' and SIDE = 'R', V es N-por-K con V2 L-por-K.
!>
!>  If STOREV = 'R' and SIDE = 'L', V es K-por-M con V2 K-por-L.
!>
!>  If STOREV = 'R' and SIDE = 'R', V es K-por-N con V2 K-por-L.
!>
