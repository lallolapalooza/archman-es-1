subroutine ctzrzf (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CTZRZF

Propósito:

!>
!> CTZRZF reduce la matriz trapezoidal superior M por N ( M<=N ) compleja A
!> a forma triangular superior mediante transformaciones unitarias.
!>
!> La matriz trapezoidal superior A se factoriza como
!>
!>    A = ( R  0 ) * Z,
!>
!> donde Z es una matriz unitaria N por N y R es una matriz triangular
!> superior M por M.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= M.
!>

A

!>          A es array COMPLEXO, dimension (LDA,N)
!>          En la entrada, la parte trapezoidal superior M por N principal de
!>          la matriz A debe contener la matriz a factorizar.
!>          En la salida, la parte triangular superior M por M principal de A
!>          contiene la matriz triangular superior R, y los elementos M+1 a
!>          N de las primeras M filas de A, junto con la matriz TAU, representan
!>          la matriz unitaria Z como un producto de M reflectores elementales.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de la matriz A.  LDA >= max(1,M).
!>

TAU

!>          TAU es array COMPLEXO, dimension (M)
!>          Los factores escalares de los reflectores elementales.
!>

WORK

!>          WORK es array COMPLEXO, dimension (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) contiene el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión de la matriz WORK.  LWORK >= max(1,M).
!>          Para un rendimiento óptimo LWORK >= M*NB, donde NB es
!>          el tamaño de bloque óptimo.
!>
!>          Si LWORK = -1, se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo de la matriz WORK, devuelve
!>          este valor como la primera entrada de la matriz WORK, y no se emite
!>          ningún mensaje de error relacionado con LWORK por XERBLA.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Detalles Adicionales:

!>
!>  La matriz unitaria N por N Z se puede calcular como:
!>
!>     Z =  Z(1)*Z(2)* ... *Z(M)
!>
!>  donde cada Z(k) de N por N viene dada por
!>
!>     Z(k) = I - tau(k)*v(k)*v(k)**H
!>
!>  con v(k) siendo el k-ésimo vector fila de la matriz M por N
!>
!>     V = ( I   A(:,M+1:N) )
!>
!>  I es la matriz identidad M por M, A(:,M+1:N)
!>  es la salida almacenada en A tras la salida de CTZRZF,
!>  y tau(k) es el k-ésimo elemento de la matriz TAU.
!>
!>
