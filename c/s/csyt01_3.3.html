subroutine csyt01_3 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

CSYT01_3

Propósito:

!>
!> CSYT01_3 reconstruye una matriz indefinida simétrica A a partir de su
!> factorización de bloque L*D*L' o U*D*U' calculada por CSYTRF_RK
!> (o CSYTRF_BK) y calcula el residuo
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> donde C es la matriz reconstruida y EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX, de dimensión (LDA,N)
!>          La matriz simétrica original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC es un array COMPLEX, de dimensión (LDAFAC,N)
!>          Diagonal de la matriz de bloques diagonales D y factores U o L
!>          calculados por CSYTRF_RK y CSYTRF_BK:
!>            a) SÓLO los elementos diagonales de la matriz simétrica de bloques diagonales
!>               D en la diagonal de A, es decir, D(k,k) = A(k,k);
!>               (los elementos superdiagonales (o subdiagonales) de D
!>                deben proporcionarse en la entrada en el array E), y
!>            b) Si UPLO = 'U': factor U en la parte superdiagonal de A.
!>               Si UPLO = 'L': factor L en la parte subdiagonal de A.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E es un array COMPLEX, de dimensión (N)
!>          En la entrada, contiene los elementos superdiagonales (o subdiagonales)
!>          de la matriz de bloques diagonales simétrica D
!>          con bloques diagonales de 1x1 o 2x2, donde
!>          Si UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) no referenciado;
!>          Si UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) no referenciado.
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>          Los índices de pivote de CSYTRF_RK (o CSYTRF_BK).
!>

C

!>          C es un array COMPLEX, de dimensión (LDC,N)
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK es un array REAL, de dimensión (N)
!>

RESID

!>          RESID es REAL
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
