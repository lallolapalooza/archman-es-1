subroutine csbmv (character uplo, integer n, integer k, complex alpha, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) x, integer incx, complex beta, complex, dimension( * ) y, integer incy)

CSBMV

Propósito:

!>
!> CSBMV  realiza la operación de matriz-vector
!>
!>    y := alpha*A*x + beta*y,
!>
!> donde alpha y beta son escalares, x e y son vectores de n elementos y
!> A es una matriz de banda simétrica de n por n, con k super-diagonales.
!> 
!>  UPLO   - CHARACTER*1
!>           En la entrada, UPLO especifica si la parte triangular superior o inferior
!>           de la matriz de banda A se está proporcionando como sigue:
!>
!>              UPLO = 'U' o 'u'   Se está proporcionando la parte triangular superior de A.
!>
!>              UPLO = 'L' o 'l'   Se está proporcionando la parte triangular inferior de A.
!>
!>           Sin cambios en la salida.
!>
!>  N      - INTEGER
!>           En la entrada, N especifica el orden de la matriz A.
!>           N debe ser al menos cero.
!>           Sin cambios en la salida.
!>
!>  K      - INTEGER
!>           En la entrada, K especifica el número de super-diagonales de la
!>           matriz A. K debe satisfacer  0 .le. K.
!>           Sin cambios en la salida.
!>
!>  ALPHA  - COMPLEX
!>           En la entrada, ALPHA especifica el escalar alpha.
!>           Sin cambios en la salida.
!>
!>  A      - Arreglo COMPLEX, dimension( LDA, N )
!>           Antes de la entrada con UPLO = 'U' o 'u', la parte principal de ( k + 1 )
!>           por n del arreglo A debe contener la parte de banda triangular superior
!>           de la matriz simétrica, proporcionada columna por columna, con la
!>           diagonal principal de la matriz en la fila ( k + 1 ) del arreglo,
!>           la primera super-diagonal comenzando en la posición 2 en la fila k,
!>           y así sucesivamente. El triángulo superior izquierdo de k por k
!>           del arreglo A no se referencia.
!>           El siguiente segmento del programa transferirá la parte triangular
!>           superior de una matriz de banda simétrica desde el almacenamiento
!>           convencional de matriz completa a almacenamiento de banda:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Antes de la entrada con UPLO = 'L' o 'l', la parte principal de ( k + 1 )
!>           por n del arreglo A debe contener la parte de banda triangular inferior
!>           de la matriz simétrica, proporcionada columna por columna, con la
!>           diagonal principal de la matriz en la fila 1 del arreglo,
!>           la primera sub-diagonal comenzando en la posición 1 en la fila 2,
!>           y así sucesivamente. El triángulo inferior derecho de k por k
!>           del arreglo A no se referencia.
!>           El siguiente segmento del programa transferirá la parte triangular
!>           inferior de una matriz de banda simétrica desde el almacenamiento
!>           convencional de matriz completa a almacenamiento de banda:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Sin cambios en la salida.
!>
!>  LDA    - INTEGER
!>           En la entrada, LDA especifica la primera dimensión de A tal como
!>           se declaró en el programa (sub)llamador. LDA debe ser al menos
!>           ( k + 1 ).
!>           Sin cambios en la salida.
!>
!>  X      - Arreglo COMPLEX, dimensión al menos
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           Antes de la entrada, el arreglo X incrementado debe contener el
!>           vector x.
!>           Sin cambios en la salida.
!>
!>  INCX   - INTEGER
!>           En la entrada, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>           Sin cambios en la salida.
!>
!>  BETA   - COMPLEX
!>           En la entrada, BETA especifica el escalar beta.
!>           Sin cambios en la salida.
!>
!>  Y      - Arreglo COMPLEX, dimensión al menos
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           Antes de la entrada, el arreglo Y incrementado debe contener el
!>           vector y. En la salida, Y se sobrescribe con el vector y actualizado.
!>
!>  INCY   - INTEGER
!>           En la entrada, INCY especifica el incremento para los elementos de
!>           Y. INCY no debe ser cero.
!>           Sin cambios en la salida.
!>
