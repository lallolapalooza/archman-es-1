Cuando abarca múltiples columnas no se usan separadores de columna entre las columnas abarcadas: se inserta una única línea horizontal ininterrumpida. Ejemplo:

using namespace FBB;
int main()
{
    CSVTable tab;
    tab.fmt() << "case" << right("length", 2) << right("weight", 1) <<
                           right("length", 2) << right("weight", 1);
    tab.sep("  ");
    tab << hline();
    tab << "" << join(4, FMT::CENTER) << "Género";
    tab << "" << hline();
    tab << "" << join(2, FMT::CENTER) << "Femenino" <<

    tab << 3 << 1.674 << 62.125 << 1.8244 << 80.201;
    tab << hline();
}
