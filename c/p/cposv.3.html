subroutine cposv (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPOSV calcula la solución al sistema de ecuaciones lineales A * X = B para matrices PO

Propósito:

!>
!> CPOSV calcula la solución a un sistema de ecuaciones lineales complejas
!>    A * X = B,
!> donde A es una matriz Hermitiana positiva definida de N por N y X y B
!> son matrices de N por NRHS.
!>
!> Se utiliza la descomposición de Cholesky para factorizar A como
!>    A = U**H* U,  si UPLO = 'U', o
!>    A = L * L**H,  si UPLO = 'L',
!> donde U es una matriz triangular superior y  L es una matriz triangular
!> inferior.  Luego, la forma factorizada de A se utiliza para resolver el sistema de
!> ecuaciones A * X = B.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El número de ecuaciones lineales, es decir, el orden de
!>          la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es un array COMPLEX, de dimensión (LDA,N)
!>          En la entrada, la matriz Hermitiana A.  Si UPLO = 'U', la parte
!>          superior triangular de N por N de A contiene la parte
!>          triangular superior de la matriz A, y la parte estrictamente
!>          triangular inferior de A no se referencia. Si UPLO = 'L', la parte
!>          inferior triangular de N por N de A contiene la parte
!>          triangular inferior de la matriz A, y la parte estrictamente
!>          triangular superior de A no se referencia.
!>
!>          En la salida, si INFO = 0, el factor U o L de la descomposición
!>          de Cholesky A = U**H*U o A = L*L**H.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

B

!>          B es un array COMPLEX, de dimensión (LDB,NRHS)
!>          En la entrada, la matriz B del lado derecho de N por NRHS.
!>          En la salida, si INFO = 0, la matriz solución X de N por NRHS.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  Salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, el menor principal de orden i
!>                de A no es positivo, por lo que la descomposición no pudo
!>                completarse y la solución no se ha calculado.
!>
