subroutine cporfsx (character uplo, character equed, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, real, dimension( * ) s, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) berr, integer n_err_bnds, real, dimension( nrhs, * ) err_bnds_norm, real, dimension( nrhs, * ) err_bnds_comp, integer nparams, real, dimension(*) params, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CPORFSX

Propósito:

!>
!>    CPORFSX mejora la solución calculada de un sistema de ecuaciones lineales
!>    cuando la matriz de coeficientes es Hermítica positiva definida, y proporciona
!>    estimaciones de los límites de error y error hacia atrás para la solución.
!>    Además del límite de error normwise, el código proporciona el límite de error
!>    componentwise máximo si es posible. Vea los comentarios para ERR_BNDS_NORM y
!>    ERR_BNDS_COMP para detalles de los límites de error.
!>
!>    El sistema original de ecuaciones lineales puede haber sido equilibrado
!>    antes de llamar a esta rutina, como se describe en los argumentos EQUED y S
!>    a continuación. En este caso, la solución y los límites de error devueltos
!>    son para el sistema original no equilibrado.
!> 
!>     Algunos parámetros opcionales están agrupados en la matriz PARAMS. Estas
!>     configuraciones determinan cómo se realiza el refinamiento, pero a menudo
!>     los valores predeterminados son aceptables. Si los valores predeterminados
!>     son aceptables, los usuarios pueden pasar NPARAMS = 0, lo que evita que el
!>     código fuente acceda al argumento PARAMS.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>       = 'U':  El triángulo superior de A se almacena;
!>       = 'L':  El triángulo inferior de A se almacena.
!>

EQUED

!>          EQUED es CHARACTER*1
!>     Especifica la forma de equilibrado que se realizó en A
!>     antes de llamar a esta rutina. Esto es necesario para calcular
!>     la solución y los límites de error correctamente.
!>       = 'N':  Sin equilibrado
!>       = 'Y':  Equilibrado de filas y columnas, es decir, A ha sido
!>               reemplazado por diag(S) * A * diag(S).
!>               El lado derecho B se ha modificado en consecuencia.
!>

N

!>          N es INTEGER
!>     El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>     El número de lados derechos, es decir, el número de columnas
!>     de las matrices B y X.  NRHS >= 0.
!>

A

!>          A es array COMPLEX, dimension (LDA,N)
!>     La matriz Hermítica A.  Si UPLO = 'U', la parte superior N-por-N
!>     de A contiene la parte triangular superior de la matriz A, y la
!>     parte estrictamente triangular inferior de A no se referencia.
!>     Si UPLO = 'L', la parte inferior N-por-N de A contiene la parte
!>     triangular inferior de la matriz A, y la parte estrictamente
!>     triangular superior de A no se referencia.
!>

LDA

!>          LDA es INTEGER
!>     La dimensión principal del array A.  LDA >= max(1,N).
!>

AF

!>          AF es array COMPLEX, dimension (LDAF,N)
!>     El factor triangular U o L de la factorización de Cholesky
!>     A = U**H*U o A = L*L**H, tal como se calculó con CPOTRF.
!>

LDAF

!>          LDAF es INTEGER
!>     La dimensión principal del array AF.  LDAF >= max(1,N).
!>

S

!>          S es array REAL, dimension (N)
!>     Los factores de escala para A.  Si EQUED = 'Y', A se multiplica a
!>     la izquierda y derecha por diag(S).  S es un argumento de entrada si FACT =
!>     'F'; de lo contrario, S es un argumento de salida.  Si FACT = 'F' y EQUED
!>     = 'Y', cada elemento de S debe ser positivo.  Si S es salida, cada
!>     elemento de S es una potencia de la base. Si S es entrada, cada
!>     elemento de S debe ser una potencia de la base para asegurar una solución
!>     y estimaciones de error fiables. El escalado por potencias de la base no
!>     causa errores de redondeo a menos que el resultado sea desbordamiento o
!>     subdesbordamiento. Los errores de redondeo durante el escalado conducen
!>     al refinamiento con una matriz que no es equivalente a la matriz de entrada,
!>     produciendo estimaciones de error que pueden no ser fiables.
!>

B

!>          B es array COMPLEX, dimension (LDB,NRHS)
!>     La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>     La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es array COMPLEX, dimension (LDX,NRHS)
!>     En la entrada, la matriz solución X, tal como la calculó SGETRS.
!>     En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>     La dimensión principal del array X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND es REAL
!>     Condición número escalado recíproco. Esta es una estimación del
!>     número de condición de Skeel recíproco de la matriz A después del
!>     equilibrado (si se realizó). Si este número es menor que la precisión
!>     de la máquina (en particular, si es cero), la matriz es singular
!>     con precisión de trabajo. Tenga en cuenta que el error puede ser
!>     aún pequeño incluso si este número es muy pequeño y la matriz parece
!>     mal condicionada.
!>

BERR

!>          BERR es array REAL, dimension (NRHS)
!>     Error relativo componente a componente hacia atrás. Este es el
!>     error relativo componente a componente de cada vector solución X(j)
!>     (es decir, el cambio relativo más pequeño en cualquier elemento de A o B
!>     que haga que X(j) sea una solución exacta).
!>

N_ERR_BNDS

!>          N_ERR_BNDS es INTEGER
!>     Número de límites de error a devolver para cada lado derecho
!>     y cada tipo (normwise o componentwise). Vea ERR_BNDS_NORM y
!>     ERR_BNDS_COMP a continuación.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM es array REAL, dimension (NRHS, N_ERR_BNDS)
!>     Para cada lado derecho, esta matriz contiene información sobre
!>     varios límites de error y números de condición correspondientes al
!>     error relativo normwise, que se define de la siguiente manera:
!>
!>     Error relativo normwise en el i-ésimo vector solución:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     La matriz está indexada por el tipo de información de error tal como se
!>     describe a continuación. Actualmente hay hasta tres piezas de información
!>     devueltas.
!>
!>     El primer índice en ERR_BNDS_NORM(i,:) corresponde al i-ésimo
!>     lado derecho.
!>
!>     El segundo índice en ERR_BNDS_NORM(:,err) contiene los siguientes
!>     tres campos:
!>     err = 1  booleano. Confiar en la respuesta si el
!>              número de condición recíproco es menor que el umbral
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  límite de error: El error directo estimado,
!>              casi con certeza dentro de un factor de 10 del error real
!>              siempre que la siguiente entrada sea mayor que el umbral
!>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP es un arreglo REAL, dimensión (NRHS, N_ERR_BNDS)
!>     Para cada segundo miembro, este arreglo contiene información
!>     sobre varios límites de error y números de condición
!>     correspondientes al error relativo componente a componente, que
!>     se define como sigue:
!>
!>     Error relativo componente a componente en el i-ésimo vector
!>     solución:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     El arreglo está indexado por el segundo miembro i (del cual
!>     depende el error relativo componente a componente), y por el
!>     tipo de información de error descrito a continuación. Actualmente
!>     se devuelven hasta tres piezas de información para cada segundo
!>     miembro. Si no se solicita precisión componente a componente
!>     (PARAMS(3) = 0.0), entonces ERR_BNDS_COMP no se accede. Si
!>     N_ERR_BNDS < 3, entonces se devuelven como máximo las primeras
!>     entradas (:,N_ERR_BNDS).
!>
!>     El primer índice en ERR_BNDS_COMP(i,:) corresponde al i-ésimo
!>     segundo miembro.
!>
!>     El segundo índice en ERR_BNDS_COMP(:,err) contiene los
!>     siguientes tres campos:
!>     err = 1  booleano. Confíe en la respuesta si el número de
!>              condición recíproco es menor que el umbral
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  límite de error: El error hacia adelante estimado, casi
!>              con certeza dentro de un factor de 10 del error real,
!>              siempre que la entrada siguiente sea mayor que el umbral
!>              sqrt(n) * slamch('Epsilon'). Este límite de error solo
!>              debe confiarse si el booleano anterior es verdadero.
!>
!>     err = 3  Número de condición recíproco: Número de condición
!>              recíproco estimado componente a componente. Se compara
!>              con el umbral sqrt(n) * slamch('Epsilon') para determinar
!>              si la estimación de error es . Estos números de condición
!>              recíprocos son 1 / (norm(Z^{-1},inf) * norm(Z,inf)) para
!>              alguna matriz Z escalada adecuadamente.
!>              Sea Z = S*(A*diag(x)), donde x es la solución para el
!>              segundo miembro actual y S escala cada fila de
!>              A*diag(x) por una potencia de la base de tal modo que
!>              todas las sumas absolutas de filas de Z sean
!>              aproximadamente 1.
!>
!>     Consulte la Nota de Trabajo de Lapack 165 para más detalles y
!>     precauciones adicionales.
!>

NPARAMS

!>          NPARAMS es INTEGER
!>     Especifica el número de parámetros establecidos en PARAMS. Si
!>     <= 0, el arreglo PARAMS nunca se referencia y se usan los
!>     valores por defecto.
!>

PARAMS

!>          PARAMS es un arreglo REAL, dimensión NPARAMS
!>     Especifica los parámetros del algoritmo. Si una entrada es
!>     < 0.0, entonces esa entrada se rellenará con el valor por
!>     defecto usado para ese parámetro. Solo se accede a las posiciones
!>     hasta NPARAMS; se usan los valores por defecto para los
!>     parámetros numerados más alto.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : Si realizar refinamiento
!>            iterativo o no.
!>         Por defecto: 1.0
!>            = 0.0:  No se realiza refinamiento y no se calculan
!>                    límites de error.
!>            = 1.0:  Usar el algoritmo de refinamiento de doble
!>                    precisión, posiblemente con cálculos de
!>                    doble-precisión simple si el entorno de compilación
!>                    no soporta DOUBLE PRECISION.
!>              (otros valores están reservados para uso futuro)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : Número máximo de cálculos de
!>            residuo permitidos para el refinamiento.
!>         Por defecto: 10
!>         Agresivo: Establecer a 100 para permitir la convergencia
!>                     usando factorizaciones aproximadas o
!>                     factorizaciones distintas de LU. Si la
!>                     factorización usa una técnica distinta de la
!>                     eliminación de Gauss, las garantías en
!>                     err_bnds_norm y err_bnds_comp pueden dejar de ser
!>                     confiables.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : Bandera que determina si el
!>            código intentará encontrar una solución con pequeño error
!>            relativo componente a componente en el algoritmo de doble
!>            precisión. Positivo es verdadero, 0.0 es falso.
!>         Por defecto: 1.0 (intentar convergencia componente a componente)
!>

WORK

!>          WORK es un arreglo COMPLEX, dimensión (2*N)
!>

RWORK

!>          RWORK es un arreglo REAL, dimensión (2*N)
!>

INFO

!>          INFO es INTEGER
!>       = 0:  Salida exitosa. La solución para cada segundo miembro está
!>         garantizada.
!>       < 0:  Si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>       > 0 y <= N:  U(INFO,INFO) es exactamente cero. La
!>         factorización se ha completado, pero el factor U es exactamente
!>         singular, por lo que no se pudieron calcular la solución y los
!>         límites de error. Se devuelve RCOND = 0.
!>       = N+J: La solución correspondiente al J-ésimo segundo miembro no
!>         está garantizada. Las soluciones correspondientes a otros
!>         segundos miembros K con K > J pueden no estar garantizadas
!>         también, pero solo se reporta el primer segundo miembro de
!>         ese tipo. Si no se solicita un pequeño error componente a
!>         componente (PARAMS(3) = 0.0), entonces el J-ésimo segundo
!>         miembro es el primero con un límite de error en norma que no
!>         está garantizado (el más pequeño J tal que
!>         ERR_BNDS_NORM(J,1) = 0.0). Por defecto (PARAMS(3) = 1.0) el
!>         J-ésimo segundo miembro es el primero con un límite de error
!>         en norma o componente a componente que no está garantizado (el
!>         más pequeño J tal que ERR_BNDS_NORM(J,1) = 0.0 o
!>         ERR_BNDS_COMP(J,1) = 0.0). Consulte la definición de
!>         ERR_BNDS_NORM(:,1) y ERR_BNDS_COMP(:,1). Para obtener
!>         información sobre todos los segundos miembros, consulte
!>         ERR_BNDS_NORM o ERR_BNDS_COMP.
!>
