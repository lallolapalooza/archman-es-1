coin_shaders_page - Shaders en Coin

Coin 2.5 añadió soporte para shaders. Los nodos principales utilizados son SoShaderProgram, SoVertexShader, SoFragmentShader y SoGeometryShader. Un típico grafo de escena con shaders se vería algo así:

Separator {
  ShaderProgram {
    shaderObject [
      VertexShader {
        sourceProgram "myvertexshader.glsl"
        parameter [
          ShaderParameter1f { name "myvertexparam" value 1.0 }
        ]
      }
      FragmentShader {
        sourceProgram "myfragmentshader.glsl"
        parameter [
          ShaderParameter1f { name "myfragmentparam" value 2.0 }
        ]
      }
    ]
  }
  Cube { }
}

Esto renderizará el cubo con los shaders de vértice y fragmento especificados en myvertexshader.glsl y myfragmentshader.glsl. Coin también soporta shaders ARB y shaders Cg (si la librería Cg está instalada). Sin embargo, recomendamos usar GLSL ya que nos centraremos principalmente en el soporte de este lenguaje de shaders.

Coin define algunos parámetros con nombre que pueden ser añadidos por el programador de la aplicación, y que serán actualizados automáticamente por Coin al recorrer el grafo de escena.

•
coin_texunit[n]_model - Se establece en 0 cuando la texturización está deshabilitada, y en SoTextureImageElement::Model si hay una textura actual en el estado para la unidad n.
•
coin_light_model - Se establece en 1 para PHONG, 0 para iluminación BASE_COLOR.
•
coin_two_sided_lighting - Se establece en 1 para dos caras, 0 para normal

Ejemplo de grafo de escena que renderiza iluminación Phong de OpenGL por fragmento para una fuente de luz. Los shaders asumen que la primera fuente de luz es una luz direccional. Este es el caso si abres el archivo en un visor examinador estándar.

El archivo iv:

Separator {
  ShaderProgram {
    shaderObject [
      VertexShader {
        sourceProgram "perpixel_vertex.glsl"
      }
      FragmentShader {
        sourceProgram "perpixel_fragment.glsl"
      }
    ]
  }
  Complexity { value 1.0 }
  Material { diffuseColor 1 0 0 specularColor 1 1 1 shininess 0.9 }
  Sphere { }
  Translation { translation 3 0 0 }
  Material { diffuseColor 0 1 0 specularColor 1 1 1 shininess 0.9 }
  Cone { }
  Translation { translation 3 0 0 }
  Material { diffuseColor 0.8 0.4 0.1 specularColor 1 1 1 shininess 0.9 }
  Cylinder { }
}

El shader de vértice (perpixel_vertex.glsl):

varying vec3 ecPosition3;
varying vec3 fragmentNormal;
void main(void)
{
  vec4 ecPosition = gl_ModelViewMatrix * gl_Vertex;
  ecPosition3 = ecPosition.xyz / ecPosition.w;
  fragmentNormal = normalize(gl_NormalMatrix * gl_Normal);
  gl_Position = ftransform();
  gl_FrontColor = gl_Color;
}

El shader de fragmento (perpixel_fragment.glsl):

varying vec3 ecPosition3;
varying vec3 fragmentNormal;
void DirectionalLight(in int i,
                      in vec3 normal,
                      inout vec4 ambient,
                      inout vec4 diffuse,
                      inout vec4 specular)
{
  float nDotVP; // normal . light direction
  float nDotHV; // normal . light half vector
  float pf;     // power factor
  nDotVP = max(0.0, dot(normal, normalize(vec3(gl_LightSource[i].position))));
  nDotHV = max(0.0, dot(normal, vec3(gl_LightSource[i].halfVector)));
  if (nDotVP == 0.0)
    pf = 0.0;
  else
    pf = pow(nDotHV, gl_FrontMaterial.shininess);
  ambient += gl_LightSource[i].ambient;
  diffuse += gl_LightSource[i].diffuse * nDotVP;
  specular += gl_LightSource[i].specular * pf;
}
void main(void)
{
  vec3 eye = -normalize(ecPosition3);
  vec4 ambient = vec4(0.0);
  vec4 diffuse = vec4(0.0);
  vec4 specular = vec4(0.0);
  vec3 color;
  DirectionalLight(0, normalize(fragmentNormal), ambient, diffuse, specular);
  color =
    gl_FrontLightModelProduct.sceneColor.rgb +
    ambient.rgb * gl_FrontMaterial.ambient.rgb +
    diffuse.rgb * gl_Color.rgb +
    specular.rgb * gl_FrontMaterial.specular.rgb;
  gl_FragColor = vec4(color, gl_Color.a);
}
