subroutine claqz1 (logical, intent(in) ilq, logical, intent(in) ilz, integer, intent(in) k, integer, intent(in) istartm, integer, intent(in) istopm, integer, intent(in) ihi, complex, dimension( lda, * ) a, integer, intent(in) lda, complex, dimension( ldb, * ) b, integer, intent(in) ldb, integer, intent(in) nq, integer, intent(in) qstart, complex, dimension( ldq, * ) q, integer, intent(in) ldq, integer, intent(in) nz, integer, intent(in) zstart, complex, dimension( ldz, * ) z, integer, intent(in) ldz)

CLAQZ1

Propósito:

!>
!>      CLAQZ1 persigue un bulto de desplazamiento 1x1 en un lápiz de matriz hacia abajo una posición
!>

Parámetros

ILQ 
!>          ILQ es LOGICAL
!>              Determina si actualizar o no la matriz Q
!>

ILZ

!>          ILZ es LOGICAL
!>              Determina si actualizar o no la matriz Z
!>

K

!>          K es INTEGER
!>              Índice que indica la posición del bulto.
!>              En la entrada, el bulto se encuentra en
!>              (A(k+1,k),B(k+1,k)).
!>              En la salida, el bulto se encuentra en
!>              (A(k+2,k+1),B(k+2,k+1)).
!>

ISTARTM

!>          ISTARTM es INTEGER
!>

ISTOPM

!>          ISTOPM es INTEGER
!>              Las actualizaciones a (A,B) se restringen a
!>              (istartm:k+2,k:istopm). Se asume
!>              sin comprobación que istartm <= k+1 y
!>              k+2 <= istopm
!>

IHI

!>          IHI es INTEGER
!>

A

!>          A es un array COMPLEX, dimension (LDA,N)
!>

LDA

!>          LDA es INTEGER
!>              La dimensión principal de A tal como se declara en
!>              el procedimiento de llamada.
!>

B

!>          B es un array COMPLEX, dimension (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>              La dimensión principal de B tal como se declara en
!>              el procedimiento de llamada.
!>

NQ

!>          NQ es INTEGER
!>              El orden de la matriz Q
!>

QSTART

!>          QSTART es INTEGER
!>              Índice de inicio de la matriz Q. Las rotaciones se aplican
!>              a las columnas k+2-qStart:k+3-qStart de Q.
!>

Q

!>          Q es un array COMPLEX, dimension (LDQ,NQ)
!>

LDQ

!>          LDQ es INTEGER
!>              La dimensión principal de Q tal como se declara en
!>              el procedimiento de llamada.
!>

NZ

!>          NZ es INTEGER
!>              El orden de la matriz Z
!>

ZSTART

!>          ZSTART es INTEGER
!>              Índice de inicio de la matriz Z. Las rotaciones se aplican
!>              a las columnas k+1-qStart:k+2-qStart de Z.
!>

Z

!>          Z es un array COMPLEX, dimension (LDZ,NZ)
!>

LDZ

!>          LDZ es INTEGER
!>              La dimensión principal de Q tal como se declara en
!>              el procedimiento de llamada.
!>
