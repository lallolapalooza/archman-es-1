subroutine clsets (integer m, integer p, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, integer lda, complex, dimension( ldb, * ) b, complex, dimension( ldb, * ) bf, integer ldb, complex, dimension( * ) c, complex, dimension( * ) cf, complex, dimension( * ) d, complex, dimension( * ) df, complex, dimension( * ) x, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 2 ) result)

CLSETS

Propósito:

!>
!> CLSETS prueba CGGLSE - una subrutina para resolver el problema de
!> mínimos cuadrados con restricciones de igualdad lineal (LSE).
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz B.  P >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B.  N >= 0.
!>

A

!>          A es un array COMPLEX, dimension (LDA,N)
!>          La matriz M por N, A.
!>

AF

!>          AF es un array COMPLEX, dimension (LDA,N)
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, Q y R.
!>          LDA >= max(M,N).
!>

B

!>          B es un array COMPLEX, dimension (LDB,N)
!>          La matriz P por N, A.
!>

BF

!>          BF es un array COMPLEX, dimension (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de los arrays B, BF, V y S.
!>          LDB >= max(P,N).
!>

C

!>          C es un array COMPLEX, dimension( M )
!>          el vector C en el problema LSE.
!>

CF

!>          CF es un array COMPLEX, dimension( M )
!>

D

!>          D es un array COMPLEX, dimension( P )
!>          el vector D en el problema LSE.
!>

DF

!>          DF es un array COMPLEX, dimension( P )
!>

X

!>          X es un array COMPLEX, dimension( N )
!>          vector solución X en el problema LSE.
!>

WORK

!>          WORK es un array COMPLEX, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>

RWORK

!>          RWORK es un array REAL, dimension (M)
!>

RESULT

!>          RESULT es un array REAL, dimension (2)
!>          Las razones de prueba:
!>            RESULT(1) = norm( A*x - c )/ norm(A)*norm(X)*EPS
!>            RESULT(2) = norm( B*x - d )/ norm(B)*norm(X)*EPS
!>
