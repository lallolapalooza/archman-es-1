subroutine clatsy (character uplo, integer n, complex, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) iseed)

CLATSY

Propósito:

!>
!> CLATSY genera una matriz de prueba especial para la factorización
!> compleja simétrica (indefinida). Los bloques pivote de la matriz
!> generada estarán en el siguiente orden:
!>    bloque pivote 2x2, no diagonalizable
!>    bloque pivote 1x1
!>    bloque pivote 2x2, diagonalizable
!>    (el ciclo se repite)
!> Se requiere un intercambio de filas para cada bloque 2x2 no
!> diagonalizable.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER
!>          Especifica si la matriz generada debe ser triangular superior
!>          o inferior.
!>          = 'U': Triangular superior
!>          = 'L': Triangular inferior
!>

N

!>          N es INTEGER
!>          La dimensión de la matriz a generar.
!>

X

!>          X es array COMPLEX, dimensión (LDX,N)
!>          La matriz generada, que consiste en bloques diagonales de 3x3 y
!>          2x2 que resultan en la secuencia pivote dada anteriormente.
!>          La matriz fuera de estos bloques diagonales es cero.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.
!>

ISEED

!>          ISEED es array INTEGER, dimensión (4)
!>          En la entrada, la semilla para el generador de números aleatorios.
!>          El último de los cuatro enteros debe ser impar. (modificado en
!>          la salida)
!>
