subroutine clags2 (logical upper, real a1, complex a2, real a3, real b1, complex b2, real b3, real csu, complex snu, real csv, complex snv, real csq, complex snq)

CLAGS2

Propósito:

!>
!> CLAGS2 calcula matrices unitarias 2x2 U, V y Q, tales
!> que si ( UPPER ) entonces
!>
!>           U**H *A*Q = U**H *( A1 A2 )*Q = ( x  0  )
!>                             ( 0  A3 )     ( x  x  )
!> y
!>           V**H*B*Q = V**H *( B1 B2 )*Q = ( x  0  )
!>                            ( 0  B3 )     ( x  x  )
!>
!> o si ( .NOT.UPPER ) entonces
!>
!>           U**H *A*Q = U**H *( A1 0  )*Q = ( x  x  )
!>                             ( A2 A3 )     ( 0  x  )
!> y
!>           V**H *B*Q = V**H *( B1 0  )*Q = ( x  x  )
!>                             ( B2 B3 )     ( 0  x  )
!> donde
!>
!>   U = (   CSU    SNU ), V = (  CSV    SNV ),
!>       ( -SNU**H  CSU )      ( -SNV**H CSV )
!>
!>   Q = (   CSQ    SNQ )
!>       ( -SNQ**H  CSQ )
!>
!> Las filas de A y B transformadas son paralelas. Además, si la
!> matriz 2x2 de entrada A no es cero, entonces la entrada (1,1)
!> transformada de A no es cero. Si las matrices de entrada A y B
!> no son cero, entonces la entrada (2,2) transformada de B no es cero,
!> excepto cuando las primeras filas de A y B de entrada son paralelas
!> y las segundas filas son cero.
!>

Parámetros

UPPER 
!>          UPPER es LOGICAL
!>          = .TRUE.: las matrices de entrada A y B son triangulares superiores.
!>          = .FALSE.: las matrices de entrada A y B son triangulares inferiores.
!>

A1

!>          A1 es REAL
!>

A2

!>          A2 es COMPLEX
!>

A3

!>          A3 es REAL
!>          A la entrada, A1, A2 y A3 son elementos de la matriz
!>          triangular superior (inferior) 2x2 de entrada A.
!>

B1

!>          B1 es REAL
!>

B2

!>          B2 es COMPLEX
!>

B3

!>          B3 es REAL
!>          A la entrada, B1, B2 y B3 son elementos de la matriz
!>          triangular superior (inferior) 2x2 de entrada B.
!>

CSU

!>          CSU es REAL
!>

SNU

!>          SNU es COMPLEX
!>          La matriz unitaria deseada U.
!>

CSV

!>          CSV es REAL
!>

SNV

!>          SNV es COMPLEX
!>          La matriz unitaria deseada V.
!>

CSQ

!>          CSQ es REAL
!>

SNQ

!>          SNQ es COMPLEX
!>          La matriz unitaria deseada Q.
!>
