subroutine clapmt (logical forwrd, integer m, integer n, complex, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) k)

CLAPMT realiza una permutación hacia adelante o hacia atrás de las columnas de una matriz.

Propósito:

!>
!> CLAPMT reorganiza las columnas de la matriz M por N de X según se especifica
!> por la permutación K(1),K(2),...,K(N) de los enteros 1,...,N.
!> Si FORWRD = .TRUE., permutación hacia adelante:
!>
!>      X(*,K(J)) se mueve a X(*,J) para J = 1,2,...,N.
!>
!> Si FORWRD = .FALSE., permutación hacia atrás:
!>
!>      X(*,J) se mueve a X(*,K(J)) para J = 1,2,...,N.
!>

Parámetros

FORWRD 
!>          FORWRD es LOGICAL
!>          = .TRUE., permutación hacia adelante
!>          = .FALSE., permutación hacia atrás
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz X. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz X. N >= 0.
!>

X

!>          X es array COMPLEX, dimension (LDX,N)
!>          En la entrada, la matriz M por N X.
!>          En la salida, X contiene la matriz X permutada.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X, LDX >= MAX(1,M).
!>

K

!>          K es array INTEGER, dimension (N)
!>          En la entrada, K contiene el vector de permutación. K se usa como
!>          espacio de trabajo interno, pero se restablece a su valor original en
!>          la salida.
!>
