subroutine claein (logical rightv, logical noinit, integer n, complex, dimension( ldh, * ) h, integer ldh, complex w, complex, dimension( * ) v, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real eps3, real smlnum, integer info)

CLAEIN calcula un vector propio derecho o izquierdo especificado de una matriz de Hessenberg superior mediante iteración inversa.

Propósito:

!>
!> CLAEIN usa iteración inversa para encontrar un vector propio derecho o izquierdo
!> correspondiente al valor propio W de una matriz de Hessenberg superior H.
!>

Parámetros

RIGHTV 
!>          RIGHTV es LOGICAL
!>          = .TRUE. : calcular vector propio derecho;
!>          = .FALSE.: calcular vector propio izquierdo.
!>

NOINIT

!>          NOINIT es LOGICAL
!>          = .TRUE. : no se proporciona vector inicial en V
!>          = .FALSE.: se proporciona vector inicial en V.
!>

N

!>          N es INTEGER
!>          El orden de la matriz H.  N >= 0.
!>

H

!>          H es un array COMPLEX, dimension (LDH,N)
!>          La matriz de Hessenberg superior H.
!>

LDH

!>          LDH es INTEGER
!>          La dimensión principal del array H.  LDH >= max(1,N).
!>

W

!>          W es COMPLEX
!>          El valor propio de H cuyo vector propio derecho o izquierdo
!>          correspondiente se va a calcular.
!>

V

!>          V es un array COMPLEX, dimension (N)
!>          A la entrada, si NOINIT = .FALSE., V debe contener un vector inicial
!>          para la iteración inversa; de lo contrario, V no necesita ser establecido.
!>          A la salida, V contiene el vector propio calculado, normalizado para que
!>          el componente de mayor magnitud tenga magnitud 1; aquí
!>          la magnitud de un número complejo (x,y) se toma como
!>          |x| + |y|.
!>

B

!>          B es un array COMPLEX, dimension (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK es un array REAL, dimension (N)
!>

EPS3

!>          EPS3 es REAL
!>          Un pequeño valor dependiente de la máquina que se utiliza para perturbar
!>          valores propios cercanos y para reemplazar pivotes cero.
!>

SMLNUM

!>          SMLNUM es REAL
!>          Un valor dependiente de la máquina cercano al umbral de subdesbordamiento.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          = 1:  la iteración inversa no convergió; V se establece en el
!>                último iterado.
!>
