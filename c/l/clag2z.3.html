subroutine clag2z (integer m, integer n, complex, dimension( ldsa, * ) sa, integer ldsa, complex*16, dimension( lda, * ) a, integer lda, integer info)

CLAG2Z convierte una matriz de precisión simple compleja, SA, a una matriz de precisión doble compleja, A.

Propósito:

!>
!> CLAG2Z convierte una matriz COMPLEJA, SA, a una matriz COMPLEJA*16, A.
!>
!> Nótese que aunque es posible el desbordamiento al convertir
!> de doble a simple, no es posible el desbordamiento al convertir
!> de simple a doble.
!>
!> Esta es una subrutina auxiliar, por lo que no hay comprobación de argumentos.
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es ENTERO
!>          El número de columnas de la matriz A.  N >= 0.
!>

SA

!>          SA es un array COMPLEJO, de dimensión (LDSA,N)
!>          En la entrada, la matriz de coeficientes SA de M por N.
!>

LDSA

!>          LDSA es ENTERO
!>          La dimensión principal del array SA.  LDSA >= max(1,M).
!>

A

!>          A es un array COMPLEJO*16, de dimensión (LDA,N)
!>          En la salida, la matriz de coeficientes A de M por N.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

INFO

!>          INFO es ENTERO
!>          = 0:  salida exitosa
!>
