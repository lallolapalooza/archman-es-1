!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICO, dimension (NTYPES)
!>          Los tipos de matrices a usar para las pruebas. Las matrices del tipo j
!>          (para 1 <= j <= NTYPES) se usan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se usa.
!>

NN

!>          NN es un ENTERO
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array de ENTEROS, dimension (NN)
!>          Los valores de la dimensión N de la matriz.
!>

NNB

!>          NNB es un ENTERO
!>          El número de valores de NB contenidos en el vector NBVAL.
!>

NBVAL

!>          NBVAL es un array de ENTEROS, dimension (NNB)
!>          Los valores del tamaño de bloque NB.
!>

NNS

!>          NNS es un ENTERO
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array de ENTEROS, dimension (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las ratios de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULTADO >= UMBRAL. Para que
!>          cada ratio de prueba se imprima, usar THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICO
!>          Bandera que indica si las salidas de error deben ser probadas.
!>

NMAX

!>          NMAX es un ENTERO
!>          El valor máximo permitido para N, usado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array COMPLEJO, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array COMPLEJO, dimension (NMAX*NMAX)
!>

AINV

!>          AINV es un array COMPLEJO, dimension (NMAX*NMAX)
!>

B

!>          B es un array COMPLEJO, dimension (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array COMPLEJO, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT es un array COMPLEJO, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK es un array COMPLEJO, dimension (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array REAL, dimension (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK es un array de ENTEROS, dimension (2*NMAX)
!>

NOUT

!>          NOUT es un ENTERO
!>          La unidad de salida.
!>
