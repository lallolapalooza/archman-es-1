DESCRIPCIÓ

N

ck_rwcohort.h proporciona una interfaz para definir bloqueos de lector-escritor que usan bloqueos de cohorte internamente para aumentar el rendimiento en arquitecturas NUMA. Véase ck_cohort(3) para más información sobre los bloqueos de

cohorte.

Antes de usar un bloqueo de cohorte de lector-escritor, el usuario debe definir un tipo de cohorte usando ya sea la CK_COHORT_PROTOTYPE(3) o la CK_COHORT_TRYLOCK_PROTOTYPE(3]] macros, y definir un tipo de bloqueo de lector-escritor usando la CK_RWCOHORT_PROTOT

YPE(3)
macro.

EJEMPL

O 
#include <stdlib.h>
#include <pthread.h>

#include <ck_pr.h>
#include <ck_cohort.h>
#include <ck_rwcohort.h>
#include <ck_spinlock.h>

/* Crear métodos de cohorte con firmas que coincidan con la firma requerida */

static void
ck_spinlock_lock_with_context(ck_spinlock_t *lock, void *context)
{
	(void)context;
	ck_spinlock_lock(lock);
	return;
}

static void
ck_spinlock_unlock_with_context(ck_spinlock_t *lock, void *context)
{
	(void)context;
	ck_spinlock_unlock(lock);
	return;
}

static bool
ck_spinlock_locked_with_context(ck_spinlock_t *lock, void *context)
{
	(void)context;
	return ck_spinlock_locked(lock);
}

/*
 * definir un tipo de cohorte llamado "test_cohort" que usará
 * los métodos anteriores tanto para sus bloqueos globales como locales
 */
CK_COHORT_PROTOTYPE(test_cohort,
	ck_spinlock_lock_with_context, ck_spinlock_unlock_with_context, ck_spinlock_locked_with_context,
	ck_spinlock_lock_with_context, ck_spinlock_unlock_with_context, ck_spinlock_locked_with_context)

/* definir un tipo de lector-escritor usando el mismo tipo de cohorte */
CK_RWCOHORT_WP_PROTOTYPE(test_cohort)

static ck_spinlock_t global_lock = CK_SPINLOCK_INITIALIZER;
static CK_COHORT_INSTANCE(test_cohort) *cohorts;
static CK_RWCOHORT_WP_INSTANCE(test_cohort) rw_cohort = CK_RWCOHORT_WP_INITIALIZER;
static unsigned int ready;

static void *
function(void *context)
{
	CK_COHORT_INSTANCE(test_cohort) *cohort = context;

	while (ck_pr_load_uint(&ready) == 0);

	while (ck_pr_load_uint(&ready) > 0) {
		/*
		 * adquirir el bloqueo de cohorte antes de realizar la sección crítica.
		 * nota que pasamos NULL para ambos argumentos de contexto global y local
		 * porque ni las funciones de bloqueo ni las de desbloqueo
		 * los usarán.
		 */
		CK_COHORT_LOCK(test_cohort, cohort, NULL, NULL);

		/* realizar sección crítica */

		/* liberar bloqueo de cohorte */
		CK_COHORT_UNLOCK(test_cohort, cohort, NULL, NULL);
	}

	return NULL;
}

int
main(void)
{
	unsigned int nthr = 4;
	unsigned int n_cohorts = 2;
	unsigned int i;

	/* asignar 2 cohortes del tipo definido */
	CK_COHORT_INSTANCE(test_cohort) *cohorts =
	    calloc(n_cohorts, sizeof(CK_COHORT_INSTANCE(test_cohort)));

	/* crear bloqueos locales para usar con cada cohorte */
	ck_spinlock_t *local_locks =
		calloc(n_cohorts, sizeof(ck_spinlock_t));

	pthread_t *threads =
		calloc(nthr, sizeof(pthread_t));

	/* inicializar cada una de las cohortes antes de usarlas */
	for (i = 0 ; i < n_cohorts ; ++i) {
		CK_COHORT_INIT(test_cohort, cohorts + i, &global_lock, local_locks + i,
			CK_COHORT_DEFAULT_LOCAL_PASS_LIMIT);
	}

	/* iniciar cada hilo y asignar cohortes equitativamente */
	for (i = 0 ; i < nthr ; ++i) {
		pthread_create(threads + i, NULL, function, cohorts + (i % n_cohorts));
	}

	ck_pr_store_uint(&ready, 1);
	sleep(10);
	ck_pr_store_uint(&ready, 0);

	for (i = 0 ; i < nthr ; ++i) {
		pthread_join(threads[i], NULL);
	}

	return
0;
}
