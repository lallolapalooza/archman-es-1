EVP_PKEY_fromdata(3ssl)

Paquete: openssl

Licencias: Apache-2.0

EVP_PKEY_FROMDATA(3ssl) OpenSSL EVP_PKEY_FROMDATA(3ssl)

NOMBRE

EVP_PKEY_fromdata_init, EVP_PKEY_fromdata, EVP_PKEY_fromdata_settable - funciones para crear claves y parámetros de clave a partir de datos proporcionados por el usuario

SINOPSIS

#include <openssl/evp.h>
int EVP_PKEY_fromdata_init(EVP_PKEY_CTX *ctx);
int EVP_PKEY_fromdata(EVP_PKEY_CTX *ctx, EVP_PKEY **ppkey, int selection,
                      OSSL_PARAM params[]);
const OSSL_PARAM *EVP_PKEY_fromdata_settable(EVP_PKEY_CTX *ctx, int selection);

DESCRIPCIÓN

Las funciones descritas aquí se utilizan para crear nuevas claves a partir de datos de clave proporcionados por el usuario, como n, e y d para un par de claves RSA mínimo.

Estas funciones utilizan un contexto EVP_PKEY_CTX , que debe crearse principalmente con EVP_PKEY_CTX_new_from_name(3) o EVP_PKEY_CTX_new_id(3).

Los datos de clave exactos que el usuario puede pasar dependen del tipo de clave. Estos se pasan como un array OSSL_PARAM(3) .

EVP_PKEY_fromdata_init() inicializa un contexto de algoritmo de clave pública para crear una clave o parámetros de clave a partir de datos del usuario.

EVP_PKEY_fromdata() crea la estructura para almacenar una clave o parámetros de clave, dados datos de params, selection y un contexto que ha sido inicializado con EVP_PKEY_fromdata_init(). El resultado se escribe en *ppkey. selection se describe en "Selecciones". Los parámetros que se pueden utilizar para varios tipos de clave se describen en las distintas secciones de "Parámetros comunes" de las páginas EVP_PKEY-RSA(7), EVP_PKEY-DSA(7), EVP_PKEY-DH(7), EVP_PKEY-EC(7), EVP_PKEY-ED448(7), EVP_PKEY-X25519(7), EVP_PKEY-X448(7), EVP_PKEY-ED25519(7), EVP_PKEY-ML-DSA(7) y EVP_PKEY-ML-KEM(7) .

EVP_PKEY_fromdata_settable() obtiene un array constante OSSL_PARAM(3) que describe los parámetros configurables que se pueden usar con EVP_PKEY_fromdata(). selection se describe en "Selecciones".

Los parámetros en el array params que no estén entre los parámetros configurables para la selección dada se ignoran.

Selecciones

Las siguientes constantes se pueden usar para selection:

EVP_PKEY_KEY_PARAMETERS
Solo se seleccionarán los parámetros de clave.
EVP_PKEY_PUBLIC_KEY
Solo se seleccionarán los componentes de la clave pública. Esto incluye los parámetros de clave opcionales.
EVP_PKEY_KEYPAIR
Se seleccionarán todos los componentes del par de claves. Esto incluye la clave privada, la clave pública y los parámetros de clave.

NOTAS

Estas funciones solo funcionan con métodos de gestión de claves provenientes de un proveedor. Esta es la función espejo de EVP_PKEY_todata(3).

El soporte para LMS está deshabilitado de forma predeterminada en tiempo de compilación. Para habilitarlo, especifique la opción de configuración de compilación enable-lms .

VALORES DE RETORNO

EVP_PKEY_fromdata_init() y EVP_PKEY_fromdata() devuelven 1 en caso de éxito y 0 o un valor negativo en caso de fallo. En particular, un valor de retorno de -2 indica que la operación no es soportada por el algoritmo de clave pública.

EJEMPLOS

Estos ejemplos son muy breves para mantenerse en el tema, que es el conjunto de funciones EVP_PKEY_fromdata() . En aplicaciones reales, los BIGNUM se manejarían y convertirían a arrays de bytes con BN_bn2nativepad(), pero eso está fuera de tema aquí.

Crear un par de claves RSA usando datos de clave en bruto

#include <openssl/evp.h>
/*
 * Estos son extremadamente pequeños para mantener simple este ejemplo. Una
 * aplicación real y segura no usará números tan pequeños. Se espera que una
 * aplicación real y segura use BIGNUMs y construya
 * este array dinámicamente.
 */
unsigned long rsa_n = 0xbc747fc5;
unsigned long rsa_e = 0x10001;
unsigned long rsa_d = 0x7b133399;
OSSL_PARAM params[] = {
    OSSL_PARAM_ulong("n", &rsa_n),
    OSSL_PARAM_ulong("e", &rsa_e),
    OSSL_PARAM_ulong("d", &rsa_d),
    OSSL_PARAM_END
};
int main()
{
    EVP_PKEY_CTX *ctx = EVP_PKEY_CTX_new_from_name(NULL, "RSA", NULL);
    EVP_PKEY *pkey = NULL;
    if (ctx == NULL
        || EVP_PKEY_fromdata_init(ctx) <= 0
        || EVP_PKEY_fromdata(ctx, &pkey, EVP_PKEY_KEYPAIR, params) <= 0)
        exit(1);
    /* Haz lo que quieras con |pkey| */
}

Crear un par de claves ECC usando datos de clave en bruto

#include <openssl/evp.h>
#include <openssl/param_build.h>
#include <openssl/ec.h>
/*
 * Datos fijos para representar la clave privada y pública.
 */
const unsigned char priv_data[] = {
    0xb9, 0x2f, 0x3c, 0xe6, 0x2f, 0xfb, 0x45, 0x68,
    0x39, 0x96, 0xf0, 0x2a, 0xaf, 0x6c, 0xda, 0xf2,
    0x89, 0x8a, 0x27, 0xbf, 0x39, 0x9b, 0x7e, 0x54,
    0x21, 0xc2, 0xa1, 0xe5, 0x36, 0x12, 0x48, 0x5d
};
/* FORMATO NO COMPRIMIDO */
const unsigned char pub_data[] = {
    POINT_CONVERSION_UNCOMPRESSED,
    0xcf, 0x20, 0xfb, 0x9a, 0x1d, 0x11, 0x6c, 0x5e,
    0x9f, 0xec, 0x38, 0x87, 0x6c, 0x1d, 0x2f, 0x58,
    0x47, 0xab, 0xa3, 0x9b, 0x79, 0x23, 0xe6, 0xeb,
    0x94, 0x6f, 0x97, 0xdb, 0xa3, 0x7d, 0xbd, 0xe5,
    0x26, 0xca, 0x07, 0x17, 0x8d, 0x26, 0x75, 0xff,
    0xcb, 0x8e, 0xb6, 0x84, 0xd0, 0x24, 0x02, 0x25,
    0x8f, 0xb9, 0x33, 0x6e, 0xcf, 0x12, 0x16, 0x2f,
    0x5c, 0xcd, 0x86, 0x71, 0xa8, 0xbf, 0x1a, 0x47
};
int main()
{
    EVP_PKEY_CTX *ctx;
    EVP_PKEY *pkey = NULL;
    BIGNUM *priv;
    OSSL_PARAM_BLD *param_bld;
    OSSL_PARAM *params = NULL;
    int exitcode = 0;
    priv = BN_bin2bn(priv_data, sizeof(priv_data), NULL);
    param_bld = OSSL_PARAM_BLD_new();
    if (priv != NULL && param_bld != NULL
        && OSSL_PARAM_BLD_push_utf8_string(param_bld, "group",
                                           "prime256v1", 0)
        && OSSL_PARAM_BLD_push_BN(param_bld, "priv", priv)
        && OSSL_PARAM_BLD_push_octet_string(param_bld, "pub",
                                            pub_data, sizeof(pub_data)))
        params = OSSL_PARAM_BLD_to_param(param_bld);
    ctx = EVP_PKEY_CTX_new_from_name(NULL, "EC", NULL);
    if (ctx == NULL
        || params == NULL
        || EVP_PKEY_fromdata_init(ctx) <= 0
        || EVP_PKEY_fromdata(ctx, &pkey, EVP_PKEY_KEYPAIR, params) <= 0) {
        exitcode = 1;
    } else {
        /* Haz lo que quieras con |pkey| */
    }
    EVP_PKEY_free(pkey);
    EVP_PKEY_CTX_free(ctx);
    OSSL_PARAM_free(params);
    OSSL_PARAM_BLD_free(param_bld);
    BN_free(priv);
    exit(exitcode);
}

Descubrir los parámetros para un tipo de clave desconocido

#include <openssl/evp.h>
#include <openssl/core.h>
/* El programa espera un tipo de clave como primer argumento */
int main(int argc, char *argv[])
{
    EVP_PKEY_CTX *ctx = EVP_PKEY_CTX_new_from_name(NULL, argv[1], NULL);
    const OSSL_PARAM *settable_params = NULL;
    if (ctx == NULL)
       exit(1);
   settable_params = EVP_PKEY_fromdata_settable(ctx, EVP_PKEY_KEYPAIR);
   if (settable_params == NULL)
        exit(1);
    for (; settable_params->key != NULL; settable_params++) {
        const char *datatype = NULL;
        switch (settable_params->data_type) {
        case OSSL_PARAM_INTEGER:
            datatype = "integer";
            break;
        case OSSL_PARAM_UNSIGNED_INTEGER:
            datatype = "unsigned integer";
            break;
        case OSSL_PARAM_UTF8_STRING:
            datatype = "printable string (utf-8 encoding expected)";
            break;
        case OSSL_PARAM_UTF8_PTR:
            datatype = "printable string pointer (se espera codificación utf-8)";
            break;
        case OSSL_PARAM_OCTET_STRING:
            datatype = "cadena de octetos";
            break;
        case OSSL_PARAM_OCTET_PTR:
            datatype = "puntero a cadena de octetos";
            break;
        }
        printf("%s : %s ", settable_params->key, datatype);
        if (settable_params->data_size == 0)
            printf("(tamaño ilimitado)\n");
        else
            printf("(tamaño máximo %zu)\n", settable_params->data_size);
    }
}

El descriptor OSSL_PARAM(3) devuelto por EVP_PKEY_fromdata_settable() también puede usarse programáticamente, por ejemplo con OSSL_PARAM_allocate_from_text(3).
