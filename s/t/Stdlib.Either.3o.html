Módulo Either
: (módulo Stdlib__Either)

tipo ('a, 'b) t =
| Left de 'a
| Right de 'b

Un valor de ('a, 'b) Either.t contiene un valor de 'a o un valor de 'b.

val left : 'a -> ('a, 'b) t

left v es Left v.

val right : 'b -> ('a, 'b) t

right v es Right v.

val is_left : ('a, 'b) t -> bool

is_left (Left v) es true, is_left (Right v) es false.

val is_right : ('a, 'b) t -> bool

is_right (Left v) es false, is_right (Right v) es true.

val get_left : ('a, 'b) t -> 'a

get_left e es v si e es Left v y lanza una excepción en caso contrario.

Lanza Invalid_argument si e es Right _.

val get_right : ('a, 'b) t -> 'b

get_right e es v si e es Right v y lanza una excepción en caso contrario.

Desde 5.4

Lanza Invalid_argument si e es Left _.

val find_left : ('a, 'b) t -> 'a option

find_left (Left v) es Some v, find_left (Right _) es None.

val find_right : ('a, 'b) t -> 'b option

find_right (Right v) es Some v, find_right (Left _) es None.

val map_left : ('a1 -> 'a2) -> ('a1, 'b) t -> ('a2, 'b) t

map_left f e es Left (f v) si e es Left v y e si e es Right _.

val map_right : ('b1 -> 'b2) -> ('a, 'b1) t -> ('a, 'b2) t

map_right f e es Right (f v) si e es Right v y e si e es Left _.

val map : left:('a1 -> 'a2) -> right:('b1 -> 'b2) -> ('a1, 'b1) t -> ('a2, 'b2) t

map ~left ~right (Left v) es Left (left v), map ~left ~right (Right v) es Right (right v).

val fold : left:('a -> 'c) -> right:('b -> 'c) -> ('a, 'b) t -> 'c

fold ~left ~right (Left v) es left v y fold ~left ~right (Right v) es right v.

val retract : ('a, 'a) t -> 'a

retract (Left v) es v y retract (Right v) es v.

val iter : left:('a -> unit) -> right:('b -> unit) -> ('a, 'b) t -> unit

iter ~left ~right (Left v) es left v y iter ~left ~right (Right v) es right v.

val for_all : left:('a -> bool) -> right:('b -> bool) -> ('a, 'b) t -> bool

for_all ~left ~right (Left v) es left v y for_all ~left ~right (Right v) es right v.

val equal : left:('a -> 'a -> bool) -> right:('b -> 'b -> bool) -> ('a, 'b) t -> ('a, 'b) t -> bool

equal ~left ~right e0 e1 comprueba la igualdad de e0 y e1 usando left y right para comparar respectivamente los valores envueltos por Left _ y Right _.

val compare : left:('a -> 'a -> int) -> right:('b -> 'b -> int) -> ('a, 'b) t -> ('a, 'b) t -> int

compare ~left ~right e0 e1 ordena totalmente e0 y e1 usando left y right para comparar respectivamente los valores envueltos por Left _ y Right _. Los valores Left _ son menores que los valores Right _.
