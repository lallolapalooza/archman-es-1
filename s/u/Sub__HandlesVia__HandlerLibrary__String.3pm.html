Esta es una biblioteca de métodos para Sub::HandlesVia.

MÉTODOS DELEGABLES

append( $tail )

Argumentos: Str.

Añade otra cadena al final de la cadena actual y actualiza el atributo.

my $object = My::Class->new( attr => 'foo' );
$object->my_append( 'bar' );
say $object->attr; ## ==> 'foobar'

chomp()

Similar a "chomp" de perlfunc.

chop()

Similar a "chop" de perlfunc.

clear()

Establece la cadena a la cadena vacía.

my $object = My::Class->new( attr => 'foo' );
$object->my_clear;
say $object->attr; ## nada

cmp( $str )

Argumentos: Str.

Devuelve "$object->attr cmp $str".

cmpi( $str )

Argumentos: Str.

Devuelve "fc($object->attr) cmp fc($str)". Utiliza "lc" en lugar de "fc" en versiones de Perl anteriores a 5.16.

contains( $str )

Argumentos: Str.

Devuelve verdadero si la cadena contiene $str.

contains_i( $str )

Argumentos: Str.

Devuelve verdadero si la cadena contiene $str sin distinguir mayúsculas y minúsculas.

ends_with( $tail )

Argumentos: Str.

Devuelve verdadero si la cadena termina con $tail.

ends_with_i( $tail )

Argumentos: Str.

Devuelve verdadero si la cadena termina con $tail sin distinguir mayúsculas y minúsculas.

eq( $str )

Argumentos: Str.

Devuelve "$object->attr eq $str".

eqi( $str )

Argumentos: Str.

Devuelve "fc($object->attr) eq fc($str)". Utiliza "lc" en lugar de "fc" en versiones de Perl anteriores a 5.16.

fc()

Devuelve fc($object->attr).

ge( $str )

Argumentos: Str.

Devuelve "$object->attr ge $str".

gei( $str )

Argumentos: Str.

Devuelve "fc($object->attr) ge fc($str)". Utiliza "lc" en lugar de "fc" en versiones de Perl anteriores a 5.16.

get()

Obtiene el valor actual de la cadena.

my $object = My::Class->new( attr => 'foo' );
say $object->my_get; ## ==> 'foo'

gt( $str )

Argumentos: Str.

Devuelve "$object->attr gt $str".

Argumentos: Str.

Devuelve "fc($object->attr) gt fc($str)". Utiliza "lc" en lugar de "fc" en versiones de Perl anteriores a 5.16.

inc()

Sub::HandlesVia::HandlerLibrary::String(3pm)

NAME

Sub::HandlesVia::HandlerLibrary::String - Implement string-related attribute handling.

SYNOPSIS

  package My::Class {
    use Moo;
    use Sub::HandlesVia;
    has attr => (
      is => 'rw',
      handles_via => 'String',
      handles => {
        attr_is_foo => [ eq => 'foo' ],
        attr_is_bar => [ eq => 'bar' ],
        attr_starts_with_baz => [ starts_with => 'baz' ],
        attr_ends_with_qux => [ ends_with => 'qux' ],
        attr_matches => [ match => qr/\A[A-Z]+\z/ ],
        attr_matches_i => [ match_i => qr/\A[a-z]+\z/ ],
        attr_ne_bar => [ ne => 'bar' ],
        attr_nei_bar => [ nei => 'bar' ],
        attr_prepend_hello => [ prepend => 'Hello, ' ],
        attr_replace_o_with_a => [ replace => 'o', => 'a' ],
        attr_replace_globally_o_with_a => [ replace_globally => 'o', => 'a' ],
        attr_uc => 'uc',
        attr_strip => 'strip',
      },
    );
  }

DESCRIPTION

This handler library provides methods for manipulating string attributes.

DELEGATABLE METHODS

append( $tail )

Arguments: Str.

Appends another string to the end of the current string and updates the attribute.

my $object = My::Class->new( attr => 'foo' );
$object->my_append( 'bar' );
say $object->attr; ## ==> 'foobar'

chomp()

Removes a trailing newline character, if any.

my $object = My::Class->new( attr => "foo\n" );
$object->my_chomp;
say $object->attr; ## ==> 'foo'

chop()

Removes the last character of the string, returning that character.

my $object = My::Class->new( attr => 'foo' );
my $last_char = $object->my_chop;
say $object->attr; ## ==> 'fo'
say $last_char;   ## ==> 'o'

contains( $needle )

Arguments: Str.

Returns true iff the string contains $needle.

contains_i( $needle )

Arguments: Str.

Returns true iff the string contains $needle case insensitively.

ends_with( $tail )

Arguments: Str.

Returns true iff the string ends with $tail.

ends_with_i( $tail )

Arguments: Str.

Returns true iff the string ends with $tail case insensitively.

eq( $str )

Arguments: Str.

Returns "$object->attr eq $str".

eqi( $str )

Arguments: Str.

Returns "fc($object->attr) eq fc($str)". Uses "lc" instead of "fc" in versions of Perl older than 5.16.

lc()

Returns lc($object->attr).

match( $regexp )

Arguments: Str|RegexpRef.

Returns true iff the string matches the regexp.

my $object = My::Class->new( attr => 'foo' );
if ( $object->my_match( '^f..$' ) ) {
  say 'matched!';
}

match_i( $regexp )

Arguments: Str|RegexpRef.

Returns true iff the string matches the regexp case-insensitively.

my $object = My::Class->new( attr => 'foo' );
if ( $object->my_match_i( '^F..$' ) ) {
  say 'matched!';
}

ne( $str )

Arguments: Str.

Returns "$object->attr ne $str".

nei( $str )

Arguments: Str.

Returns "fc($object->attr) ne fc($str)". Uses "lc" instead of "fc" in versions of Perl older than 5.16.

prepend( $head )

Arguments: Str.

Prepends another string to the start of the current string and updates the attribute.

my $object = My::Class->new( attr => 'foo' );
$object->my_prepend( 'bar' );
say $object->attr; ## ==> 'barfoo'

"replace( $regexp, $replacement )"

Arguments: Str|RegexpRef, Str|CodeRef.

Replaces the first regexp match within the string with the replacement string.

my $object = My::Class->new( attr => 'foo' );
$object->my_replace( 'o' => 'a' );
say $object->attr; ## ==> 'fao'
my $object2 = My::Class->new( attr => 'foo' );
$object2->my_replace( qr/O/i => sub { return 'e' } );
say $object2->attr; ## ==> 'feo'

"replace_globally( $regexp, $replacement )"

Arguments: Str|RegexpRef, Str|CodeRef.

Replaces the all regexp matches within the string with the replacement string.

my $object = My::Class->new( attr => 'foo' );
$object->my_replace_globally( 'o' => 'a' );
say $object->attr; ## ==> 'faa'
my $object2 = My::Class->new( attr => 'foo' );
$object2->my_replace_globally( qr/O/i => sub { return 'e' } );
say $object2->attr; ## ==> 'fee'

reset()

Resets the attribute to its default value, or an empty string if it has no default.

set( $value )

Arguments: Str.

Sets the string to a new value.

my $object = My::Class->new( attr => 'foo' );
$object->my_set( 'bar' );
say $object->attr; ## ==> 'bar'

starts_with( $head )

Arguments: Str.

Returns true iff the string starts with $head.

starts_with_i( $head )

Arguments: Str.

Returns true iff the string starts with $head case-insensitively.

"substr( $start, $length?, $replacement? )"

Arguments: Int, Optional[Int], Optional[Str].

Like "substr" from perlfunc, but is not an lvalue.

strip()

Removes leading and trailing whitespace from the string.

my $object = My::Class->new( attr => "  foo  " );
$object->my_strip;
say $object->attr; ## ==> 'foo'

trim()

Like "trim" from builtin, but in-place.

uc()

Returns uc($object->attr).

EXTENDED EXAMPLES

Using eq for Enum

use strict;
use warnings;

package My::Person {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard qw( Str Enum );

  has name => (
    is => 'ro',
    isa => Str,
    required => 1,
  );

  has status => (
    is => 'rwp',
    isa => Enum[ 'alive', 'dead' ],
    handles_via => 'String',
    handles => {
      is_alive => [ eq  => 'alive' ],
      is_dead  => [ eq  => 'dead' ],
      kill     => [ set => 'dead' ],
    },
    default => 'alive',
  );

  # Note: method modifiers work on delegated methods
  #
  before kill => sub {
    my $self = shift;
    warn "overkill" if $self->is_dead;
  };
}

my $bob = My::Person->new( name => 'Robert' );
say $bob->is_alive; ## ==> true
say $bob->is_dead;  ## ==> false
$bob->kill;
say $bob->is_alive; ## ==> false
say $bob->is_dead;  ## ==> true

See also MooX::Enumeration and MooseX::Enumeration.

Match with curried regexp

use strict;
use warnings;

package My::Component {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard qw( Str Int );

  has id => (
    is => 'ro',
    isa => Int,
    required => 1,
  );

  has name => (
    is => 'ro',
    isa => Str,
    required => 1,
    handles_via => 'String',
    handles => {
      name_is_safe_filename => [ match => qr/\A[A-Za-z0-9]+\z/ ],
      _lc_name => 'lc',
    },
  );

  sub config_filename {
    my $self = shift;
    if ( $self->name_is_safe_filename ) {
      return sprintf( '%s.ini', $self->_lc_name );
    }
    return sprintf( 'component-%d.ini', $self->id );
  }
}

my $foo = My::Component->new( id => 42, name => 'Foo' );
say $foo->config_filename; ## ==> 'foo.ini'

my $bar4 = My::Component->new( id => 99, name => 'Bar #4' );
say $bar4->config_filename; ## ==> 'component-99.ini'
