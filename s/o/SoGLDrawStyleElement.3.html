SoGLDrawStyleElement — OpenGL draw style element

Detailed Description

This element contains the OpenGL draw style information. The draw style indicates how primitives should be rendered.

The possible values are defined in the SoGLDrawStyleElement enumeration.

enum SoGLDrawStyleElement {
    SoGLDrawStyleElement::NONE,       // No explicit draw style set
    SoGLDrawStyleElement::POINTS,     // Render points
    SoGLDrawStyleElement::LINES,      // Render lines
    SoGLDrawStyleElement::LINE_STRIP, // Render line strip
    SoGLDrawStyleElement::LINE_LOOP,  // Render line loop
    SoGLDrawStyleElement::TRIANGLES,  // Render triangles
    SoGLDrawStyleElement::TRIANGLE_STRIP, // Render triangle strip
    SoGLDrawStyleElement::TRIANGLE_FAN, // Render triangle fan
    SoGLDrawStyleElement::QUADS,      // Render quads
    SoGLDrawStyleElement::QUAD_STRIP,   // Render quad strip
    SoGLDrawStyleElement::POLYGON     // Render polygon
}
