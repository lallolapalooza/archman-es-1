NOMBRE

TIFFcolor - rutinas de conversión de color

SINOPSIS

#include <tiffio.h>
int TIFFYCbCrToRGBInit(TIFFYCbCrToRGB *ycbcr, float *luma, float *refBlackWhite)
void TIFFYCbCrtoRGB(TIFFYCbCrToRGB *ycbcr, uint32_t Y, int32_t Cb, int32_t Cr, uint32_t *R, uint32_t *G, uint32_t *B)
int TIFFCIELabToRGBInit(TIFFCIELabToRGB *cielab, const TIFFDisplay *displayw, float *refWhite)
void TIFFCIELabToXYZ(TIFFCIELabToRGB *cielab, uint32_t L, int32_t a, int32_t b, float *X, float *Y, float *Z)
void TIFFXYZToRGB(TIFFCIELabToRGB *cielab, float X, float Y, float Z, uint32_t *R, uint32_t *G, uint32_t *B)

DESCRIPCIÓN

TIFF admite varios espacios de color para las imágenes almacenadas en ese formato. Generalmente existe el problema de que la aplicación maneje los datos correctamente y convierta entre diferentes espacios de color para fines de visualización e impresión. Para simplificar esta tarea, libtiff implementa varias rutinas de conversión de color. En particular, estas rutinas se utilizan en la interfaz TIFFRGBAImage.

TIFFYCbCrToRGBInit() se utiliza para inicializar el estado de conversión de YCbCr a RGB. La asignación y liberación de la estructura ycbcr pertenece al programador. TIFFYCbCrToRGB se define en tiffio.h como:

typedef struct {                /* Soporte YCbCr->RGB */
        TIFFRGBValue* clamptab; /* tabla de limitación de rango */
        int*          Cr_r_tab;
        int*          Cb_b_tab;
        int32_t*      Cr_g_tab;
        int32_t*      Cb_g_tab;
        int32_t*      Y_tab;
} TIFFYCbCrToRGB;

luma es un array de float de tres valores que representan las proporciones de rojo, verde y azul en la luminancia, Y (ver la sección 21 de la especificación TIFF 6.0, donde se discuten las imágenes YCbCr). TIFFTAG_YCBCRCOEFFICIENTS contiene estos valores en el archivo TIFF. refBlackWhite es un array de float de 6 valores que especifica un par de valores de datos de imagen de headroom y footroom (códigos) para cada componente de imagen (ver la sección 20 de la especificación TIFF 6.0 donde se discuten los campos de colorimetría). TIFFTAG_REFERENCEBLACKWHITE es responsable de almacenar estos valores en el archivo TIFF. El siguiente fragmento de código debería ayudar a entender la técnica:

float *luma, *refBlackWhite;
uint16_t hs, vs;
/* Inicializar estructuras */
ycbcr = (TIFFYCbCrToRGB*)
    _TIFFmalloc(TIFFroundup(sizeof(TIFFYCbCrToRGB), sizeof(long))
        + 4*256*sizeof(TIFFRGBValue)
        + 2*256*sizeof(int)
        + 3*256*sizeof(int32_t));
if (ycbcr == NULL) {
    TIFFError("YCbCr->RGB",
    "No hay espacio para el estado de conversión YCbCr->RGB");
    exit(0);
}
TIFFGetFieldDefaulted(tif, TIFFTAG_YCBCRCOEFFICIENTS, &luma);
TIFFGetFieldDefaulted(tif, TIFFTAG_REFERENCEBLACKWHITE, &refBlackWhite);
if (TIFFYCbCrToRGBInit(ycbcr, luma, refBlackWhite) < 0) {
    exit(0);
}
/* Iniciar conversión */
uint32_t r, g, b;
uint32_t Y;
int32_t Cb, Cr;
for each pixel in image {
    TIFFYCbCrtoRGB(img->ycbcr, Y, Cb, Cr, &r, &g, &b);
}
/* Liberar estructura de estado */
_TIFFfree(ycbcr);

TIFFCIELabToRGBInit() inicializa el estado de conversión de CIE L*a*b* 1976 a RGB. TIFFCIELabToRGB se define como:

#define CIELABTORGB_TABLE_RANGE 1500
typedef struct {                                /* Soporte CIE Lab 1976->RGB */
    int      range;                             /* Tamaño de la tabla de conversión */
    float    rstep, gstep, bstep;
    float    X0, Y0, Z0;                        /* Punto blanco de referencia */
    TIFFDisplay display;
    float    Yr2r[CIELABTORGB_TABLE_RANGE + 1]; /* Conversión de Yr a r */
    float    Yg2g[CIELABTORGB_TABLE_RANGE + 1]; /* Conversión de Yg a g */
    float    Yb2b[CIELABTORGB_TABLE_RANGE + 1]; /* Conversión de Yb a b */
} TIFFCIELabToRGB;

display es una descripción del dispositivo de visualización, declarada como:

typedef struct {
    float d_mat[3][3]; /* Matriz XYZ -> luminancia */
    float d_YCR;       /* Salida de luz para blanco de referencia */
    float d_YCG;
    float d_YCB;
    uint32_t d_Vrwr;   /* Valores de píxel para blanco de referencia */
    uint32_t d_Vrwg;
    uint32_t d_Vrwb;
    float d_Y0R;       /* Salida de luz residual para píxel negro */
    float d_Y0G;
    float d_Y0B;
    float d_gammaR;    /* Valores gamma para los tres cañones */
    float d_gammaG;
    float d_gammaB;
} TIFFDisplay;

Por ejemplo, se puede usar el dispositivo sRGB, que tiene los siguientes parámetros:

TIFFDisplay display_sRGB = {
    {   /* Matriz XYZ -> luminancia */
        {  3.2410F, -1.5374F, -0.4986F },
        {  -0.9692F, 1.8760F, 0.0416F },
        {  0.0556F, -0.2040F, 1.0570F }
    },
    100.0F, 100.0F, 100.0F, /* Salida de luz para blanco de referencia */
    255, 255, 255,          /* Valores de píxel para blanco de referencia */
    1.0F, 1.0F, 1.0F,       /* Salida de luz residual para píxel negro */
    2.4F, 2.4F, 2.4F,       /* Valores gamma para los tres cañones */
};

refWhite es la temperatura de color del blanco de referencia. TIFFTAG_WHITEPOINT contiene la cromaticidad del punto blanco de la imagen desde donde se puede calcular el blanco de referencia usando las siguientes fórmulas:

refWhite_Y = 100.0
refWhite_X = whitePoint_x / whitePoint_y * refWhite_Y
refWhite_Z = (1.0 - whitePoint_x - whitePoint_y) / whitePoint_y * refWhite_X

La conversión en sí se realiza en dos pasos: en el primero convertiremos CIE L*a*b* 1976 a CIE XYZ usando la rutina TIFFCIELabToXYZ(), y en el segundo paso convertiremos CIE XYZ a RGB usando TIFFXYZToRGB(). Mire el siguiente ejemplo de código:

float   *whitePoint;
float   refWhite[3];
/* Inicializar estructuras */
img->cielab = (TIFFCIELabToRGB *)
    _TIFFmalloc(sizeof(TIFFCIELabToRGB));
if (!cielab) {
    TIFFError("CIE L*a*b*->RGB",
        "No hay espacio para el estado de conversión CIE L*a*b*->RGB.");
    exit(0);
}
TIFFGetFieldDefaulted(tif, TIFFTAG_WHITEPOINT, &whitePoint);
refWhite[1] = 100.0F;
refWhite[0] = whitePoint[0] / whitePoint[1] * refWhite[1];
refWhite[2] = (1.0F - whitePoint[0] - whitePoint[1])
    / whitePoint[1] * refWhite[1];
if (TIFFCIELabToRGBInit(cielab, &display_sRGB, refWhite) < 0) {
    TIFFError("CIE L*a*b*->RGB",
        "Fallo al inicializar el estado de conversión CIE L*a*b*->RGB.");
    _TIFFfree(cielab);
    exit(0);
}
/* Ahora podemos empezar a convertir */
uint32_t r, g, b;
uint32_t L;
int32_t a, b;
float X, Y, Z;
for each pixel in image {
    TIFFCIELabToXYZ(cielab, L, a, b, &X, &Y, &Z);
    TIFFXYZToRGB(cielab, X, Y, Z, &r, &g, &b);
}
/* No olvides liberar la estructura de estado */
_TIFFfree(cielab);
