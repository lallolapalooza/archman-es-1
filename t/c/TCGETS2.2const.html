NOMBRE

TCGETS, TCSETS, TCSETSW, TCSETSF - obtener y configurar atributos de la terminal

BIBLIOTECA

Biblioteca C estándar (libc,  -lc)

DESCRIPCIÓN

TCGETS
Equivalente a tcgetattr(fd, argp).
Obtener la configuración actual del puerto serie.
TCSETS
Equivalente a tcsetattr(fd, TCSANOW, argp).
Configurar la configuración actual del puerto serie.
TCSETSW
Equivalente a tcsetattr(fd, TCSADRAIN, argp).
Permitir que el buffer de salida se vacíe y configurar la configuración actual del puerto serie.
TCSETSF
Equivalente a tcsetattr(fd, TCSAFLUSH, argp).
Permitir que el buffer de salida se vacíe, descartar la entrada pendiente y configurar la configuración actual del puerto serie.

Los siguientes cuatro ioctls son similares a TCGETS, TCSETS, TCSETSW, TCSETSF, excepto que toman un struct termios2 en lugar de un struct termios. Si el miembro c_cflag contiene el flag BOTHER, entonces la velocidad en baudios se almacena en los miembros c_ispeed y c_ospeed como valores enteros. Estos ioctls no están soportados en todas las arquitecturas.

TCSETS2
TCSETSW2

Los siguientes cuatro ioctls son similares a TCGETS, TCSETS, TCSETSW, TCSETSF, excepto que toman un struct termio en lugar de un struct termios.

TCSETA

VALOR DE RETORNO

En caso de éxito, se devuelve 0. En caso de error, se devuelve -1 y errno se establece para indicar el error.

ERRORES

EPERM
Permiso insuficiente.

ADVERTENCIAS

struct termios desde <asm/termbits.h> es diferente e incompatible con struct termios desde <termios.h>. Estas llamadas ioctl requieren struct termios desde <asm/termbits.h>.

EJEMPLOS

Obtener o configurar la velocidad en baudios en el puerto serie.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER no está soportado\n");
    /* El programa puede fallar en TCGETS/TCSETS con constantes Bnnn */
    exit(EXIT_FAILURE);
#else
    /* Declara la estructura tio, su tipo depende de las ioctls soportadas */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "Uso: %s dispositivo [salida [entrada]]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* Obtener la configuración actual del puerto serie mediante ioctl soportada */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* Cambiar la velocidad en baudios cuando hay más argumentos */
    if (argc == 3 || argc == 4) {
        /* Limpiar la velocidad de salida actual y rellenar un nuevo valor */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* Limpiar la velocidad de entrada actual y rellenar un nuevo valor */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* Cuando el argumento 4 no se proporciona, reutiliza la velocidad de salida */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* Configurar la nueva configuración del puerto serie mediante ioctl soportada */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* Y obtener los valores que realmente fueron configurados */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("velocidad de salida: %u\n", tio.c_ospeed);
    printf("velocidad de entrada: %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}
