NOMBRE

TCGETS, TCSETS, TCSETSW, TCSETSF, TCGETS2, TCSETS2, TCSETSW2, TCSETSF2, TCGETA, TCSETA, TCSETAW, TCSETAF - obtener y establecer atributos del terminal

DESCRIPCIÓN

TCGETS
Equivalente a tcgetattr(fd, argp).
Obtiene la configuración actual del puerto serie.
TCSETS
Equivalente a tcsetattr(fd, TCSANOW, argp).
Establece la configuración actual del puerto serie.
TCSETSW
Equivalente a tcsetattr(fd, TCSADRAIN, argp).
Permite que el búfer de salida se vacíe y establece la configuración actual del puerto serie.
TCSETSF
Equivalente a tcsetattr(fd, TCSAFLUSH, argp).
Permite que el búfer de salida se vacíe, descarta la entrada pendiente y establece la configuración actual del puerto serie.

Los siguientes cuatro ioctls son iguales a TCGETS, TCSETS, TCSETSW, TCSETSF, excepto que toman un struct termios2 * en lugar de un struct termios *. Si el miembro de la estructura c_cflag contiene la bandera BOTHER, entonces la velocidad en bauds se almacena en los miembros de la estructura c_ispeed y c_ospeed como valores enteros. Estos ioctls no son compatibles con todas las arquitecturas.

VALOR DE RETORNO

En caso de éxito, se devuelve 0. En caso de error, se devuelve -1 y se establece errno para indicar el error.

ERRORES

EPERM
Permiso insuficiente.

ADVERTENCIAS

struct termios de <asm/termbits.h> es diferente e incompatible con struct termios de <termios.h>. Estas llamadas ioctl requieren struct termios de <asm/termbits.h>.

EJEMPLOS

Obtener o establecer una velocidad en bauds arbitraria en el puerto serie.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER no es compatible\n");
    /* El programa puede recurrir a TCGETS/TCSETS con constantes Bnnn */
    exit(EXIT_FAILURE);
#else
    /* Declarar estructura tio, su tipo depende del ioctl soportado */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "Uso: %s dispositivo [salida [entrada] ]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* Obtener la configuración actual del puerto serie mediante ioctl soportado */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* Cambiar la velocidad en bauds cuando se proporcionaron argumentos adicionales */
    if (argc == 3 || argc == 4) {
        /* Limpiar la velocidad en bauds de salida actual y rellenar un nuevo valor */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* Limpiar la velocidad en bauds de entrada actual y rellenar un nuevo valor */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* Cuando el 4to argumento no se proporciona, reutilizar la velocidad en bauds de salida */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* Establecer la nueva configuración del puerto serie mediante ioctl soportado */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* Y obtener los nuevos valores que fueron realmente configurados */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("velocidad en bauds de salida: %u\n", tio.c_ospeed);
    printf("velocidad en bauds de entrada: %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}
