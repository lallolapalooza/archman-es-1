TCSETS(2const) TCSETS(2const)

NOMBRE

TCGETS, TCSETS, TCSETSW, TCSETSF, TCGETS2, TCSETS2, TCSETSW2, TCSETSF2, TCGETA, TCSETA, TCSETAW, TCSETAF - obtener y establecer atributos de terminal

BIBLIOTECA

Biblioteca C estándar (libc, -lc)

SINOPSIS

#include <asm/termbits.h>  /* Definición de constantes TC* */
#include <sys/ioctl.h>
int ioctl(int fd, TCGETS, struct termios *argp);
int ioctl(int fd, TCSETS, const struct termios *argp);
int ioctl(int fd, TCSETSW, const struct termios *argp);
int ioctl(int fd, TCSETSF, const struct termios *argp);
int ioctl(int fd, TCGETS2, struct termios2 *argp);
int ioctl(int fd, TCSETS2, const struct termios2 *argp);
int ioctl(int fd, TCSETSW2, const struct termios2 *argp);
int ioctl(int fd, TCSETSF2, const struct termios2 *argp);
int ioctl(int fd, TCGETA, struct termio *argp);
int ioctl(int fd, TCSETA, const struct termio *argp);
int ioctl(int fd, TCSETAW, const struct termio *argp);
int ioctl(int fd, TCSETAF, const struct termio *argp);
#include <asm/termbits.h>
struct termios;
struct termios2;
struct termio;

DESCRIPCIÓN

TCGETS
Equivalente a tcgetattr(fd, argp).
Obtener la configuración actual del puerto serie.
TCSETS
Equivalente a tcsetattr(fd, TCSANOW, argp).
Establecer la configuración actual del puerto serie.
TCSETSW
Equivalente a tcsetattr(fd, TCSADRAIN, argp).
Permitir que el búfer de salida se drene y establecer la configuración actual del puerto serie.
TCSETSF
Equivalente a tcsetattr(fd, TCSAFLUSH, argp).
Permitir que el búfer de salida se drene, descartar la entrada pendiente y establecer la configuración actual del puerto serie.

Los siguientes cuatro ioctls son iguales a TCGETS, TCSETS, TCSETSW, TCSETSF, excepto que toman un struct termios2 * en lugar de un struct termios *. Si el miembro de la estructura c_cflag contiene la bandera BOTHER, entonces la velocidad en baudios se almacena en los miembros de la estructura c_ispeed y c_ospeed como valores enteros. Estos ioctls no son compatibles en todas las arquitecturas.

VALOR DE RETORNO

Al éxito, se devuelve 0. En caso de error, se devuelve -1 y errno se establece para indicar el error.

ERRORES

EPERM
Permiso insuficiente.

HISTORIA

ADVERTENCIAS

struct termios de <asm/termbits.h> es diferente e incompatible con struct termios de <termios.h>. Estas llamadas ioctl requieren struct termios de <asm/termbits.h>.

EJEMPLOS

Obtener o establecer una velocidad en baudios arbitraria en el puerto serie.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER no es compatible\n");
    /* El programa puede recurrir a TCGETS/TCSETS con constantes Bnnn */
    exit(EXIT_FAILURE);
#else
    /* Declarar la estructura tio, su tipo depende del ioctl soportado */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "Uso: %s dispositivo [salida [entrada] ]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* Obtener la configuración actual del puerto serie a través del ioctl soportado */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* Cambiar la velocidad en baudios cuando se proporcionaron más argumentos */
    if (argc == 3 || argc == 4) {
        /* Limpiar la velocidad en baudios de salida actual y rellenar un nuevo valor */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* Limpiar la velocidad en baudios de entrada actual y rellenar un nuevo valor */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* Cuando el 4to argumento no se proporciona, se reutiliza la velocidad en baudios de salida */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* Establecer la nueva configuración del puerto serie a través del ioctl soportado */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* Y obtener los nuevos valores que fueron realmente configurados */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("velocidad en baudios de salida: %u\n", tio.c_ospeed);
    printf("velocidad en baudios de entrada: %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}

TCSETS(2)

Linux Programmer's Manual

NAME

tcsetpgrp - set the foreground process group ID of the terminal

LIBRARY

Standard C library (libc, -lc)

SYNOPSIS

#include <termios.h>

pid_t tcsetpgrp(int fd, pid_t pgrp);

DESCRIPTION

The tcsetpgrp() function sets the foreground process group ID of the terminal associated with the file descriptor fd to pgrp.

The terminal associated with fd must be a controlling terminal.

A process can only change the foreground process group ID of a terminal if it is a member of that terminal's session.

A process group is a set of processes, all with the same process group ID. The process group ID is a positive integer.

The foreground process group is the process group that is allowed to receive terminal-related signals (like SIGINT, SIGQUIT, SIGTSTP) and to read from and write to the terminal.

Normally, when a process starts a new session using setsid(2), it becomes the foreground process group of its controlling terminal. When a process in the foreground process group performs an ioctl(2) with TIOCSCTTY, it becomes the foreground process group.

Other processes in the same session can change the foreground process group ID of the terminal to one of the existing process groups in the session using tcsetpgrp(). This is typically done by a shell when it starts a background job.

RETURN VALUE

On success, tcsetpgrp() returns the new foreground process group ID, which is equal to pgrp.

On error, tcsetpgrp() returns (pid_t) -1 and sets errno to indicate the error.

ERRORS

EBADF
fd is not a valid file descriptor.
ENOTTY
The file descriptor fd is not associated with a terminal.
EPERM
The calling process is not a member of the session that owns the terminal.
EINVAL
pgrp is not a positive integer or is not a process group ID of the calling process's session.

HISTORY

This function first appeared in POSIX.1-2001.

CAVEATS

The behavior of this function is unspecified if the terminal is not a controlling terminal.

EXAMPLES

The following example sets the foreground process group ID of the terminal associated with standard input to the process group ID of the calling process.

#include <stdio.h>
#include <unistd.h>
#include <termios.h>
#include <errno.h>

int main(void) {
    pid_t current_pgid;
    int result;

    current_pgid = getpgrp(); // Get the current process group ID

    result = tcsetpgrp(STDIN_FILENO, current_pgid);
    if (result == -1) {
        perror("tcsetpgrp");
        return 1;
    }

    printf("Foreground process group set to %d\n", result);

    return 0;
}
