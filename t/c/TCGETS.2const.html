TCGETS(2const) TCGETS(2const)

NOMBRE

TCGETS, TCSETS, TCSETSW, TCSETSF - obtener y configurar atributos del puerto serial

LIBRERÍA

Librería estándar (libc, -lc)

SINOPSIS

#include <asm/termbits.h>  /* Definición de constantes TC* */
#include <sys/ioctl.h>
int ioctl(int fd, TCGETS, struct termios *argp);
int ioctl(int fd, TCSETS, const struct termios *argp);
int ioctl(int fd, TCSETSW, const struct termios *argp);
int ioctl(int fd, TCSETSF, const struct termios *argp);
int ioctl(int fd, TCGETS2, struct termios2 *argp);
int ioctl(int fd, TCSETS2, const struct termios2 *argp);
int ioctl(int fd, TCSETSW2, const struct termios2 *argp);
int ioctl(int fd, TCSETSF2, const struct termios2 *argp);
int ioctl(int fd, TCGETA, struct termio *argp);
int ioctl(int fd, TCSETA, const struct termio *argp);
int ioctl(int fd, TCSETAW, const struct termio *argp);
int ioctl(int fd, TCSETAF, const struct termio *argp);
#include <asm/termbits.h>
struct termios;
struct termios2;
struct termio;

DESCRIPCIÓN

TCGETS
Equivalente a tcgetattr(fd, argp).
Obtener las configuraciones actuales del puerto serial.
TCSETS
Equivalente a tcsetattr(fd, TCSANOW, argp).
Configurar las configuraciones actuales del puerto serial.
TCSETSW
Equivalente a tcsetattr(fd, TCSADRAIN, argp).
Permitir que el buffer de salida se vacíe y configurar las configuraciones actuales del puerto serial.
TCSETSF
Equivalente a tcsetattr(fd, TCSAFLUSH, argp).
Permitir que el buffer de salida se vacíe, descartar la entrada pendiente y configurar las configuraciones actuales del puerto serial.

Los siguientes cuatro ioctls son similares a TCGETS, TCSETS, TCSETSW, TCSETSF, excepto que toman un struct termios2* en lugar de un struct termios*. Si el miembro c_cflag contiene el flag BOTHER, entonces la tasa de baudios se almacena en los miembros c_ispeed y c_ospeed como valores enteros. Estos ioctls no están soportados en todas las arquitecturas.

VALOR DE RETORNO

En éxito, se devuelve 0. En error, -1 y errno se establece para indicar el error.

ERRORES

EPERM
Permiso insuficiente.

HISTORIA

TCGETS2
TCSETS2
TCSETSW2
TCSETSF2
Linux 2.6.20.

CAVEATS

struct termios desde <asm/termbits.h> es diferente e incompatible con struct termios desde <termios.h>. Estos ioctls requieren struct termios desde <asm/termbits.h>.

EJEMPLOS

Obtener o configurar cualquier tasa de baudios en el puerto serial.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER no está soportado\n");
    /* El programa puede fallar en TCGETS/TCSETS con constantes Bnnn */
    exit(EXIT_FAILURE);
#else
    /* Declara la estructura tio, su tipo depende del soporte de ioctl */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "Uso: %s device [output [input]]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* Obtener las configuraciones actuales del puerto serial mediante el ioctl soportado */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* Cambiar la tasa de baudios cuando hay más argumentos */
    if (argc == 3 || argc == 4) {
        /* Limpiar la tasa de baudios de salida actual y rellenar un valor nuevo */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* Limpiar la tasa de baudios de entrada actual y rellenar un valor nuevo */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* Cuando el cuarto argumento no está proporcionado, reutiliza la tasa de baudios de salida */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* Configurar las nuevas configuraciones del puerto serial mediante el ioctl soportado */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* Y obtener los valores que realmente fueron configurados */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("tasa de baudios de salida: %u\n", tio.c_ospeed);
    printf("tasa de baudios de entrada: %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}

NAME

tcgets - get terminal attributes

LIBRARY

Terminal Control Library (libtermios, -ltermios)

SYNOPSIS

#include <termios.h>

int tcgets(int fd, struct termios *termios_p);

DESCRIPTION

The tcgets function retrieves the current settings for the terminal associated with the file descriptor fd and stores them in the structure pointed to by termios_p.

RETURN VALUE

On success, tcgets returns 0. On error, it returns -1 and sets errno to indicate the error.

ERRORS

EBADF
The file descriptor fd is not a valid open file descriptor.
ENOTTY
The file descriptor fd is not associated with a terminal.
EFAULT
termios_p is not a valid pointer.

HISTORY

The tcgets function first appeared in POSIX.1-1990.

CAVEATS

This function is part of the POSIX standard but may not be available on all systems. On Linux, the equivalent function is tcgetattr.

EXAMPLES

None.

SEE ALSO

tcsetattr(3), termios(3)
