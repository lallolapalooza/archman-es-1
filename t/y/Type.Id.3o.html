Identificadores de tipos.

Un identificador de tipo es un valor que denota un tipo. Dados dos identificadores de tipos, se pueden probar para Type.Id.provably_equal para demostrar que denotan el mismo tipo. Nótese que:

-Identificadores desiguales no implican tipos desiguales: un tipo dado puede ser denotado por más de un identificador.

-Los identificadores de tipos pueden ser empaquetados (marshalled), pero obtienen una identidad nueva y distinta al desempaquetar, por lo que las igualdades se pierden.

Ver un Type.Id.example de uso.

Identificadores de tipos

type !'a t

El tipo para identificadores del tipo 'a .

val make : unit -> 'a t

make () es un nuevo identificador de tipo.

val uid : 'a t -> int

uid id es un identificador único en tiempo de ejecución para id .

val provably_equal : 'a t -> 'b t -> ('a, 'b) Type.eq option

provably_equal i0 i1 es Some Equal si el identificador i0 es igual a i1 y None de lo contrario.

Ejemplo

Lo siguiente muestra cómo se pueden usar los identificadores de tipos para implementar un diccionario simple de clave-valor heterogéneo. En contraste con los valores de Map cuyas claves mapean a un único tipo homogéneo de valores, este diccionario puede asociar un tipo de valor diferente a cada clave.

(** Diccionarios heterogéneos. *)
module Dict : sig
  type t
  (** El tipo para diccionarios. *)
  type 'a key
  (** El tipo para claves que enlazan valores del tipo ['a]. *)
  val key : unit -> 'a key
  (** [key ()] es una nueva clave de diccionario. *)
  val empty : t
  (** [empty] es el diccionario vacío. *)
  val add : 'a key -> 'a -> t -> t
  (** [add k v d] es [d] con [k] enlazado a [v]. *)
  val remove : 'a key -> t -> t
  (** [remove k d] es [d] con el último enlace de [k] eliminado. *)
  val find : 'a key -> t -> 'a option
  (** [find k d] es el enlace de [k] en [d], si existe. *)
end = struct
  type 'a key = 'a Type.Id.t
  type binding = B : 'a key * 'a -> binding
  type t = (int * binding) list
  let key () = Type.Id.make ()
  let empty = []
  let add k v d = (Type.Id.uid k, B (k, v)) :: d
  let remove k d = List.remove_assoc (Type.Id.uid k) d
  let find : type a. a key -> t -> a option = fun k d ->
    match List.assoc_opt (Type.Id.uid k) d with
    | None -> None
    | Some (B (k', v)) ->
        match Type.Id.provably_equal k k' with
        | Some Type.Equal -> Some v
        | None -> assert false
end
