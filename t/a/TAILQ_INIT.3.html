NOMBRE

#include <sys/queue.h>
TAILQ_ENTRY(TYPE);

DESCRIPCIÓN

Estas macros definen y operan sobre colas de colas doblemente enlazadas.

En las definiciones de las macros, TYPE es el nombre de una estructura definida por el usuario, que debe contener un campo de tipo TAILQ_ENTRY, nombrado NAME. El argumento HEADNAME es el nombre de una estructura definida por el usuario que debe declararse usando la macro TAILQ_HEAD().

Creación

Una cola de colas está encabezada por una estructura definida por la macro TAILQ_HEAD(). Esta estructura contiene un par de punteros, uno al primer elemento de la cola y el otro al último elemento de la cola. Los elementos están doblemente enlazados para que un elemento arbitrario pueda ser eliminado sin recorrer la cola. Los nuevos elementos pueden ser añadidos a la cola después de un elemento existente, antes de un elemento existente, al principio de la cola o al final de la cola. Una estructura TAILQ_HEAD se declara de la siguiente manera:


TAILQ_HEAD(HEADNAME, TYPE) head;

donde struct HEADNAME es la estructura a definir, y struct TYPE es el tipo de los elementos a enlazar en la cola. Un puntero a la cabeza de la cola puede declararse posteriormente como:


struct HEADNAME *headp;

(Los nombres head y headp son seleccionables por el usuario.)

TAILQ_ENTRY() declara una estructura que conecta los elementos en la cola.

TAILQ_HEAD_INITIALIZER() se evalúa como un inicializador para la cola head.

TAILQ_INIT() inicializa la cola referenciada por

TAILQ_EMPTY() se evalúa como verdadero si no hay elementos en la cola. head.

Inserción

TAILQ_INSERT_HEAD() inserta el nuevo elemento elm al principio de la cola.

TAILQ_INSERT_TAIL() inserta el nuevo elemento elm al final de la cola.

TAILQ_INSERT_BEFORE() inserta el nuevo elemento elm antes del elemento listelm.

TAILQ_INSERT_AFTER() inserta el nuevo elemento elm después del elemento listelm.

Recorrido

TAILQ_FIRST() devuelve el primer elemento de la cola, o NULL si la cola está vacía.

TAILQ_LAST() devuelve el último elemento de la cola. Si la cola está vacía, el valor de retorno es NULL.

TAILQ_PREV() devuelve el elemento anterior de la cola, o NULL si este elemento es el primero.

TAILQ_NEXT() devuelve el siguiente elemento de la cola, o NULL si este elemento es el último.

TAILQ_FOREACH() recorre la cola referenciada por head en dirección hacia adelante, asignando cada elemento por vez a var. var se establece a NULL si el bucle finaliza normalmente, o si no había elementos.

TAILQ_FOREACH_REVERSE() recorre la cola referenciada por head en dirección hacia atrás, asignando cada elemento por vez a var.

Eliminación

TAILQ_REMOVE() elimina el elemento elm de la cola.

Otras características

TAILQ_CONCAT() concatena la cola encabezada por head2 al final de la encabezada por head1, eliminando todas las entradas de la primera.

VALOR DE RETORNO

TAILQ_EMPTY() devuelve un valor distinto de cero si la cola está vacía, y cero si la cola contiene al menos una entrada.

TAILQ_FIRST(), TAILQ_LAST(), TAILQ_PREV(), y TAILQ_NEXT() devuelven un puntero a la estructura TYPE primera, última, anterior o siguiente, respectivamente.

TAILQ_HEAD_INITIALIZER() devuelve un inicializador que puede ser asignado a la cola head.

ESTÁNDARES

BSD.

HISTORIA

4.4BSD.

ADVERTENCIAS

TAILQ_FOREACH() y TAILQ_FOREACH_REVERSE() no permiten que var sea eliminado o liberado dentro del bucle, ya que interferiría con el recorrido. TAILQ_FOREACH_SAFE() y TAILQ_FOREACH_REVERSE_SAFE(), que están presentes en los BSD pero no en glibc, corrigen esta limitación al permitir que var sea eliminado de forma segura de la lista y liberado desde dentro del bucle sin interferir con el recorrido.

EJEMPLOS

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    TAILQ_ENTRY(entry) entries;             /* Cola de colas */
};
TAILQ_HEAD(tailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct tailhead head;                   /* Cabeza de la cola de colas */
    int i;
    TAILQ_INIT(&head);                      /* Inicializar la cola */
    n1 = malloc(sizeof(struct entry));      /* Insertar al principio */
    TAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* Insertar al final */
    TAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* Insertar después */
    TAILQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* Insertar antes */
    TAILQ_INSERT_BEFORE(n2, n3, entries);
    TAILQ_REMOVE(&head, n2, entries);       /* Borrado */
    free(n2);
                                            /* Recorrido hacia adelante */
    i = 0;
    TAILQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* Recorrido hacia atrás */
    TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
        printf("%i\n", np->data);
                                            /* Borrado de cola de colas */
    n1

NOMBRE

TAILQ_INIT, TAILQ_INSERT_HEAD, TAILQ_INSERT_TAIL, TAILQ_INSERT_AFTER, TAILQ_INSERT_BEFORE, TAILQ_REMOVE, TAILQ_REPLACE, TAILQ_PREPEND — macros para manejo de listas enlazadas doblemente terminadas

BIBLIOTECA

#include <sys/queue.h>

SINOPSIS

    TAILQ_HEAD(list_name, type);
    TAILQ_HEAD_INITIALIZER(list_head);
    TAILQ_INIT(list_head);
    TAILQ_INSERT_HEAD(list_head, elm, field);
    TAILQ_INSERT_TAIL(list_head, elm, field);
    TAILQ_INSERT_AFTER(list_head, list_elm, elm, field);
    TAILQ_INSERT_BEFORE(list_head, list_elm, elm, field);
    TAILQ_REMOVE(list_head, elm, field);
    TAILQ_REPLACE(list_head, elm, new_elm, field);
    TAILQ_PREPEND(list_head, elm, field);

DESCRIPCIÓN

Estas macros proporcionan una implementación de listas doblemente enlazadas con colas. Una lista doblemente enlazada es una lista de estructuras enlazadas donde cada nodo contiene punteros tanto al nodo anterior como al siguiente. Las listas doblemente enlazadas se utilizan a menudo en el kernel de Linux y en muchas aplicaciones.

Las macros operan sobre una estructura de lista definida usando TAILQ_HEAD. La lista misma se representa como una estructura de puntero a un nodo de encabezado. Los nodos individuales son estructuras que contienen los datos y los punteros de enlace.

Los argumentos son los siguientes:

list_head
Un puntero a la estructura de encabezado de la lista.
list_name
El nombre que se le dará a la estructura de encabezado de la lista y al tipo de nodo.
type
El tipo de la estructura de nodo que se contendrá en la lista.
elm
Un puntero al elemento que se va a insertar o eliminar.
list_elm
Un puntero a un elemento existente en la lista. Se utiliza para insertar un nuevo elemento antes o después de él.
field
El nombre del campo dentro de la estructura del nodo que contiene los punteros de enlace. Este campo debe ser una estructura de tipo LIST_ENTRY o TAILQ_ENTRY.

TAILQ_HEAD(list_name, type) define dos tipos: list_name, que es la estructura de encabezado de la lista, y list_name##_entry, que es la estructura de nodo de la lista. Por ejemplo, TAILQ_HEAD(my_list_head, my_node) define struct my_list_head y struct my_list_head_entry.

TAILQ_HEAD_INITIALIZER(list_head) es una constante de inicialización que se puede utilizar para inicializar una lista antes de que se declare la variable de encabezado.

TAILQ_INIT(list_head) inicializa una lista vacía. Debe llamarse antes de que la lista se use para cualquier operación de inserción o eliminación.

TAILQ_INSERT_HEAD(list_head, elm, field) inserta el elemento elm al principio de la lista.

TAILQ_INSERT_TAIL(list_head, elm, field) inserta el elemento elm al final de la lista.

TAILQ_INSERT_AFTER(list_head, list_elm, elm, field) inserta el elemento elm después del elemento list_elm en la lista.

TAILQ_INSERT_BEFORE(list_head, list_elm, elm, field) inserta el elemento elm antes del elemento list_elm en la lista.

TAILQ_REMOVE(list_head, elm, field) elimina el elemento elm de la lista.

TAILQ_REPLACE(list_head, elm, new_elm, field) reemplaza el elemento elm en la lista con el elemento new_elm. El puntero elm se elimina de la lista y new_elm se inserta en su lugar. La ubicación de new_elm es la misma que la de elm.

TAILQ_PREPEND(list_head, elm, field) es un sinónimo de TAILQ_INSERT_HEAD.

VALOR DE RETORNO

Estas macros no devuelven ningún valor.

ESTÁNDARES

Estas macros se derivan de la implementación de listas enlazadas de BSD y se utilizan comúnmente en sistemas tipo Unix. No forman parte de ningún estándar POSIX formal.

HISTORIA

La implementación de listas enlazadas TAILQ apareció por primera vez en FreeBSD 2.2.

ADVERTENCIAS

Al utilizar estas macros, debe asegurarse de que los punteros de enlace en la estructura del nodo estén configurados correctamente. La manipulación incorrecta de los punteros de enlace puede provocar corrupción de la lista o accesos inválidos a la memoria.

La macro TAILQ_REMOVE asume que el elemento que se está eliminando está realmente en la lista. Si el elemento no está en la lista, el comportamiento es indefinido.

La macro TAILQ_REPLACE elimina el elm original y luego inserta new_elm. Si elm no está en la lista, el comportamiento es indefinido. new_elm no debe ser el mismo que elm.

EJEMPLOS

El siguiente ejemplo muestra cómo utilizar las macros TAILQ para crear y manipular una lista de enteros.

#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>

struct my_node {
    int data;
    TAILQ_ENTRY(my_node) entries;
};

TAILQ_HEAD(my_list_head, my_node);

int
main() {
    struct my_list_head head = TAILQ_HEAD_INITIALIZER(head);
    struct my_node *n1, *n2;
    int i;

    // Insertar elementos en la lista
    for (i = 0; i < 5; i++) {
        n1 = malloc(sizeof(struct my_node));
        n1->data = i;
        TAILQ_INSERT_TAIL(&head, n1, entries);
    }

    // Recorrer la lista e imprimir los elementos
    printf("Lista original:\n");
    TAILQ_FOREACH(n1, &head, entries) {
        printf("%d ", n1->data);
    }
    printf("\n");

    // Insertar un elemento en la cabeza
    n1 = malloc(sizeof(struct my_node));
    n1->data = -1;
    TAILQ_INSERT_HEAD(&head, n1, entries);

    // Recorrer la lista después de la inserción en la cabeza
    printf("Lista después de TAILQ_INSERT_HEAD:\n");
    TAILQ_FOREACH(n1, &head, entries) {
        printf("%d ", n1->data);
    }
    printf("\n");

    // Eliminar el segundo elemento (con valor 1)
    n1 = TAILQ_FIRST(&head); // n1 apunta a -1
    n2 = TAILQ_NEXT(n1, entries); // n2 apunta a 0
    n2 = TAILQ_NEXT(n2, entries); // n2 apunta a 1 (el que queremos eliminar)
    if (n2 != NULL) {
        printf("Eliminando elemento con valor %d\n", n2->data);
        TAILQ_REMOVE(&head, n2, entries);
        free(n2);
    }

    // Recorrer la lista después de la eliminación
    printf("Lista después de TAILQ_REMOVE:\n");
    TAILQ_FOREACH(n1, &head, entries) {
        printf("%d ", n1->data);
    }
    printf("\n");

    // Reemplazar el primer elemento (con valor -1) con un nuevo elemento
    n1 = TAILQ_FIRST(&head); // n1 apunta a -1
    if (n1 != NULL) {
        struct my_node *new_node = malloc(sizeof(struct my_node));
        new_node->data = 100;
        printf("Reemplazando elemento con valor %d con %d\n", n1->data, new_node->data);
        TAILQ_REPLACE(&head, n1, new_node, entries);
        free(n1); // Liberar el nodo antiguo
    }

    // Recorrer la lista después del reemplazo
    printf("Lista después de TAILQ_REPLACE:\n");
    TAILQ_FOREACH(n1, &head, entries) {
        printf("%d ", n1->data);
    }
    printf("\n");


    // Liberar toda la memoria asignada
    n1 = TAILQ_FIRST(&head);
    while (n1 != NULL) {
        n2 = TAILQ_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    TAILQ_INIT(&head);
    exit(EXIT_SUCCESS);
}
