module T : Map.OrderedType

include Map.S

val of_list : (key * 'a) list -> 'a t

val disjoint_union : ?eq:('a -> 'a -> bool) -> ?print:(Format.formatter -> 'a -> unit) -> 'a t -> 'a t -> 'a t

disjoint_union m1 m2 contiene todas las asociaciones de m1 y m2 . Si alguna asociación está presente en ambas y el valor asociado no es igual, se produce un Fatal_error

val union_right : 'a t -> 'a t -> 'a t

union_right m1 m2 contiene todas las asociaciones de m1 y m2 . Si alguna asociación está presente en ambas, se toma la de m2

val union_left : 'a t -> 'a t -> 'a t

union_left m1 m2 = union_right m2 m1

val union_merge : ('a -> 'a -> 'a) -> 'a t -> 'a t -> 'a t
