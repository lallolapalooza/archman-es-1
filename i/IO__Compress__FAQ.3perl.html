A continuación se muestra un módulo de compresión para Apache con mod_perl, llamado "Apache::GZip", tomado de http://perl.apache.org/docs/tutorials/tips/mod_perl_tricks/mod_perl_tricks.html#On_the_Fly_Compressio

package Apache::GZip;
#File: Apache::GZip.pm
use strict vars;
use Apache::Constants ':common';
use Compress::Zlib;
use IO::File;
use constant GZIP_MAGIC => 0x1f8b;
use constant OS_MAGIC => 0x03;
sub handler {
    my $r = shift;
    my ($fh,$gz);
    my $file = $r->filename;
    return DECLINED unless $fh=IO::File->new($file);
    $r->header_out('Content-Encoding'=>'gzip');
    $r->send_http_header;
    return OK if $r->header_only;
    tie *STDOUT,'Apache::GZip',$r;
    print($_) while <$fh>;
    untie *STDOUT;
    return OK;
}
sub TIEHANDLE {
    my($class,$r) = @_;
    # initialize a deflation stream
    my $d = deflateInit(-WindowBits=>-MAX_WBITS()) || return undef;
    # gzip header -- don't ask how I found out
    $r->print(pack("nccVcc",GZIP_MAGIC,Z_DEFLATED,0,time(),0,OS_MAGIC));
    return bless { r   => $r,
                   crc =>  crc32(undef),
                   d   => $d,
                   l   =>  0
                 },$class;
}
sub PRINT {
    my $self = shift;
    foreach (@_) {
      # deflate the data
      my $data = $self->{d}->deflate($_);
      $self->{r}->print($data);
      # keep track of its length and crc
      $self->{l} += length($_);
      $self->{crc} = crc32($_,$self->{crc});
    }
}
sub DESTROY {
   my $self = shift;
   # flush the output buffers
   my $data = $self->{d}->flush;
   $self->{r}->print($data);
   # print the CRC and the total length (uncompressed)
   $self->{r}->print(pack("LL",@{$self}{qw/crc l/}));
}
1;

NAME

IO::Compress::FAQ - Frequently Asked Questions about IO::Compress

DESCRIPTION

This document answers questions that users often ask about the IO::Compress suite of modules.

GENERAL

Q: What is IO::Compress?

A: IO::Compress is a collection of modules that provide an interface to several compression formats, including gzip, bzip2, lzma, xz, lzip, lzop, lzf and zstd. It works by providing filehandle-like objects that compress data as it is written to them, and uncompress data as it is read from them.

Q: What is the difference between IO::Compress::* and IO::Uncompress::*?

A: IO::Compress::* modules are used to compress data as you write it. IO::Uncompress::* modules are used to uncompress data as you read it.

Q: Which compression format should I use?

A: That depends on your needs.

  • gzip is widely supported and offers a good balance between speed and compression ratio.
  • bzip2 offers better compression than gzip, but is slower.
  • lzma and xz offer the best compression ratios, but are much slower than gzip or bzip2.
  • lzip offers compression ratios similar to bzip2 but uses a different algorithm (Burrows-Wheeler transform).
  • lzop is very fast but offers lower compression ratios.
  • lzf is also very fast and offers a reasonable compression ratio.
  • zstd is a modern algorithm offering excellent compression ratios and very fast compression and decompression speeds.

Q: How do I use IO::Compress?

A: It's simple. Just use one of the IO::Compress::* modules to create a compressed filehandle, and then write to it as you would any other filehandle.

    use IO::Compress::Gzip;
    my $gzip_file = IO::Compress::Gzip->new( "compressed.gz" )
        or die "Cannot open compressed.gz: $!\n";
    print $gzip_file "This is line 1.\n";
    print $gzip_file "This is line 2.\n";
    $gzip_file->close();

Q: How do I use IO::Uncompress?

A: Similar to IO::Compress, but you read from the uncompressed filehandle.

    use IO::Uncompress::Gunzip;
    my $gunzip_file = IO::Uncompress::Gunzip->new( "compressed.gz" )
        or die "Cannot open compressed.gz: $!\n";
    while ( my $line = <$gunzip_file> ) {
        print $line;
    }
    $gunzip_file->close();

Q: Can I compress to a scalar?

A: Yes, you can use the IO::Compress::* modules to compress data into a scalar variable. Just pass a reference to a scalar instead of a filename.

    use IO::Compress::Gzip;
    my $compressed_data;
    my $gzip_handle = IO::Compress::Gzip->new( \$compressed_data )
        or die "Cannot compress to scalar: $!\n";
    print $gzip_handle "Some data\n";
    $gzip_handle->close();
    # $compressed_data now holds the compressed data

Q: Can I uncompress from a scalar?

A: Yes, you can use the IO::Uncompress::* modules to uncompress data from a scalar variable. Just pass a reference to a scalar containing the compressed data.

    use IO::Uncompress::Gunzip;
    my $compressed_data = ...; # Assume this holds compressed data
    my $gunzip_handle = IO::Uncompress::Gunzip->new( \$compressed_data )
        or die "Cannot uncompress from scalar: $!\n";
    my $uncompressed_data = do { local $/ = undef; $gunzip_handle->getline };
    $gunzip_handle->close();
    # $uncompressed_data now holds the uncompressed data

Q: Can I chain compressions?

A: Yes, you can chain compressions by passing a compressed filehandle to another compress module.

    use IO::Compress::Gzip;
    use IO::Compress::Deflate;
    my $gzip_file = IO::Compress::Gzip->new( "compressed.gz" )
        or die "Cannot open compressed.gz: $!\n";
    my $deflate_handle = IO::Compress::Deflate->new( $gzip_file )
        or die "Cannot create Deflate handle: $!\n";
    print $deflate_handle "Some data\n";
    $deflate_handle->close(); # Close Deflate handle first
    $gzip_file->close();     # Then close Gzip handle

Q: What happens if I write to a closed handle?

A: Writing to a closed handle will cause an error.

Q: What happens if I read from a closed handle?

A: Reading from a closed handle will return undef (or an empty string if $/ is set to the empty string).

ZIP

The IO::Uncompress::Unzip module allows you to read files from a zip archive.

Q: How do I read a specific file from a zip archive?

A: You can specify the member you want to read when creating the Unzip object.

    use IO::Uncompress::Unzip;
    my $zip = IO::Uncompress::Unzip->new( "archive.zip", "file.txt" )
        or die "Cannot open archive.zip: $!\n";
    my $content = do { local $/; $zip->getline };
    $zip->close();

Q: How do I read all the files from a zip archive?

A: You can iterate over the members of the archive using the members() method.

    use IO::Uncompress::Unzip;
    my $zip = IO::Uncompress::Unzip->new( "archive.zip" )
        or die "Cannot open archive.zip: $!\n";
    for my $member ( $zip->members ) {
        my $fh = $zip->get_member_fh( $member );
        if ( $fh ) {
            my $content = do { local $/; $fh->getline };
            print "Contents of $member:\n$content\n";
            $zip->release_member_fh( $fh );
        }
    }
    $zip->close();

Q: How do I extract all the files from a zip archive?

A: You can use the extract() method.

    use IO::Uncompress::Unzip;
    my $zip = IO::Uncompress::Unzip->new( "archive.zip" )
        or die "Cannot open archive.zip: $!\n";
    $zip->extract() or die "Extraction failed: $!\n";
    $zip->close();

Q: How do I extract a specific file from a zip archive?

A: You can use the extract_member() method.

    use IO::Uncompress::Unzip;
    my $zip = IO::Uncompress::Unzip->new( "archive.zip" )
        or die "Cannot open archive.zip: $!\n";
    $zip->extract_member( "file.txt" ) or die "Extraction failed: $!\n";
    $zip->close();

GZIP

The IO::Compress::Gzip and IO::Uncompress::Gunzip modules handle gzip compression.

Q: How do I create a gzip compressed file?

A: Use IO::Compress::Gzip.

    use IO::Compress::Gzip;
    my $gz = IO::Compress::Gzip->new( "file.gz" )
        or die "Cannot open file.gz: $!\n";
    print $gz "This is some data.\n";
    $gz->close();

Q: How do I read a gzip compressed file?

A: Use IO::Uncompress::Gunzip.

    use IO::Uncompress::Gunzip;
    my $gz = IO::Uncompress::Gunzip->new( "file.gz" )
        or die "Cannot open file.gz: $!\n";
    while ( my $line = <$gz> ) {
        print $line;
    }
    $gz->close();

Q: Can I control the compression level?

A: Yes, you can pass the Level option to IO::Compress::Gzip->new().

    my $gz = IO::Compress::Gzip->new( "file.gz", Level => 9 ) # Best compression
    my $gz = IO::Compress::Gzip->new( "file.gz", Level => 1 ) # Fastest compression

Q: Can I control the compression strategy?

A: Yes, you can pass the Strategy option.

    my $gz = IO::Compress::Gzip->new( "file.gz", Strategy => 'Filtered' )

The available strategies are Default, Filtered, HuffmanOnly, RLE, and Fixed.

ZLIB

The IO::Compress::Zlib and IO::Uncompress::Inflate modules provide a low-level interface to the zlib compression library.

These modules are generally used when you need more control over the compression process, or when dealing with data that is not in a standard format like gzip.

Q: What is the difference between IO::Uncompress::Inflate and IO::Uncompress::Gunzip?

A: IO::Uncompress::Gunzip is a wrapper around IO::Uncompress::Inflate that specifically handles the gzip file format. IO::Uncompress::Inflate works with raw deflate streams.

Q: How do I read a raw deflate stream?

A: Use IO::Uncompress::RawInflate.

    use IO::Uncompress::RawInflate;
    my $raw = IO::Uncompress::RawInflate->new( $fh )
        or die "Cannot create RawInflate handle: $!\n";
    while ( my $line = <$raw> ) {
        print $line;
    }
    $raw->close();

Q: How do I write a raw deflate stream?

A: Use IO::Compress::RawDeflate.

    use IO::Compress::RawDeflate;
    my $raw = IO::Compress::RawDeflate->new( $fh )
        or die "Cannot create RawDeflate handle: $!\n";
    print $raw "This is some data.\n";
    $raw->close();

    my $bz2 = IO::Uncompress::Bunzip2->new( "file.bz2" )
        or die "Cannot open file.bz2: $!\n";
    while ( my $line = <$bz2> ) {
        print $line;
    }
    $bz2->close();

Q: Can I control the block size?

A: Yes, you can pass the BlockSize option to IO::Compress::Bzip2->new().

    my $bz2 = IO::Compress::Bzip2->new( "file.bz2", BlockSize => 9 ) # Largest block size

HTTP & NETWORK

Q: Can I use IO::Compress to send compressed data over a network?

A: Yes, you can pass a network socket filehandle to the IO::Compress::* modules.

    use IO::Socket::INET;
    use IO::Compress::Gzip;
    my $socket = IO::Socket::INET->new( PeerAddr => 'example.com', PeerPort => 80 )
        or die "Cannot connect: $!\n";
    my $gz = IO::Compress::Gzip->new( $socket )
        or die "Cannot create Gzip handle: $!\n";
    print $gz "GET / HTTP/1.0\r\n\r\n";
    $gz->close(); # Important: close the compress handle first
    close $socket;

Q: Can I use IO::Uncompress to receive compressed data over a network?

A: Yes, you can pass a network socket filehandle to the IO::Uncompress::* modules.

    use IO::Socket::INET;
    use IO::Uncompress::Gunzip;
    my $server = IO::Socket::INET->new( Listen => 10000, Reuse => 1 )
        or die "Cannot create socket: $!\n";
    my $client = $server->accept();
    my $gz = IO::Uncompress::Gunzip->new( $client )
        or die "Cannot create Gunzip handle: $!\n";
    while ( my $line = <$gz> ) {
        print $line;
    }
    $gz->close();
    close $client;
    close $server;

MISC

Q: How do I get more information about a compression error?

A: The error message will usually contain details about the error. You can also check the special variable $IO::Compress::Error or $IO::Uncompress::Error.

Q: Can I compress/uncompress data in memory without using filehandles?

A: Yes, you can use the compress and uncompress subroutines provided by the underlying compression modules (e.g., Zlib::MemGzip, Compress::Bzip2, etc.). However, the filehandle interface is generally easier to use.

Q: Can I compress/uncompress different types of data?

A: Yes, these modules work with any kind of binary data.

Using "InputLength" to uncompress data embedded in a larger file/buffer.

A fairly common use-case is where compressed data is embedded in a larger file/buffer and you want to read both.

As an example consider the structure of a zip file. This is a well-defined file format that mixes both compressed and uncompressed sections of data in a single file.

For the purposes of this discussion you can think of a zip file as sequence of compressed data streams, each of which is prefixed by an uncompressed local header. The local header contains information about the compressed data stream, including the name of the compressed file and, in particular, the length of the compressed data stream.

To illustrate how to use "InputLength" here is a script that walks a zip file and prints out how many lines are in each compressed file (if you intend write code to walking through a zip file for real see "Walking through a zip file" in IO::Uncompress::Unzip ). Also, although this example uses the zlib-based compression, the technique can be used by the other "IO::Uncompress::*" modules.

use strict;
use warnings;
use IO::File;
use IO::Uncompress::RawInflate qw(:all);
use constant ZIP_LOCAL_HDR_SIG  => 0x04034b50;
use constant ZIP_LOCAL_HDR_LENGTH => 30;
my $file = $ARGV[0] ;
my $fh = IO::File->new( "<$file" )
            or die "Cannot open '$file': $!\n";
while (1)
{
    my $sig;
    my $buffer;
    my $x ;
    ($x = $fh->read($buffer, ZIP_LOCAL_HDR_LENGTH)) == ZIP_LOCAL_HDR_LENGTH
        or die "Truncated file: $!\n";
    my $signature = unpack ("V", substr($buffer, 0, 4));
    last unless $signature == ZIP_LOCAL_HDR_SIG;
    # Read Local Header
    my $gpFlag             = unpack ("v", substr($buffer, 6, 2));
    my $compressedMethod   = unpack ("v", substr($buffer, 8, 2));
    my $compressedLength   = unpack ("V", substr($buffer, 18, 4));
    my $uncompressedLength = unpack ("V", substr($buffer, 22, 4));
    my $filename_length    = unpack ("v", substr($buffer, 26, 2));
    my $extra_length       = unpack ("v", substr($buffer, 28, 2));
    my $filename ;
    $fh->read($filename, $filename_length) == $filename_length
        or die "Truncated file\n";
    $fh->read($buffer, $extra_length) == $extra_length
        or die "Truncated file\n";
    if ($compressedMethod != 8 && $compressedMethod != 0)
    {
        warn "Skipping file '$filename' - not deflated $compressedMethod\n";
        $fh->read($buffer, $compressedLength) == $compressedLength
            or die "Truncated file\n";
        next;
    }
    if ($compressedMethod == 0 && $gpFlag & 8 == 8)
    {
        die "Streamed Stored not supported for '$filename'\n";
    }
    next if $compressedLength == 0;
    # Done reading the Local Header
    my $inf = IO::Uncompress::RawInflate->new( $fh,
                        Transparent => 1,
                        InputLength => $compressedLength )
      or die "Cannot uncompress $file [$filename]: $RawInflateError\n"  ;
    my $line_count = 0;
    while (<$inf>)
    {
        ++ $line_count;
    }
    print "$filename: $line_count\n";
}

The majority of the code above is concerned with reading the zip local header data. The code that I want to focus on is at the bottom.

while (1) {
    # read local zip header data
    # get $filename
    # get $compressedLength
    my $inf = IO::Uncompress::RawInflate->new( $fh,
                        Transparent => 1,
                        InputLength => $compressedLength )
      or die "Cannot uncompress $file [$filename]: $RawInflateError\n"  ;
    my $line_count = 0;
    while (<$inf>)
    {
        ++ $line_count;
    }
    print "$filename: $line_count\n";
}

The call to "IO::Uncompress::RawInflate" creates a new filehandle $inf that can be used to read from the parent filehandle $fh, uncompressing it as it goes. The use of the "InputLength" option will guarantee that at most $compressedLength bytes of compressed data will be read from the $fh filehandle (The only exception is for an error case like a truncated file or a corrupt data stream).

This means that once RawInflate is finished $fh will be left at the byte directly after the compressed data stream.

Now consider what the code looks like without "InputLength"

while (1) {
    # read local zip header data
    # get $filename
    # get $compressedLength
    # read all the compressed data into $data
    read($fh, $data, $compressedLength);
    my $inf = IO::Uncompress::RawInflate->new( \$data,
                        Transparent => 1 )
      or die "Cannot uncompress $file [$filename]: $RawInflateError\n"  ;
    my $line_count = 0;
    while (<$inf>)
    {
        ++ $line_count;
    }
    print "$filename: $line_count\n";
}

The difference here is the addition of the temporary variable $data. This is used to store a copy of the compressed data while it is being uncompressed.

If you know that $compressedLength isn't that big then using temporary storage won't be a problem. But if $compressedLength is very large or you are writing an application that other people will use, and so have no idea how big $compressedLength will be, it could be an issue.

Using "InputLength" avoids the use of temporary storage and means the application can cope with large compressed data streams.

One final point -- obviously "InputLength" can only be used whenever you know the length of the compressed data beforehand, like here with a zip file.
