XcmsColor(3) FUNCIONES DE XLIB XcmsColor(3)

NOMBRE

XcmsColor, XcmsRGB, XcmsRGBi, XcmsCIEXYZ, XcmsCIEuvY, XcmsCIExyY, XcmsCIELab, XcmsCIELuv, XcmsTekHVC, XcmsPad - Estructura de color Xcms

ESTRUCTURAS

La estructura para XcmsColor contiene:

typedef unsigned long XcmsColorFormat;			/* Formato de especificación de color */
typedef struct {
	union {
		XcmsRGB RGB;
		XcmsRGBi RGBi;
		XcmsCIEXYZ CIEXYZ;
		XcmsCIEuvY CIEuvY;
		XcmsCIExyY CIExyY;
		XcmsCIELab CIELab;
		XcmsCIELuv CIELuv;
		XcmsTekHVC TekHVC;
		XcmsPad Pad;
	} spec;
	unsigned long pixel;
	XcmsColorFormat format;
} XcmsColor;			/* Estructura de color Xcms */
typedef double XcmsFloat;
typedef struct {
	unsigned short red;	/* 0x0000 a 0xffff */
	unsigned short green;	/* 0x0000 a 0xffff */
	unsigned short blue;	/* 0x0000 a 0xffff */
} XcmsRGB;		/* Dispositivo RGB */
typedef struct {
	XcmsFloat red;	/* 0.0 a 1.0 */
	XcmsFloat green;	/* 0.0 a 1.0 */
	XcmsFloat blue;	/* 0.0 a 1.0 */
} XcmsRGBi;		/* Intensidad RGB */
typedef struct {
	XcmsFloat X;
	XcmsFloat Y;	/* 0.0 a 1.0 */
	XcmsFloat Z;
} XcmsCIEXYZ;		/* CIE XYZ */
typedef struct {
	XcmsFloat u_prime;	/* 0.0 a ~0.6 */
	XcmsFloat v_prime;	/* 0.0 a ~0.6 */
	XcmsFloat Y; 	/* 0.0 a 1.0 */
} XcmsCIEuvY;		/* CIE u'v'Y */
typedef struct {
	XcmsFloat x; 	/* 0.0 a ~.75 */
	XcmsFloat y; 	/* 0.0 a ~.85 */
	XcmsFloat Y; 	/* 0.0 a 1.0 */
} XcmsCIExyY;		/* CIE xyY */
typedef struct {
	XcmsFloat L_star; 	/* 0.0 a 100.0 */
	XcmsFloat a_star;
	XcmsFloat b_star;
} XcmsCIELab;		/* CIE L*a*b* */
typedef struct {
	XcmsFloat L_star; 	/* 0.0 a 100.0 */
	XcmsFloat u_star;
	XcmsFloat v_star;
} XcmsCIELuv;		/* CIE L*u*v* */
typedef struct {
	XcmsFloat H; 	/* 0.0 a 360.0 */
	XcmsFloat V; 	/* 0.0 a 100.0 */
	XcmsFloat C; 	/* 0.0 a 100.0 */
} XcmsTekHVC;		/* TekHVC */
typedef struct {
	XcmsFloat pad0;
	XcmsFloat pad1;
	XcmsFloat pad2;
	XcmsFloat pad3;
} XcmsPad;		/* cuatro doubles */

DESCRIPCIÓN

La estructura XcmsColor contiene una unión de subestructuras, cada una soportando la codificación de especificación de color para un espacio de color particular.
