val reset : unit -> unit

borra toda la información de perfil registrada

val record_call : ?accumulate:bool -> string -> (unit -> 'a) -> 'a

record_call pass f llama a f y registra su información de perfil.

val record : ?accumulate:bool -> string -> ('a -> 'b) -> 'a -> 'b

record pass f arg registra la información de perfil de f arg
