NOMBRE

Pango::Language - envoltorio para PangoLanguage

JERARQUÍA

Glib::Boxed
+----Pango::Language

MÉTODOS

language = Pango::Language->get_default

Desde: pango 1.16

dirección = Pango->find_base_dir ($texto)

•
$texto (cadena)

Desde: pango 1.4

language = Pango::Language->from_string ($language)

•
$language (cadena)

booleano = $language->includes_script ($script)

•
$script (Pango::Script)

Desde: pango 1.4

booleano = $language->matches ($range_list)

•
$range_list (cadena)

cadena = $language->to_string
