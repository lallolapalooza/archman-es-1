DESCRIPTION

Estas funciones sirven para extender y/o modificar el comportamiento de las funciones de la familia printf(3).

register_printf_specifier()

Esta función registra un especificador de conversión personalizado para las funciones de la familia printf(3).

spec
El carácter que se utilizará como especificador de conversión en la cadena de formato.
func
La función de callback que se ejecutará por parte de las funciones de la familia printf(3) para formatear los argumentos de entrada en el flujo de salida stream.
stream
El flujo de salida donde se imprimirá el resultado de la formateo. Este flujo representa transparentemente el resultado, incluso en el caso de funciones que escriben en un string.
info
La estructura que contiene la información de contexto, incluyendo los modificadores especificados en la cadena de formato. Esto contiene los mismos valores que en arginfo.
args
Array de punteros a los argumentos de la función de la familia printf(3)-like.
arginfo
La función de callback que se ejecutará por parte de las funciones de la familia printf(3) para saber cuántos argumentos se deben parsear para el especificador personalizado y también sus tipos.
info
La estructura que contiene la información de contexto, incluyendo los modificadores especificados en la cadena de formato. Esto contiene los mismos valores que en func.
n
El número de argumentos restantes a parsear.
argtypes
Esta matriz debe ser establecida para definir el tipo de cada uno de los argumentos a parsear. Cada elemento en la matriz representa un de los argumentos a parsear, en el mismo orden que se pasan a la función printf(3)-like. Cada elemento debe ser establecido a un tipo base (PA_*) desde el enum anterior, o un tipo personalizado, y opcionalmente OR con un modificador de longitud (PA_FLAG_*).

El tipo se determina usando uno de los siguientes constantes:

PA_INT
int.
PA_CHAR
int, convertido a char.
PA_WCHAR
wchar_t.
PA_STRING
const char*, una cadena terminada en '\0'.
PA_WSTRING
const wchar_t*, una cadena de caracteres ancho terminada en '\0'.
PA_POINTER
void*.
PA_FLOAT
float.
PA_DOUBLE
double.
PA_LAST
TODO.
size
Para tipos personalizados, también debe especificarse el tamaño del tipo (en bytes) a través de esta matriz. De lo contrario, dejarlo sin usar.

arginfo se llama antes de func, y prepara información necesaria para llamar a func.

RETURN VALUE

register_printf_specifier(), register_printf_modifier() y register_printf_type() retornan 0 en caso de éxito, o -1 en caso de error.

Callbacks

La función de callback de tipo printf_function debe retornar el número de caracteres escritos, o -1 en error.

La función de callback de tipo printf_arginfo_size_function debe retornar el número de argumentos a parsear por este especificador. También pasa información sobre el tipo de esos argumentos a la llamada mediante argtypes. En error, debe retornar -1.

ERRORS

EINVAL
El especificador no era un carácter válido.

HISTORIA

register_printf_function(3) es una función antigua similar a register_printf_specifier(), y ahora está desactivada. Esta función no puede manejar tipos definidos por el usuario.

register_printf_specifier() sustituye a register_printf_function(3).

EJEMPLOS

El siguiente programa de ejemplo registra los especificadores 'b' y 'B' para imprimir enteros en formato binario, reflejando las reglas de otros convertidores de tipo sin signo como 'x' y 'u'. Esto se puede utilizar antes de C23.

/* Este código está en el dominio público */
#include <err.h>
#include <limits.h>
#include <stddef.h>
#include <stdint.h>;
#include <stdio.h>;
#include <stdlib.h>;
#include <string.h>;
#include <sys/param.h>;
#include <printf.h>;
#define GROUP_SEP  '\'';
struct Printf_Pad {
    char    ch;
    size_t  len;
};
static int b_printf(FILE *stream, const struct printf_info *info,
                    const void *const args[]);
static int b_arginf_sz(const struct printf_info *info,
                    size_t n, int argtypes[n], int size[n]);
static uintmax_t b_value(const struct printf_info *info,
                    const void *arg);
static size_t b_bin_repr(char bin[UINTMAX_WIDTH],
                    const struct printf_info *info, const void *arg);
static size_t b_bin_len(const struct printf_info *info,
                    ptrdiff_t min_len);
static size_t b_pad_len(const struct printf_info *info,
                    ptrdiff_t bin_len);
static ssize_t b_print_prefix(FILE *stream,
                    const struct printf_info *info);
static ssize_t b_pad_zeros(FILE *stream, const struct printf_info *info,
                    ptrdiff_t min_len);
static ssize_t b_print_number(FILE *stream,
                    const struct printf_info *info,
                    const char bin[UINTMAX_WIDTH],
                    size_t min_len, size_t bin_len);
static char pad_ch(const struct printf_info *info);
static ssize_t pad_spaces(FILE *stream, size_t pad_len);
int
main(void)
{
    if (register_printf_specifier('b', b_printf, b_arginf_sz) == -1)
        err(EXIT_FAILURE, "register_printf_specifier('b', ...)");
    if (register_printf_specifier('B', b_printf, b_arginf_sz) == -1)
        err(EXIT_FAILURE, "register_printf_specifier('B', ...)");
    printf("....----....----....----....----\n");
    printf("%llb;\n", 0x5Ellu);
    printf("%lB;\n", 0x5Elu);
    printf("%b;\n", 0x5Eu);
    printf("%hB;\n", 0x5Eu);
    printf("%hhb;\n", 0x5Eu);
    printf("%jb;\n", (uintmax_t)0x5E);
    printf("%zb;\n", (size_t)0x5E);
    printf("....----....----....----....----\n");
    printf("%#b;\n", 0x5Eu);
    printf("%#B;\n", 0x5Eu);
    printf("....----....----....----....----\n");
    printf("%10b;\n", 0x5Eu);
    printf("%010b;\n", 0x5Eu);
    printf("%.10b;\n", 0x5Eu);

    printf("....----....----....----....----\n");
    printf("%'B;\n", 0x5Eu);
    printf("....----....----....----....----\n");
    printf("....----....----....----....----\n");
    printf("%#16.12b;\n", 0xAB);
    printf("%-#'20.12b;\n", 0xAB);

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* Alineamiento con espacios (derecha) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Prefijo "0b"/"0B" */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Alineamiento con ceros (izquierda) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Imprimir número (incluyendo ceros para llenar la precisión) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* short y char se promocionan a int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* Imprimir ceros para llenar la precisión */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* Imprimir número */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
