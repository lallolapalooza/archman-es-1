PA_CHAR(3const) PA_CHAR(3const)

NOMBRE

PA_CHAR, PA_INT, PA_POINTER, PA_STRING, PA_WSTRING, PA_FLOAT, PA_DOUBLE, PA_LONG, PA_LONG_LONG, PA_LONG_DOUBLE, PA_PTR, PA_WCHAR - tipos de argumento para las funciones de impresión personalizadas

SINTAXIS

#include <printf.h>
enum {
    PA_INT,
    PA_CHAR,
    PA_WCHAR,
    PA_STRING,
    PA_WSTRING,
    PA_POINTER,
    PA_FLOAT,
    PA_DOUBLE,
    PA_LAST
};
#define PA_FLAG_LONG         (1 << 8)
#define PA_FLAG_LONG_LONG    (1 << 9)
#define PA_FLAG_SHORT        (1 << 10)
#define PA_FLAG_PTR          (1 << 11)
#define PA_FLAG_LONG_DOUBLE  (1 << 12)

typedef ... printf_arginfo_size_function(...);
... printf_arginfo_size_function(..., int argtypes[], int size[]);

DESCRIPCIÓN

Estas constantes se utilizan para especificar el tipo de los argumentos que se pasan a una función de impresión personalizada, como las que se registran con la función register_printf_specifier(3). La función register_printf_specifier(3) llama a una función de callback proporcionada por el usuario (tipo printf_arginfo_size_function) para determinar el número y el tipo de argumentos que espera un especificador de formato personalizado.

Los tipos base son:

PA_INT
Un entero con signo.
PA_CHAR
Un carácter, tratado como un entero.
PA_WCHAR
Un carácter ancho (normalmente de 16 o 32 bits).
PA_STRING
Un puntero a una cadena de caracteres terminada en nulo (char*).
PA_WSTRING
Un puntero a una cadena de caracteres anchos terminada en nulo (wchar_t*).
PA_POINTER
Un puntero genérico (void*).
PA_FLOAT
Un número de punto flotante de precisión simple (float).
PA_DOUBLE
Un número de punto flotante de doble precisión (double).
PA_LAST
Se utiliza para indicar el final de la lista de tipos de argumento conocidos.

Los siguientes flags se pueden aplicar a los tipos base utilizando el operador OR a nivel de bits (|) para indicar modificadores de longitud:

PA_FLAG_LONG
Indica que el argumento es un tipo largo (por ejemplo, long int, long double).
PA_FLAG_LONG_LONG
Indica que el argumento es un tipo de longitud doble (por ejemplo, long long int).
PA_FLAG_SHORT
Indica que el argumento es un tipo corto (por ejemplo, short int, signed char).
PA_FLAG_PTR
Indica que el tipo de argumento es un puntero. Esto es útil para distinguir entre diferentes tipos de punteros si es necesario.
PA_FLAG_LONG_DOUBLE
Indica que el argumento es un long double.

La función register_printf_specifier(3) llama a la función printf_arginfo_size_function proporcionada para determinar cuántos argumentos esperar y sus tipos. El resultado de esta función indica el número total de argumentos a extraer de la lista de argumentos variables. Los tipos de estos argumentos se especifican en el array argtypes. El array size se puede utilizar para especificar el tamaño de tipos de datos personalizados. Por ejemplo, si se define un tipo personalizado para un entero de 128 bits, el campo correspondiente en size contendría el tamaño en bytes de ese tipo.

EJEMPLO

Para registrar un especificador de formato personalizado 'H' que imprime un entero largo sin signo, podría usar:

int my_printf_arginfo(const struct printf_info *info, size_t n,
                      int argtypes[n], int size[n])
{
    if (n >= 1) {
        argtypes[0] = PA_INT | PA_FLAG_LONG; // unsigned long int
        // No es necesario especificar el tamaño ya que es un tipo estándar
    }
    return 1; // Espera 1 argumento
}

int my_printf_func(FILE *stream, const struct printf_info *info,
                   const void *const args[])
{
    unsigned long val = *(const unsigned long *)args[0];
    // Lógica de impresión para unsigned long
    return fprintf(stream, "%lu", val);
}

// ... en algún lugar de la inicialización ...
register_printf_specifier('H', my_printf_func, my_printf_arginfo);

En este ejemplo, my_printf_arginfo informa que se espera un argumento de tipo unsigned long (combinando PA_INT y PA_FLAG_LONG) y que es el único argumento. Luego, my_printf_func recibe este argumento y lo imprime en el flujo especificado.

VER TAMBIÉN

printf(3), register_printf_specifier(3)

STANDARDS

GNU.

HISTORY

register_printf_function(3) es una función antigua similar a register_printf_specifier(), y está obsoleta. Esa función no puede manejar tipos definidos por el usuario.

register_printf_specifier() reemplaza a register_printf_function(3).

EJEMPLOS

El siguiente programa de ejemplo registra los especificadores 'b' y 'B' para imprimir enteros en formato binario, reflejando las reglas para otros especificadores de conversión sin signo como 'x' y 'u'. Esto se puede utilizar para imprimir en binario antes de C23.

#include <limits.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/param.h>
#include <printf.h>
#define GROUP_SEP  '\'
struct Printf_Pad {
    char    ch;
    size_t  len;
};

                    const void *const args[]);

                    const struct printf_info *info, const void *arg);

                    ptrdiff_t min_len);
static size_t b_pad_len(const struct printf_info *info,
                    ptrdiff_t bin_len);
static ssize_t b_print_prefix(FILE *stream,
                    const struct printf_info *info);
static ssize_t b_pad_zeros(FILE *stream, const struct printf_info *info,

                    const char bin[UINTMAX_WIDTH],
                    size_t min_len, size_t bin_len);
static char b_pad_ch(const struct printf_info *info);

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = b_pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* Relleno con ' ' (alineado a la derecha) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Prefijo "0b"/"0B" */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Relleno con '0' */
    if (pad.ch == '0') {
        tmp = b_pad_zeros(stream, info, min_len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Imprimir número (incluyendo ceros a la izquierda para completar la precisión) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    /* Relleno con ' ' (alineado a la izquierda) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* short y char se promocionan a int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* Imprimir ceros a la izquierda para completar la precisión */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* Imprimir número */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
b_pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
