tipo constante = {
pconst_desc : constant_desc ;
pconst_loc : Location.t ;
}

tipo constant_desc =
| Pconst_integer de string * char option
(* Constantes enteras como 3 3l 3L 3n .

Los sufijos [g-z][G-Z] son aceptados por el parser. Los sufijos excepto 'l' , 'L' y 'n' son rechazados por el verificador de tipos
*)
| Pconst_char de char
(* Caracteres como 'c' .
*)
| Pconst_string de string * Location.t * string option
(* Constantes de cadena como "constant" o {delim|other const|delim} .

La ubicación cubre el contenido de la cadena, sin los delimitadores.
*)
| Pconst_float de string * char option
(* Constantes flotantes como 3.4 , 2e5 o 1.4e-4 .

Los sufijos g-z G-Z son aceptados por el parser. Los sufijos son rechazados por el verificador de tipos.
*)

tipo location_stack = Location.t list

Puntos de extensión

tipo atributo = {
attr_name : string Asttypes.loc ;
attr_payload : payload ;
attr_loc : Location.t ;
}

Atributos como [@id ARG] y [@@id ARG] .

Contenedores de metadatos que se pasan alrededor del AST. El compilador ignora los atributos desconocidos.

tipo extension = string Asttypes.loc * payload

Puntos de extensión como [%id ARG] y [%%id ARG] .

Sublenguaje de plantilla - rechazado por el verificador de tipos.

tipo attributes = list de atributos

tipo payload =
| PStr de estructura
| PSig de signatura
(* : SIG en un atributo o un punto de extensión
*)
| PTyp de tipo_base
(* : T en un atributo o un punto de extensión
*)
| PPat de patrón * opción de expresión
(* ? P o ? P cuando E en un atributo o un punto de extensión
*)

Expresiones de tipo

tipo tipo_base = {
ptyp_desc : tipo_base_desc ;
ptyp_loc : Location.t ;
ptyp_loc_stack : location_stack ;
ptyp_attributes : attributes ; (* ... [@id1] [@id2]


*)
}

tipo tipo_base_desc =
| Ptyp_any (* _


*)
| Ptyp_var de string
(* Variables de tipo como 'a


*)
| Ptyp_arrow de Asttypes.arg_label * tipo_base * tipo_base
(* Ptyp_arrow(lbl, T1, T2) representa:

- T1 -> T2 cuando lbl es Asttypes.arg_label.Nolabel ,

- ~l:T1 -> T2 cuando lbl es Asttypes.arg_label.Labelled ,

- ?l:T1 -> T2 cuando lbl es Asttypes.arg_label.Optional .


*)
| Ptyp_tuple de (string option * tipo_base) list
(* Ptyp_tuple(tl) representa un tipo producto:

- T1 * ... * Tn cuando tl es (None, T1); ...; (None, Tn)

- L1:T1 * ... * Ln:Tn cuando tl es (Some L1, T1); ...; (Some Ln, Tn)

- Una mezcla, por ejemplo L1:T1 * T2 cuando tl es (Some L1, T1); (None, T2)

Invariante: n >= 2 .
*)
| Ptyp_constr de Longident.t Asttypes.loc * tipo_base list
(* Ptyp_constr(lident, l) representa:

- tconstr cuando l=[] ,

- T tconstr cuando l=[T] ,

- (T1, ..., Tn) tconstr cuando l=[T1 ; ... ; Tn] .


*)
| Ptyp_object de list de campo_objetivo * Asttypes.closed_flag
(* Ptyp_object([ l1:T1; ...; ln:Tn ], flag) representa:

- < l1:T1; ...; ln:Tn > cuando flag es Asttypes.closed_flag.Closed ,

- < l1:T1; ...; ln:Tn; .. > cuando flag es Asttypes.closed_flag.Open .


*)
| Ptyp_class de Longident.t Asttypes.loc * tipo_base list
(* Ptyp_class(tconstr, l) representa:

- #tconstr cuando l=[] ,

- T #tconstr cuando l=[T] ,

- (T1, ..., Tn) #tconstr cuando l=[T1 ; ... ; Tn] .


*)
| Ptyp_alias de tipo_base * string Asttypes.loc
(* T as 'a .
*)
| Ptyp_variant de list de campo_variante * Asttypes.closed_flag * list de etiqueta
(* Ptyp_variant([`A;`B], flag, labels) representa:

- [ `A|`B ] cuando flag es Asttypes.closed_flag.Closed , y labels es None ,

- [> `A|`B ] cuando flag es Asttypes.closed_flag.Open , y labels es None ,

- [< `A|`B ] cuando flag es Asttypes.closed_flag.Closed , y labels es Some [] ,

- [< `A|`B > `X `Y ] cuando flag es Asttypes.closed_flag.Closed , y labels es Some ["X";"Y"] .


*)
| Ptyp_poly de list de string Asttypes.loc * tipo_base
(* 'a1 ... 'an. T

Puede aparecer solo en el siguiente contexto:

- Como Parsetree.core_type de un nodo Parsetree.pattern_desc.Ppat_constraint que corresponde a una restricción en un bind de let:

let x : 'a1 ... 'an. T = e ...

- Como Parsetree.class_field_kind.Cfk_virtual para métodos (no valores).

- Como Parsetree.core_type de un nodo Parsetree.class_type_field_desc.Pctf_method .

- Como Parsetree.core_type de un nodo Parsetree.expression_desc.Pexp_poly .

- Como Parsetree.label_declaration.pld_type de un nodo Parsetree.label_declaration .

- Como Parsetree.core_type de un nodo Parsetree.core_type_desc.Ptyp_object .

- Como Parsetree.value_description.pval_type de un nodo Parsetree.value_description .


*)
| Ptyp_package de package_type
(* (module S) .
*)
| Ptyp_open de Longident.t Asttypes.loc * tipo_base
(* M.(T)


*)
| Ptyp_extension de extension
(* [%id] .
*)

tipo package_type = {
ppt_path : Longident.t Asttypes.loc ;
ppt_cstrs : (Longident.t Asttypes.loc * tipo_base) list ;
ppt_loc : Location.t ; es true y l es [] ,

- `A de T cuando b es false y l es [T] ,

- `A de T1 & .. & Tn cuando b es false y l es [T1;...Tn] ,

- `A de & T1 & .. & Tn cuando b es true y l es [T1;...Tn] .

- El campo bool es verdadero si la etiqueta contiene un constructor constante (vacío).

- & ocurre cuando se usan varios tipos para el mismo constructor (ver 4.2 en el manual)


*)
| Rinherit de core_type
(* [ | t ]


*)

pof_desc : object_field_desc ;
pof_loc : Location.t ;
pof_attributes : attributes ;
}

tipo object_field_desc =
| Otag de Asttypes.label Asttypes.loc * core_type
| Oinherit de core_type

tipo pattern = {
ppat_desc : pattern_desc ;
ppat_loc : Location.t ;
ppat_loc_stack : location_stack ;
ppat_attributes : attributes ; (* ... [@id1] [@id2]


*)
}

tipo pattern_desc =
| Ppat_any (* El patrón _ .
*)
| Ppat_var de string Asttypes.loc
(* Un patrón de variable como x


*)
| Ppat_alias de pattern * string Asttypes.loc
(* Un patrón de alias como P como 'a


*)
| Ppat_constant de constant
(* Patrones como 1 , 'a' , "true" , 1.0 , 1l , 1L , 1n


*)
| Ppat_interval de constant * constant
(* Patrones como 'a'..'z' .

Otras formas de intervalo son reconocidas por el analizador sintáctico pero rechazadas por el verificador de tipos.
*)
| Ppat_tuple de (string option * pattern) list * Asttypes.closed_flag
(* Ppat_tuple(pl, Closed) representa

- (P1, ..., Pn) cuando pl es (None, P1); ...; (None, Pn)

- (~L1:P1, ..., ~Ln:Pn) cuando pl es (Some L1, P1); ...; (Some Ln, Pn)

-Una mezcla, por ejemplo, (~L1:P1, P2) cuando pl es (Some L1, P1); (None, P2)

Ppat_tuple(pl, Open) es similar, pero indica que el patrón además termina en un .. .

Invariante:

-Si está cerrado, n >= 2 .

-Si está abierto, n >= 1 .


*)
| Ppat_construct de Longident.t Asttypes.loc * (string Asttypes.loc list * pattern) option
(* Ppat_construct(C, args) representa:

- C cuando args es None ,

- C P cuando args es Some ([], P)

- C (P1, ..., Pn) cuando args es Some ([], Ppat_tuple [P1; ...; Pn])

- C (tipo a b) P cuando args es Some ([a; b], P)


*)
| Ppat_variant de Asttypes.label * pattern option
(* Ppat_variant(`A, pat) representa:

- `A cuando pat es None ,

- `A P cuando pat es Some P


*)
| Ppat_record de (Longident.t Asttypes.loc * pattern) list * Asttypes.closed_flag
(* Ppat_record([(l1, P1) ; ... ; (ln, Pn)], flag) representa:

- { l1=P1; ...; ln=Pn } cuando flag es Asttypes.closed_flag.Closed

- { l1=P1; ...; ln=Pn; _} cuando flag es Asttypes.closed_flag.Open

Invariante: n > 0


*)
| Ppat_array de pattern list
(* Patrón [| P1; ...; Pn |]


*)
| Ppat_or de pattern * pattern
(* Patrón P1 | P2


*)
| Ppat_constraint de pattern * core_type
(* Patrón (P : T)


*)
| Ppat_type de Longident.t Asttypes.loc
(* Patrón #tconst


*)
| Ppat_lazy de pattern
(* Patrón lazy P


*)
| Ppat_unpack de string option Asttypes.loc
(* Ppat_unpack(s) representa:

- (module P) cuando s es Some "P"

- (module _) cuando s es None

Nota: (module P : S) se representa como Ppat_constraint(Ppat_unpack(Some "P"), Ptyp_package S)


*)
| Ppat_exception de pattern
(* Patrón exception P


*)
| Ppat_effect de pattern * pattern
| Ppat_extension de extension
(* Patrón [%id]


*)
| Ppat_open de Longident.t Asttypes.loc * pattern
(* Patrón M.(P)


*)

Expresiones de valor

tipo expression = {
pexp_desc : expression_desc ;
pexp_loc : Location.t ;
pexp_loc_stack : location_stack ;
pexp_attributes : attributes ; (* ... [@id1] [@id2]


*)
}

tipo expression_desc =
| Pexp_ident de Longident.t Asttypes.loc
(* Identificadores como x y M.x


*)
| Pexp_constant de constant
(* Constantes de expresiones como 1 , 'a' , "true" , 1.0 , 1l , 1L , 1n


*)
| Pexp_let de Asttypes.rec_flag * value_binding list * expression
(* Pexp_let(flag, [(P1,E1) ; ... ; (Pn,En)], E) representa:

- let P1 = E1 and ... and Pn = EN in E cuando flag es Asttypes.rec_flag.Nonrecursive ,

- let rec P1 = E1 and ... and Pn = EN in E cuando flag es Asttypes.rec_flag.Recursive .


*)
| Pexp_function de function_param list * type_constraint option * function_body
(* Pexp_function ([P1; ...; Pn], C, body) representa cualquier construcción que involucre fun o function , incluyendo:

- fun P1 ... Pn -> E cuando body = Pfunction_body E

- fun P1 ... Pn -> function p1 -> e1 | ... | pm -> em cuando body = Pfunction_cases [ p1 -> e1; ...; pm -> em ]

C representa una restricción de tipo o una coerción colocada inmediatamente antes de la flecha, por ejemplo, fun P1 ... Pn : ty -> ... cuando C = Some (Pconstraint ty) .

Una función debe tener parámetros: en Pexp_function (params, _, body) , si params no contiene un Pparam_val _ , body debe ser Pfunction_cases _ .
*)
| Pexp_apply de expression * (Asttypes.arg_label * expression) list
(* Pexp_apply(E0, [(l1, E1) ; ... ; (ln, En)]) representa E0 ~l1:E1 ... ~ln:En

li puede ser Asttypes.arg_label.Nolabel (argumento sin etiqueta), Asttypes.arg_label.Labelled (argumentos etiquetados) o Asttypes.arg_label.Optional (argumento opcional).

Invariante: n > 0


*)
| Pexp_match de expression * case list
(* match E0 with P1 -> E1 | ... | Pn -> En


*)
| Pexp_try de expression * case list
(* try E0 with P1 -> E1 | ... | Pn -> En


*)
| Pexp_tuple de (string option * expression) list
(* Pexp_tuple(el) representa

- (E1, ..., En) cuando el es (None, E1); ...; (None, En)

- (~L1:E1, ..., ~Ln:En) cuando el es (Some L1, E1); ...; (Some Ln, En)

-Una mezcla, por ejemplo, (~L1:E1, E2) cuando el es (Some L1, E1); (None, E2)

Invariante: n >= 2


*)
| Pexp_construct de Longident.t Asttypes.loc * expression option
(* Pexp_construct(C, exp) representa:

- C cuando exp es None ,

- C E cuando exp es Some E ,

- C (E1, ..., En) cuando exp es Some (Pexp_tuple[E1;...;En])


*)
| Pexp_variant de Asttypes.label * expression option
(* Pexp_variant(`A, exp) representa

- `A cuando exp es None

- `A E cuando exp es Some E


*)
| Pexp_record de (Longident.t Asttypes.loc * expression) list * expression option
(* Pexp_record([(l1,P1) ; ... ; (ln,Pn)], exp0) representa

- { l1=P1; ...; ln=Pn } cuando exp0 es None

- { E0 with l1=P1; ...; ln=Pn } cuando exp0 es Some E0

Invariante: n > 0


*)
| Pexp_field de expression * Longident.t Asttypes.loc
(* E.l


*)
| Pexp_setfield de expression * Longident.t Asttypes.loc * expression
(* E1.l <- E2


*)
| Pexp_array de expression list
(* [| E1; ...; En |]


*)
| Pexp_ifthenelse de expression * expression * expression option
(* if E1 then E2 else E3


*)
| Pexp_sequence de expression * expression
(* E1; E2


*)

Structure language

This section describes the Parsetree.structure, Parsetree.structure_item and Parsetree.module_type types, which represent the structure of a compiled OCaml file.

Structure of a file

type structure = {
pstr_items : structure_item list ;
pstr_frac_pos : float ;
}

type structure_item = {
pstr_desc : structure_item_desc ;
pstr_loc : Location.t ;
pstr_attributes : attributes ;
}

type structure_item_desc =
| Pstr_eval of expression
(* e. This is the code for the file's last expression. *)
| Pstr_value of Asttypes.rec_flag * value_binding list
(* let ...

let rec ...


*)
| Pstr_effect of expression
(* e. This is the code for an expression whose value is discarded. *)
| Pstr_module of module_binding
(* module M = E


*)
| Pstr_recmodule of module_binding list
(* module rec M1 = E1 and ... and Mn = En


*)
| Pstr_modtype of module_type_declaration
(* module type M = Mtype


*)
| Pstr_open of open_description
(* open M


*)
| Pstr_class of class_description list
(* class c = ...


*)
| Pstr_primitive of value_description list
(* external e: t = external_call


*)
| Pstr_type of type_declaration list
(* type t = ...


*)
| Pstr_exception of Pext_decl
(* exception E of ...


*)
| Pstr_module_type of module_type
(* module type f = ...


*)
| Pstr_open of open_description
(* open M


*)
| Pstr_class_type of class_type_declaration list
(* class type c = ...


*)
| Pstr_include of include_description
(* include M


*)
| Pstr_attribute of attribute
(* [@@@attribute]


*)
| Pstr_extension of extension
(* [%%extension]


*)
}

type module_binding = {
pmod_name : string Asttypes.loc ;
pmod_expr : module_expr ;
pmod_attributes : attributes ; (* ... [@@id1] [@@id2]


*)
}

type module_expr = {
pmod_desc : module_expr_desc ;
pmod_loc : Location.t ;
pmod_attributes : attributes ; (* ... [@id1] [@id2]


*)
}

type module_expr_desc =
|pmod_ident of Longident.t Asttypes.loc
(* M


*)
| Pmod_structure of structure
(* struct ... end
