subroutine dlaqr5 (logical wantt, logical wantz, integer kacc22, integer n, integer ktop, integer kbot, integer nshfts, double precision, dimension( * ) sr, double precision, dimension( * ) si, double precision, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( ldu, * ) u, integer ldu, integer nv, double precision, dimension( ldwv, * ) wv, integer ldwv, integer nh, double precision, dimension( ldwh, * ) wh, integer ldwh)

DLAQR5 realiza un único barrido QR multi-instrucción de bulto pequeño.

Propósito:

!>
!>    DLAQR5, llamado por DLAQR0, realiza un
!>    único barrido QR multi-instrucción de bulto pequeño.
!>

Parámetros

WANTT 
!>          WANTT es LOGICAL
!>             WANTT = .true. si se está calculando el factor de Schur
!>             cuasi-triangular.  WANTT se establece en .false. en caso contrario.
!>

WANTZ

!>          WANTZ es LOGICAL
!>             WANTZ = .true. si se está calculando el factor de Schur
!>             ortogonal.  WANTZ se establece en .false. en caso contrario.
!>

KACC22

!>          KACC22 es INTEGER con valor 0, 1, o 2.
!>             Especifica el modo de cálculo de las actualizaciones ortogonales
!>             lejanas de la diagonal.
!>        = 0: DLAQR5 no acumula reflexiones y no usa multiplicación matriz-matriz
!>             para actualizar las entradas de la matriz lejanas de la diagonal.
!>        = 1: DLAQR5 acumula reflexiones y usa multiplicación matriz-matriz
!>             para actualizar las entradas de la matriz lejanas de la diagonal.
!>        = 2: Igual que KACC22 = 1. Esta opción se usaba para explotar
!>             la estructura de 2x2 durante las multiplicaciones de matrices, pero
!>             esto ya no es compatible.
!>

N

!>          N es INTEGER
!>             N es el orden de la matriz de Hessenberg H sobre la que opera esta
!>             subrutina.
!>

KTOP

!>          KTOP es INTEGER
!>

KBOT

!>          KBOT es INTEGER
!>             Estas son las primeras y últimas filas y columnas de un
!>             bloque diagonal aislado sobre el que se aplicará el barrido QR.
!>             Se asume sin verificación que
!>                       o bien KTOP = 1  o   H(KTOP,KTOP-1) = 0
!>             y
!>                       o bien KBOT = N  o   H(KBOT+1,KBOT) = 0.
!>

NSHFTS

!>          NSHFTS es INTEGER
!>             NSHFTS indica el número de instrucciones simultáneas. NSHFTS
!>             debe ser positivo y par.
!>

SR

!>          SR es array DOUBLE PRECISION, dimension (NSHFTS)
!>

SI

!>          SI es array DOUBLE PRECISION, dimension (NSHFTS)
!>             SR contiene las partes reales y SI contiene las partes imaginarias
!>             de las NSHFTS instrucciones de origen que definen el barrido QR multi-instrucción.
!>             En la salida, SR y SI pueden ser reordenados.
!>

H

!>          H es array DOUBLE PRECISION, dimension (LDH,N)
!>             En la entrada, H contiene una matriz de Hessenberg. En la salida,
!>             se aplica un barrido QR multi-instrucción con instrucciones SR(J)+i*SI(J)
!>             al bloque diagonal aislado en las filas y columnas KTOP
!>             a KBOT.
!>

LDH

!>          LDH es INTEGER
!>             LDH es la dimensión principal de H tal como se declara en el
!>             procedimiento de llamada. LDH >= MAX(1,N).
!>

ILOZ

!>          ILOZ es INTEGER
!>

IHIZ

!>          IHIZ es INTEGER
!>             Especifican las filas de Z a las que se deben aplicar las transformaciones
!>             si WANTZ es .TRUE.. 1 <= ILOZ <= IHIZ <= N
!>

Z

!>          Z es array DOUBLE PRECISION, dimension (LDZ,IHIZ)
!>             Si WANTZ = .TRUE., entonces la transformación de similitud ortogonal
!>             del barrido QR se acumula en
!>             Z(ILOZ:IHIZ,ILOZ:IHIZ) desde la derecha.
!>             Si WANTZ = .FALSE., entonces Z no se referencia.
!>

LDZ

!>          LDZ es INTEGER
!>             LDA es la dimensión principal de Z tal como se declara en
!>             el procedimiento de llamada. LDZ >= N.
!>

V

!>          V es array DOUBLE PRECISION, dimension (LDV,NSHFTS/2)
!>

LDV

!>          LDV es INTEGER
!>             LDV es la dimensión principal de V tal como se declara en
!>             el procedimiento de llamada. LDV >= 3.
!>

U

!>          U es array DOUBLE PRECISION, dimension (LDU,2*NSHFTS)
!>

LDU

!>          LDU es INTEGER
!>             LDU es la dimensión principal de U tal como se declara en
!>             el procedimiento de llamada. LDU >= 2*NSHFTS.
!>

NV

!>          NV es INTEGER
!>             NV es el número de filas en WV disponibles como espacio de trabajo.
!>             NV >= 1.
!>

WV

!>          WV es array DOUBLE PRECISION, dimension (LDWV,2*NSHFTS)
!>

LDWV

!>          LDWV es INTEGER
!>             LDWV es la dimensión principal de WV tal como se declara en
!>             el procedimiento de llamada. LDWV >= NV.
!>

NH

!>          NH es INTEGER
!>             NH es el número de columnas en el array WH disponibles como espacio de trabajo.
!>             NH >= 1.
!>

WH

!>          WH es array DOUBLE PRECISION, dimension (LDWH,NH)
!>

LDWH

!>          LDWH es INTEGER
!>             Dimensión principal de WH tal como se declara en el
!>             procedimiento de llamada. LDWH >= 2*NSHFTS.
!>
