subroutine dlsets (integer m, integer p, integer n, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, integer lda, double precision, dimension( ldb, * ) b, double precision, dimension( ldb, * ) bf, integer ldb, double precision, dimension( * ) c, double precision, dimension( * ) cf, double precision, dimension( * ) d, double precision, dimension( * ) df, double precision, dimension( * ) x, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

DLSETS

Propósito:

!>
!> DLSETS prueba DGGLSE - una subrutina para resolver el problema del
!> cuadrado mínimo lineal con restricciones de igualdad (LSE).
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz B.  P >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B.  N >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, dimensión (LDA,N)
!>          La matriz M-por-N A.
!>

AF

!>          AF es un array DOUBLE PRECISION, dimensión (LDA,N)
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, Q y R.
!>          LDA >= max(M,N).
!>

B

!>          B es un array DOUBLE PRECISION, dimensión (LDB,N)
!>          La matriz P-por-N A.
!>

BF

!>          BF es un array DOUBLE PRECISION, dimensión (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de los arrays B, BF, V y S.
!>          LDB >= max(P,N).
!>

C

!>          C es un array DOUBLE PRECISION, dimensión( M )
!>          el vector C en el problema LSE.
!>

CF

!>          CF es un array DOUBLE PRECISION, dimensión( M )
!>

D

!>          D es un array DOUBLE PRECISION, dimensión( P )
!>          el vector D en el problema LSE.
!>

DF

!>          DF es un array DOUBLE PRECISION, dimensión( P )
!>

X

!>          X es un array DOUBLE PRECISION, dimensión( N )
!>          el vector solución X en el problema LSE.
!>

WORK

!>          WORK es un array DOUBLE PRECISION, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (M)
!>

RESULT

!>          RESULT es un array DOUBLE PRECISION, dimensión (2)
!>          Las razones de prueba:
!>            RESULT(1) = norm( A*x - c )/ norm(A)*norm(X)*EPS
!>            RESULT(2) = norm( B*x - d )/ norm(B)*norm(X)*EPS
!>
