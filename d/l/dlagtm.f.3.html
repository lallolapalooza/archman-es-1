subroutine dlagtm (character trans, integer n, integer nrhs, double precision alpha, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( ldx, * ) x, integer ldx, double precision beta, double precision, dimension( ldb, * ) b, integer ldb)

DLAGTM realiza un producto matriz-matriz de la forma C = αAB+βC, donde A es una matriz tridiagonal, B y C son matrices rectangulares, y α y β son escalares, que pueden ser 0, 1 o -1.

Propósito:

!>
!> DLAGTM realiza un producto matriz-matriz de la forma
!>
!>    B := alpha * A * X + beta * B
!>
!> donde A es una matriz tridiagonal de orden N, B y X son matrices N por NRHS,
!> y alpha y beta son escalares reales, cada uno de los cuales puede ser
!> 0., 1. o -1.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la operación aplicada a A.
!>          = 'N':  Sin transposición, B := alpha * A * X + beta * B
!>          = 'T':  Transpuesta,    B := alpha * A'* X + beta * B
!>          = 'C':  Transpuesta conjugada = Transpuesta
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices X y B.
!>

ALPHA

!>          ALPHA es DOUBLE PRECISION
!>          El escalar alpha.  ALPHA debe ser 0., 1. o -1.; de lo contrario,
!>          se asume que es 0.
!>

DL

!>          DL es array DOUBLE PRECISION, dimensión (N-1)
!>          Los elementos subdiagonales (n-1) de T.
!>

D

!>          D es array DOUBLE PRECISION, dimensión (N)
!>          Los elementos diagonales de T.
!>

DU

!>          DU es array DOUBLE PRECISION, dimensión (N-1)
!>          Los elementos superdiagonales (n-1) de T.
!>

X

!>          X es array DOUBLE PRECISION, dimensión (LDX,NRHS)
!>          La matriz X de N por NRHS.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(N,1).
!>

BETA

!>          BETA es DOUBLE PRECISION
!>          El escalar beta.  BETA debe ser 0., 1. o -1.; de lo contrario,
!>          se asume que es 1.
!>

B

!>          B es array DOUBLE PRECISION, dimensión (LDB,NRHS)
!>          En la entrada, la matriz B de N por NRHS.
!>          En la salida, B se sobrescribe con la expresión matricial
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(N,1).
!>
