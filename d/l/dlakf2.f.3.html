Documentación de función/subrutina

subroutine dlakf2 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( lda, * ) b, double precision, dimension( lda, * ) d, double precision, dimension( lda, * ) e, double precision, dimension( ldz, * ) z, integer ldz)

DLAKF2

Propósito:

!>
!> Formar la matriz 2*M*N por 2*M*N
!>
!>        Z = [ kron(In, A)  -kron(B', Im) ]
!>            [ kron(In, D)  -kron(E', Im) ],
!>
!> donde In es la matriz identidad de tamaño n y X' es la transpuesta
!> de X. kron(X, Y) es el producto de Kronecker entre las matrices X
!> y Y.
!>

Parámetros

M 
!>          M es INTEGER
!>          Tamaño de la matriz, debe ser >= 1.
!>

N

!>          N es INTEGER
!>          Tamaño de la matriz, debe ser >= 1.
!>

A

!>          A es DOUBLE PRECISION, dimension ( LDA, M )
!>          La matriz A en la matriz de salida Z.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de A, B, D y E. ( LDA >= M+N )
!>

B

!>          B es DOUBLE PRECISION, dimension ( LDA, N )
!>

D

!>          D es DOUBLE PRECISION, dimension ( LDA, M )
!>

E

!>          E es DOUBLE PRECISION, dimension ( LDA, N )
!>
!>          Las matrices utilizadas para formar la matriz de salida Z.
!>

Z

!>          Z es DOUBLE PRECISION, dimension ( LDZ, 2*M*N )
!>          La matriz resultante de Kronecker M*N*2 por M*N*2 (ver arriba.)
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal de Z. ( LDZ >= 2*M*N )
!>
