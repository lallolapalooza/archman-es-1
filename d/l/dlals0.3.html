subroutine dlals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, double precision, dimension( ldgnum, * ) givnum, integer ldgnum, double precision, dimension( ldgnum, * ) poles, double precision, dimension( * ) difl, double precision, dimension( ldgnum, * ) difr, double precision, dimension( * ) z, integer k, double precision c, double precision s, double precision, dimension( * ) work, integer info)

DLALS0 aplica factores de multiplicación inversa en la resolución del problema de mínimos cuadrados utilizando el enfoque de división y conquista SVD.

Propósito:

!>
!> DLALS0 aplica inversamente los factores multiplicadores de la matriz
!> de vectores singulares izquierdos o derechos de una matriz diagonal
!> añadida a la matriz del lado derecho B al resolver el problema de mínimos
!> cuadrados utilizando el enfoque SVD de división y conquista.
!>
!> Para la matriz de vectores singulares izquierdos, se involucran tres tipos
!> de matrices ortogonales:
!>
!> (1L) Rotaciones de Givens: el número de tales rotaciones es GIVPTR; los
!>      pares de columnas/filas a los que se aplicaron están almacenados en GIVCOL;
!>      y los valores C y S de estas rotaciones están almacenados en GIVNUM.
!>
!> (2L) Permutación. La (NL+1)-ésima fila de B se moverá a la primera
!>      fila, y para J=2:N, la J-ésima fila de B se moverá a la
!>      J-ésima fila.
!>
!> (3L) La matriz de vectores singulares izquierdos de la matriz restante.
!>
!> Para la matriz de vectores singulares derechos, se involucran cuatro tipos
!> de matrices ortogonales:
!>
!> (1R) La matriz de vectores singulares derechos de la matriz restante.
!>
!> (2R) Si SQRE = 1, una rotación de Givens adicional para generar el espacio
!>      nulo derecho.
!>
!> (3R) La transformación inversa de (2L).
!>
!> (4R) La transformación inversa de (1L).
!>

Parámetros

ICOMPQ 
!>          ICOMPQ es INTEGER
!>         Especifica si se van a calcular los vectores singulares en
!>         forma factorizada:
!>         = 0: Matriz de vectores singulares izquierdos.
!>         = 1: Matriz de vectores singulares derechos.
!>

NL

!>          NL es INTEGER
!>         La dimensión de fila del bloque superior. NL >= 1.
!>

NR

!>          NR es INTEGER
!>         La dimensión de fila del bloque inferior. NR >= 1.
!>

SQRE

!>          SQRE es INTEGER
!>         = 0: el bloque inferior es una matriz cuadrada de NRxNR.
!>         = 1: el bloque inferior es una matriz rectangular de NRx(NR+1).
!>
!>         La matriz bidiagonal tiene dimensión de fila N = NL + NR + 1,
!>         y dimensión de columna M = N + SQRE.
!>

NRHS

!>          NRHS es INTEGER
!>         El número de columnas de B y BX. NRHS debe ser al menos 1.
!>

B

!>          B es un array DOUBLE PRECISION, de dimensión ( LDB, NRHS )
!>         En la entrada, B contiene los lados derechos del problema de mínimos
!>         cuadrados en las filas 1 hasta M. En la salida, B contiene
!>         la solución X en las filas 1 hasta N.
!>

LDB

!>          LDB es INTEGER
!>         La dimensión principal de B. LDB debe ser al menos
!>         max(1,MAX( M, N ) ).
!>

BX

!>          BX es un array DOUBLE PRECISION, de dimensión ( LDBX, NRHS )
!>

LDBX

!>          LDBX es INTEGER
!>         La dimensión principal de BX.
!>

PERM

!>          PERM es un array INTEGER, de dimensión ( N )
!>         Las permutaciones (de deflación y ordenación) aplicadas
!>         a los dos bloques.
!>

GIVPTR

!>          GIVPTR es INTEGER
!>         El número de rotaciones de Givens que ocurrieron en este
!>         subproblema.
!>

GIVCOL

!>          GIVCOL es un array INTEGER, de dimensión ( LDGCOL, 2 )
!>         Cada par de números indica un par de filas/columnas
!>         involucradas en una rotación de Givens.
!>

LDGCOL

!>          LDGCOL es INTEGER
!>         La dimensión principal de GIVCOL, debe ser al menos N.
!>

GIVNUM

!>          GIVNUM es un array DOUBLE PRECISION, de dimensión ( LDGNUM, 2 )
!>         Cada número indica el valor C o S utilizado en la
!>         rotación de Givens correspondiente.
!>

LDGNUM

!>          LDGNUM es INTEGER
!>         La dimensión principal de los arrays DIFR, POLES y
!>         GIVNUM, debe ser al menos K.
!>

POLES

!>          POLES es un array DOUBLE PRECISION, de dimensión ( LDGNUM, 2 )
!>         En la entrada, POLES(1:K, 1) contiene los nuevos valores
!>         singulares obtenidos al resolver la ecuación secular, y
!>         POLES(1:K, 2) es un array que contiene los polos en la
!>         ecuación secular.
!>

DIFL

!>          DIFL es un array DOUBLE PRECISION, de dimensión ( K ).
!>         En la entrada, DIFL(I) es la distancia entre el I-ésimo valor
!>         singular actualizado (sin deflación) y el I-ésimo valor
!>         singular antiguo (sin deflación).
!>

DIFR

!>          DIFR es un array DOUBLE PRECISION, de dimensión ( LDGNUM, 2 ).
!>         En la entrada, DIFR(I, 1) contiene las distancias entre el I-ésimo
!>         valor singular actualizado (sin deflación) y el (I+1)-ésimo
!>         valor singular antiguo (sin deflación). Y DIFR(I, 2) es el
!>         factor de normalización para el I-ésimo vector singular derecho.
!>

Z

!>          Z es un array DOUBLE PRECISION, de dimensión ( K )
!>         Contiene los componentes del vector de fila de actualización
!>         ajustado por deflación.
!>

K

!>          K es INTEGER
!>         Contiene la dimensión de la matriz no desinflada,
!>         este es el orden de la ecuación secular relacionada. 1 <= K <= N.
!>

C

!>          C es DOUBLE PRECISION
!>         C contiene basura si SQRE = 0 y el valor C de una rotación
!>         de Givens relacionada con el espacio nulo derecho si SQRE = 1.
!>

S

!>          S es DOUBLE PRECISION
!>         S contiene basura si SQRE = 0 y el valor S de una rotación
!>         de Givens relacionada con el espacio nulo derecho si SQRE = 1.
!>

WORK

!>          WORK es un array DOUBLE PRECISION, de dimensión ( K )
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa.
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
