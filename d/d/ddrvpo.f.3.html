subroutine ddrvpo (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, double precision thresh, logical tsterr, integer nmax, double precision, dimension( * ) a, double precision, dimension( * ) afac, double precision, dimension( * ) asav, double precision, dimension( * ) b, double precision, dimension( * ) bsav, double precision, dimension( * ) x, double precision, dimension( * ) xact, double precision, dimension( * ) s, double precision, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

DDRVPO

Propósito:

!>
!> DDRVPO prueba las rutinas de controlador DPOSV y -SVX.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de vectores del lado derecho que se generarán para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si las salidas de error deben ser probadas.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, utilizado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array DOUBLE PRECISION, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array DOUBLE PRECISION, dimension (NMAX*NMAX)
!>

ASAV

!>          ASAV es un array DOUBLE PRECISION, dimension (NMAX*NMAX)
!>

B

!>          B es un array DOUBLE PRECISION, dimension (NMAX*NRHS)
!>

BSAV

!>          BSAV es un array DOUBLE PRECISION, dimension (NMAX*NRHS)
!>

X

!>          X es un array DOUBLE PRECISION, dimension (NMAX*NRHS)
!>

XACT

!>          XACT es un array DOUBLE PRECISION, dimension (NMAX*NRHS)
!>

S

!>          S es un array DOUBLE PRECISION, dimension (NMAX)
!>

WORK

!>          WORK es un array DOUBLE PRECISION, dimension
!>                      (NMAX*max(3,NRHS))
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (NMAX+2*NRHS)
!>

IWORK

!>          IWORK es un array INTEGER, dimension (NMAX)
!>

NOUT

!>          NOUT es INTEGER
!>          La unidad de número para la salida.
!>
