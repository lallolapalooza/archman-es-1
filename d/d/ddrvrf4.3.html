subroutine ddrvrf4 (integer nout, integer nn, integer, dimension( nn ) nval, double precision thresh, double precision, dimension( ldc, * ) c1, double precision, dimension( ldc, *) c2, integer ldc, double precision, dimension( * ) crf, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d_work_dlange)

DDRVRF4

Propósito:

!>
!> DDRVRF4 prueba las rutinas RFP de LAPACK:
!>     DSFRK
!>

Parámetros

NOUT 
!>          NOUT es INTEGER
!>                El número de unidad para la salida.
!>

NN

!>          NN es INTEGER
!>                El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>                Los valores de la dimensión N de la matriz.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>                El valor umbral para las relaciones de prueba. Un resultado se
!>                incluye en el archivo de salida si RESULT >= THRESH. Para
!>                que se imprima cada relación de prueba, use THRESH = 0.
!>

C1

!>          C1 es un array DOUBLE PRECISION,
!>                dimension (LDC,NMAX)
!>

C2

!>          C2 es un array DOUBLE PRECISION,
!>                dimension (LDC,NMAX)
!>

LDC

!>          LDC es INTEGER
!>                La dimensión principal del array A.
!>                LDA >= max(1,NMAX).
!>

CRF

!>          CRF es un array DOUBLE PRECISION,
!>                dimension ((NMAX*(NMAX+1))/2).
!>

A

!>          A es un array DOUBLE PRECISION,
!>                dimension (LDA,NMAX)
!>

LDA

!>          LDA es INTEGER
!>                La dimensión principal del array A. LDA >= max(1,NMAX).
!>

D_WORK_DLANGE

!>          D_WORK_DLANGE es un array DOUBLE PRECISION, dimension (NMAX)
!>
