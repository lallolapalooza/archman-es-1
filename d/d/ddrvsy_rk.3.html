subrutina ddrvsy_rk (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, double precision thresh, logical tsterr, integer nmax, double precision, dimension( * ) a, double precision, dimension( * ) afac, double precision, dimension( * ) e, double precision, dimension( * ) ainv, double precision, dimension( * ) b, double precision, dimension( * ) x, double precision, dimension( * ) xact, double precision, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

DDRVSY_RK

Propósito:

!> DDRVSY_RK prueba las rutinas de control DSYSV_RK.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matrices a ser utilizados para la prueba. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se usan para la prueba si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se usa.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>          Los valores de la dimensión N de la matriz.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de vectores del lado derecho a ser generados para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las razones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada razón de prueba se imprima, usar THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si las salidas de error deben ser probadas.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, usado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array DOUBLE PRECISION, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array DOUBLE PRECISION, dimension (NMAX*NMAX)
!>

E

!>          E es un array DOUBLE PRECISION, dimension (NMAX)
!>

AINV

!>          AINV es un array DOUBLE PRECISION, dimension (NMAX*NMAX)
!>

B

!>          B es un array DOUBLE PRECISION, dimension (NMAX*NRHS)
!>

X

!>          X es un array DOUBLE PRECISION, dimension (NMAX*NRHS)
!>

XACT

!>          XACT es un array DOUBLE PRECISION, dimension (NMAX*NRHS)
!>

WORK

!>          WORK es un array DOUBLE PRECISION, dimension (NMAX*max(2,NRHS))
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (NMAX+2*NRHS)
!>

IWORK

!>          IWORK es un array INTEGER, dimension (2*NMAX)
!>

NOUT

!>          NOUT es INTEGER
!>          La unidad de número para la salida.
!>
