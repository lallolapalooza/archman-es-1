subrutina ddrvgb (lógico, dimensión( * ) dotype, entero nn, entero, dimensión( * ) nval, entero nrhs, doble precisión thresh, lógico tsterr, doble precisión, dimensión( * ) a, entero la, doble precisión, dimensión( * ) afb, entero lafb, doble precisión, dimensión( * ) asav, doble precisión, dimensión( * ) b, doble precisión, dimensión( * ) bsav, doble precisión, dimensión( * ) x, doble precisión, dimensión( * ) xact, doble precisión, dimensión( * ) s, doble precisión, dimensión( * ) work, doble precisión, dimensión( * ) rwork, entero, dimensión( * ) iwork, entero nout)

DDRVGB

Propósito:

!>
!> DDRVGB prueba las rutinas de control DGBSV y -SVX.
!>

Parámetros

DOTYPE 
!>          DOTYPE es una matriz LÓGICA, de dimensión (NTYPES)
!>          Los tipos de matrices a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es ENTERO
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es una matriz ENTERA, de dimensión (NN)
!>          Los valores de la dimensión de columna N de la matriz.
!>

NRHS

!>          NRHS es ENTERO
!>          El número de vectores del lado derecho que se generarán para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es DOBLE PRECISIÓN
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para tener
!>          cada relación de prueba impresa, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LÓGICO
!>          Indicador que señala si se deben probar las salidas de error.
!>

A

!>          A es una matriz de DOBLE PRECISIÓN, de dimensión (LA)
!>

LA

!>          LA es ENTERO
!>          La longitud de la matriz A. LA >= (2*NMAX-1)*NMAX
!>          donde NMAX es la entrada más grande en NVAL.
!>

AFB

!>          AFB es una matriz de DOBLE PRECISIÓN, de dimensión (LAFB)
!>

LAFB

!>          LAFB es ENTERO
!>          La longitud de la matriz AFB. LAFB >= (3*NMAX-2)*NMAX
!>          donde NMAX es la entrada más grande en NVAL.
!>

ASAV

!>          ASAV es una matriz de DOBLE PRECISIÓN, de dimensión (LA)
!>

B

!>          B es una matriz de DOBLE PRECISIÓN, de dimensión (NMAX*NRHS)
!>

BSAV

!>          BSAV es una matriz de DOBLE PRECISIÓN, de dimensión (NMAX*NRHS)
!>

X

!>          X es una matriz de DOBLE PRECISIÓN, de dimensión (NMAX*NRHS)
!>

XACT

!>          XACT es una matriz de DOBLE PRECISIÓN, de dimensión (NMAX*NRHS)
!>

S

!>          S es una matriz de DOBLE PRECISIÓN, de dimensión (2*NMAX)
!>

WORK

!>          WORK es una matriz de DOBLE PRECISIÓN, de dimensión
!>                      (NMAX*max(3,NRHS,NMAX))
!>

RWORK

!>          RWORK es una matriz de DOBLE PRECISIÓN, de dimensión
!>                      (NMAX+2*NRHS)
!>

IWORK

!>          IWORK es una matriz ENTERA, de dimensión (2*NMAX)
!>

NOUT

!>          NOUT es ENTERO
!>          El número de unidad para la salida.
!>
