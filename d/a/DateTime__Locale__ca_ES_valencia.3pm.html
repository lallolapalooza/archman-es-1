Patrones de Strftime

%c (%A, %-d %B de %Y, %T) - formato de fecha y hora

 2008-02-05T18:30:30 = dimarts, %-d de febrer de 2008, 18:30:30
 1995-12-22T09:05:02 = divendres, %-d de desembre de 1995, 09:05:02
-0010-09-15T04:44:23 = dissabte, %-d de setembre de -10, 04:44:23

%x (%-d/%-m/%y) - formato de fecha

 2008-02-05T18:30:30 = %-d/%-m/08
 1995-12-22T09:05:02 = %-d/%-m/95
-0010-09-15T04:44:23 = %-d/%-m/10

%X (%T) - formato de hora

 2008-02-05T18:30:30 = 18:30:30
 1995-12-22T09:05:02 = 09:05:02
-0010-09-15T04:44:23 = 04:44:23
