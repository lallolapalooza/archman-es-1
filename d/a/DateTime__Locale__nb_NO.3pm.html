Patrones de Strftime

%c (%a %d. %b %Y a las %H.%M %z) - formato de fecha y hora

 2008-02-05T18:30:30 = mar 05 feb 2008 a las 18.30 +0000
 1995-12-22T09:05:02 = vie 22 dic 1995 a las 09.05 +0000
-0010-09-15T04:44:23 = sáb 15 sep -10 a las 04.44 +0000

%x (%d. %b %Y) - formato de fecha

 2008-02-05T18:30:30 = 05 feb 2008
 1995-12-22T09:05:02 = 22 dic 1995
-0010-09-15T04:44:23 = 15 sep -10

 2008-02-05T18:30:30 = a las 18.30 +0000
