Patrones de Strftime

%c (%a %b %e %H:%M:%S %Y) - formato de fecha y hora

 2008-02-05T18:30:30 = otrd. febr.  5 18:30:30 2008
 1995-12-22T09:05:02 = piektd. dec. 22 09:05:02 1995
-0010-09-15T04:44:23 = sestd. sept. 15 04:44:23 -10

%x (%m/%d/%y) - formato de fecha

 2008-02-05T18:30:30 = 02/05/08
 1995-12-22T09:05:02 = 12/22/95
-0010-09-15T04:44:23 = 09/15/10

%X (%H:%M:%S) - formato de hora

 2008-02-05T18:30:30 = 18:30:30
 1995-12-22T09:05:02 = 09:05:02
-0010-09-15T04:44:23 = 04:44:23
