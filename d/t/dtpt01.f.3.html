subroutine dtpt01 (character uplo, character diag, integer n, double precision, dimension( * ) ap, double precision, dimension( * ) ainvp, double precision rcond, double precision, dimension( * ) work, double precision resid)

DTPT01

Propósito:

!>
!> DTPT01 calcula el residuo para una matriz triangular A multiplicada por su
!> inversa cuando A está almacenada en formato empaquetado:
!>    RESID = norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es superior o inferior triangular.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitriangular o no.
!>          = 'N':  No unitriangular
!>          = 'U':  Unitriangular
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

AP

!>          AP es un array DOUBLE PRECISION, de dimensión (N*(N+1)/2)
!>          La matriz triangular original A, empaquetada
!>          columna a columna en un array lineal. La j-ésima columna de A se almacena
!>          en el array AP como sigue:
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) para j<=i<=n.
!>

AINVP

!>          AINVP es un array DOUBLE PRECISION, de dimensión (N*(N+1)/2)
!>          En la entrada, la inversa (triangular) de la matriz A, empaquetada
!>          columna a columna en un array lineal como en AP.
!>          En la salida, el contenido de AINVP se destruye.
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          El recíproco del número de condición de A, calculado como
!>          1/(norm(A) * norm(AINV)).
!>

WORK

!>          WORK es un array DOUBLE PRECISION, de dimensión (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS )
!>
