subroutine dorgtr (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DORGTR

Propósito:

!>
!> DORGTR genera una matriz ortogonal real Q que se define como el
!> producto de n-1 reflectores elementales de orden N, tal como los devuelve
!> DSYTRD:
!>
!> si UPLO = 'U', Q = H(n-1) . . . H(2) H(1),
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(n-1).
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U': El triángulo superior de A contiene los reflectores elementales
!>                 de DSYTRD;
!>          = 'L': El triángulo inferior de A contiene los reflectores elementales
!>                 de DSYTRD.
!>

N

!>          N es INTEGER
!>          El orden de la matriz Q. N >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, dimensión (LDA,N)
!>          En la entrada, los vectores que definen los reflectores elementales,
!>          tal como los devuelve DSYTRD.
!>          En la salida, la matriz ortogonal N-por-N, Q.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,N).
!>

TAU

!>          TAU es un array DOUBLE PRECISION, dimensión (N-1)
!>          TAU(i) debe contener el factor escalar del reflector
!>          elemental H(i), tal como lo devuelve DSYTRD.
!>

WORK

!>          WORK es un array DOUBLE PRECISION, dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK. LWORK >= max(1,N-1).
!>          Para un rendimiento óptimo LWORK >= (N-1)*NB, donde NB es
!>          el tamaño de bloque óptimo.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK y XERBLA no emite
!>          ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el argumento i tuvo un valor ilegal
!>
