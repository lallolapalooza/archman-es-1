subroutine dpst01 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, double precision, dimension( ldperm, * ) perm, integer ldperm, integer, dimension( * ) piv, double precision, dimension( * ) rwork, double precision resid, integer rank)

DPST01

Propósito:

!>
!> DPST01 reconstruye una matriz simétrica semidefinida positiva A
!> a partir de sus factores L o U y la matriz de permutación P y calcula
!> el residuo
!>    norm( P*L*L'*P' - A ) / ( N * norm(A) * EPS ) o
!>    norm( P*U'*U*P' - A ) / ( N * norm(A) * EPS ),
!> donde EPS es el epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO is CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N is INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          La matriz simétrica original A.
!>

LDA

!>          LDA is INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC is DOUBLE PRECISION array, dimension (LDAFAC,N)
!>          El factor L o U de la factorización L*L' o U'*U
!>          de A.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          La dimensión principal del array AFAC.  LDAFAC >= max(1,N).
!>

PERM

!>          PERM is DOUBLE PRECISION array, dimension (LDPERM,N)
!>          Sobrescrito con la matriz reconstruida, y luego con la
!>          diferencia P*L*L'*P' - A (o P*U'*U*P' - A)
!>

LDPERM

!>          LDPERM is INTEGER
!>          La dimensión principal del array PERM.
!>          LDAPERM >= max(1,N).
!>

PIV

!>          PIV is INTEGER array, dimension (N)
!>          PIV es tal que las entradas no nulas son
!>          P( PIV( K ), K ) = 1.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U'*U - A) / ( N * norm(A) * EPS )
!>

RANK

!>          RANK is INTEGER
!>          número de valores singulares no nulos de A.
!>
