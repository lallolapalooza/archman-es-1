subroutine dporfsx (character uplo, character equed, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldaf, * ) af, integer ldaf, double precision, dimension( * ) s, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) berr, integer n_err_bnds, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, integer nparams, double precision, dimension( * ) params, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPORFSX

Propósito:

!>
!>    DPORFSX mejora la solución calculada de un sistema de ecuaciones
!>    lineales cuando la matriz de coeficientes es simétrica definida positiva,
!>    y proporciona cotas de error y estimaciones de error hacia atrás
!>    para la solución. Además de la cota de error normwise, el código
!>    proporciona una cota de error componente a componente máxima si es posible. Ver
!>    los comentarios para ERR_BNDS_NORM y ERR_BNDS_COMP para detalles de las
!>    cotas de error.
!>
!>    El sistema original de ecuaciones lineales puede haber sido 
!>    equilibrado antes de llamar a esta rutina, como se describe en los
!>    argumentos EQUED y S abajo. En este caso, la solución y las cotas de
!>    error devueltas son para el sistema original no equilibrado.
!> 
!>     Algunos parámetros opcionales están agrupados en la matriz PARAMS. Estos
!>     ajustes determinan cómo se realiza el refinamiento, pero a menudo
!>     los valores predeterminados son aceptables. Si los valores predeterminados son aceptables, los usuarios
!>     pueden pasar NPARAMS = 0 que evita que el código fuente acceda
!>     al argumento PARAMS.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>       = 'U':  La parte triangular superior de A está almacenada;
!>       = 'L':  La parte triangular inferior de A está almacenada.
!>

EQUED

!>          EQUED es CHARACTER*1
!>     Especifica la forma de equilibrado que se realizó en A
!>     antes de llamar a esta rutina. Esto es necesario para calcular
!>     la solución y las cotas de error correctamente.
!>       = 'N':  Sin equilibrado
!>       = 'Y':  Equilibrado de filas y columnas, es decir, A ha sido
!>               reemplazado por diag(S) * A * diag(S).
!>               El lado derecho B ha sido modificado en consecuencia.
!>

N

!>          N es INTEGER
!>     El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>     El número de lados derechos, es decir, el número de columnas
!>     de las matrices B y X.  NRHS >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, dimensión (LDA,N)
!>     La matriz simétrica A.  Si UPLO = 'U', la parte triangular
!>     superior N por N de A contiene la parte triangular superior
!>     de la matriz A, y la parte triangular estrictamente inferior de A
!>     no se referencia. Si UPLO = 'L', la parte triangular
!>     inferior N por N de A contiene la parte triangular inferior
!>     de la matriz A, y la parte triangular estrictamente superior de A
!>     no se referencia.
!>

LDA

!>          LDA es INTEGER
!>     La dimensión principal del array A.  LDA >= max(1,N).
!>

AF

!>          AF es un array DOUBLE PRECISION, dimensión (LDAF,N)
!>     El factor triangular U o L de la factorización de Cholesky
!>     A = U**T*U o A = L*L**T, tal como se calculó con DPOTRF.
!>

LDAF

!>          LDAF es INTEGER
!>     La dimensión principal del array AF.  LDAF >= max(1,N).
!>

S

!>          S es un array DOUBLE PRECISION, dimensión (N)
!>     Los factores de escala para A. Si EQUED = 'Y', A se multiplica a
!>     la izquierda y a la derecha por diag(S).  S es un argumento de entrada si FACT =
!>     'F'; de lo contrario, S es un argumento de salida. Si FACT = 'F' y EQUED
!>     = 'Y', cada elemento de S debe ser positivo. Si S es una salida, cada
!>     elemento de S es una potencia de la base. Si S es una entrada, cada
!>     elemento de S debe ser una potencia de la base para garantizar una solución fiable
!>     y estimaciones de error. El escalado por potencias de la base no causa
!>     errores de redondeo a menos que el resultado sea subdesbordado o
!>     desbordado.
!>     Los errores de redondeo durante el escalado conducen a un refinamiento con una matriz que
!>     no es equivalente a la matriz de entrada, produciendo estimaciones de error
!>     que pueden no ser fiables.
!>

B

!>          B es un array DOUBLE PRECISION, dimensión (LDB,NRHS)
!>     La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>     La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es un array DOUBLE PRECISION, dimensión (LDX,NRHS)
!>     En la entrada, la matriz solución X, tal como se calculó con DGETRS.
!>     En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>     La dimensión principal del array X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>     Condición inversa escalada. Esta es una estimación del
!>     número de condición de Skeel inverso de la matriz A después del
!>     equilibrado (si se realizó). Si este es menor que la precisión de la máquina
!>     (en particular, si es cero), la matriz es singular
!>     para la precisión de trabajo. Tenga en cuenta que el error puede ser pequeño incluso
!>     si este número es muy pequeño y la matriz parece mal condicionada.
!>

BERR

!>          BERR es un array DOUBLE PRECISION, dimensión (NRHS)
!>     Error relativo componente a componente hacia atrás. Este es el
!>     error relativo componente a componente de cada vector solución X(j)
!>     (es decir, el menor cambio relativo en cualquier elemento de A o B que
!>     haga que X(j) sea una solución exacta).
!>

N_ERR_BNDS

!>          N_ERR_BNDS es INTEGER
!>     Número de cotas de error a devolver para cada lado derecho
!>     y cada tipo (normwise o componente a componente). Ver ERR_BNDS_NORM y
!>     ERR_BNDS_COMP abajo.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM es un array DOUBLE PRECISION, dimensión (NRHS, N_ERR_BNDS)
!>     Para cada lado derecho, este array contiene información sobre
!>     varias cotas de error y números de condición correspondientes al
!>     error relativo normwise, que se define como sigue:
!>
!>     Error relativo normwise en el i-ésimo vector solución:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     El array está indexado por el tipo de información de error como se describe
!>     abajo. Actualmente hay hasta tres piezas de información
!>     devueltas.
!>
!>     El primer índice en ERR_BNDS_NORM(i,:) corresponde al i-ésimo
!>     lado derecho.
!>
!>     El segundo índice en ERR_BNDS_NORM(:,err) con
contiene los siguientes !> tres campos: !> err = 1 booleano. Confiar en la respuesta si el !> número de condición recíproco es menor que el umbral !> sqrt(n) * dlamch('Epsilon'). !> !> err = 2 límite de error: El error directo estimado, !> casi con certeza dentro de un factor de 10 del error !> real siempre que la siguiente entrada sea mayor que el umbral !> sqrt(n) * dlamch('Epsilon'). Este límite de error solo !> debe confiarse si el booleano anterior es verdadero. !> !> err = 3 Número de condición recíproco: Número de condición !> recíproco estimado en norma. Se compara con el umbral !> sqrt(n) * dlamch('Epsilon') para determinar si la !> estimación de error es . Estos números de condición !> recíprocos son 1 / (norm(Z^{-1},inf) * norm(Z,inf)) para alguna !> matriz Z escalada adecuadamente. !> Sea Z = S*A, donde S escala cada fila por una potencia de la !> base de tal modo que todas las sumas absolutas de filas de Z sean aproximadamente 1. !> !> Consulte la Nota de Trabajo de Lapack 165 para más detalles y precauciones !> adicionales. !>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP es un arreglo de PRECISIÓN DOBLE, dimensión (NRHS, N_ERR_BNDS)
!>     Para cada segundo miembro, este arreglo contiene información sobre
!>     varios límites de error y números de condición correspondientes al
!>     error relativo componente a componente, que se define como sigue:
!>
!>     Error relativo componente a componente en el i-ésimo vector solución:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     El arreglo está indexado por el segundo miembro i (del cual
!>     depende el error relativo componente a componente), y el tipo de
!>     información de error descrito a continuación. Actualmente se devuelven hasta tres
!>     piezas de información para cada segundo miembro. Si no se solicita
!>     precisión componente a componente (PARAMS(3) = 0.0), entonces
!>     no se accede a ERR_BNDS_COMP. Si N_ERR_BNDS < 3, entonces se devuelven como máximo
!>     las primeras (:,N_ERR_BNDS) entradas.
!>
!>     El primer índice en ERR_BNDS_COMP(i,:) corresponde al i-ésimo
!>     segundo miembro.
!>
!>     El segundo índice en ERR_BNDS_COMP(:,err) contiene los siguientes
!>     tres campos:
!>     err = 1  booleano. Confiar en la respuesta si el
!>              número de condición recíproco es menor que el umbral
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  límite de error: El error directo estimado,
!>              casi con certeza dentro de un factor de 10 del error
!>              real siempre que la siguiente entrada sea mayor que el umbral
!>              sqrt(n) * dlamch('Epsilon'). Este límite de error solo
!>              debe confiarse si el booleano anterior es verdadero.
!>
!>     err = 3  Número de condición recíproco: Número de condición
!>              recíproco estimado componente a componente. Se compara con el umbral
!>              sqrt(n) * dlamch('Epsilon') para determinar si la
!>              estimación de error es . Estos números de condición
!>              recíprocos son 1 / (norm(Z^{-1},inf) * norm(Z,inf)) para alguna
!>              matriz Z escalada adecuadamente.
!>              Sea Z = S*(A*diag(x)), donde x es la solución para el
!>              segundo miembro actual y S escala cada fila de
!>              A*diag(x) por una potencia de la base de tal modo que todas las sumas absolutas
!>              de filas de Z sean aproximadamente 1.
!>
!>     Consulte la Nota de Trabajo de Lapack 165 para más detalles y precauciones
!>     adicionales.
!>

NPARAMS

!>          NPARAMS es INTEGER
!>     Especifica el número de parámetros establecidos en PARAMS. Si <= 0, el
!>     arreglo PARAMS nunca se referencia y se usan valores predeterminados.
!>

PARAMS

!>          PARAMS es un arreglo de PRECISIÓN DOBLE, dimensión (NPARAMS)
!>     Especifica parámetros del algoritmo. Si una entrada es < 0.0, entonces
!>     esa entrada se rellenará con el valor predeterminado usado para ese
!>     parámetro. Solo se accede a las posiciones hasta NPARAMS; se usan
!>     los valores predeterminados para los parámetros de número más alto.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : Si realizar refinamiento
!>            iterativo o no.
!>         Predeterminado: 1.0D+0
!>            = 0.0:  No se realiza refinamiento y no se calculan
!>                    límites de error.
!>            = 1.0:  Usar el algoritmo de refinamiento de doble precisión,
!>                    posiblemente con cálculos de doble simple si el
!>                    entorno de compilación no soporta PRECISIÓN DOBLE.
!>              (otros valores están reservados para uso futuro)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : Número máximo de cálculos
!>            de residuo permitidos para el refinamiento.
!>         Predeterminado: 10
!>         Agresivo: Establecer a 100 para permitir la convergencia usando
!>                     factorizaciones aproximadas o factorizaciones distintas de LU. Si
!>                     la factorización usa una técnica distinta de la
!>                     eliminación gaussiana, las garantías en
!>                     err_bnds_norm y err_bnds_comp pueden dejar de ser
!>                     confiables.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : Indicador que determina si el código
!>            intentará encontrar una solución con pequeño error relativo
!>            componente a componente en el algoritmo de doble precisión. Positivo
!>            es verdadero, 0.0 es falso.
!>         Predeterminado: 1.0 (intentar convergencia componente a componente)
!>

WORK

!>          WORK es un arreglo de PRECISIÓN DOBLE, dimensión (4*N)
!>

IWORK

!>          IWORK es un arreglo de INTEGER, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>       = 0:  Salida exitosa. La solución para cada segundo miembro está
!>         garantizada.
!>       < 0:  Si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>       > 0 y <= N:  U(INFO,INFO) es exactamente cero. La factorización
!>         se ha completado, pero el factor U es exactamente singular, por lo
!>         que no se pudieron calcular la solución y los límites de error. Se devuelve
!>         RCOND = 0.
!>       = N+J: La solución correspondiente al J-ésimo segundo miembro no
!>         está garantizada. Las soluciones correspondientes a otros
!>         segundos miembros K con K > J pueden no estar garantizadas también, pero
!>         solo se informa del primer segundo miembro de ese tipo. Si no se solicita
!>         un pequeño error componente a componente (PARAMS(3) = 0.0) entonces
!>         el J-ésimo segundo miembro es el primero con un límite de error
!>         en norma que no está garantizado (el más pequeño J tal
!>         que ERR_BNDS_NORM(J,1) = 0.0). Por defecto (PARAMS(3) = 1.0)
!>         el J-ésimo segundo miembro es el primero con un límite de error
!>         en norma o componente a componente que no está garantizado (el más pequeño
!>         J tal que ERR_BNDS_NORM(J,1) = 0.0 o
!>         ERR_BNDS_COMP(J,1) = 0.0). Consulte la definición de
!>         ERR_BNDS_NORM(:,1) y ERR_BNDS_COMP(:,1). Para obtener información
!>         sobre todos los segundos miembros, verifique ERR_BNDS_NORM o
!>         ERR_BNDS_COMP.
!>
