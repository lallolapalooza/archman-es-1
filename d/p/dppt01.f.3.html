subroutine dppt01 (character uplo, integer n, double precision, dimension( * ) a, double precision, dimension( * ) afac, double precision, dimension( * ) rwork, double precision resid)

DPPT01

Propósito:

!>
!> DPPT01 reconstruye una matriz A simétrica definida positiva empaquetada
!> a partir de su factorización L*L' o U'*U y calcula la norma residual
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) o
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> donde EPS es el epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array de DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          La matriz simétrica original A, almacenada como una matriz
!>          triangular empaquetada.
!>

AFAC

!>          AFAC es un array de DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          En la entrada, el factor L o U de la factorización L*L' o U'*U
!>          de A, almacenado como una matriz triangular empaquetada.
!>          Sobreescrito con la matriz reconstruida, y luego con la
!>          diferencia L*L' - A (o U'*U - A).
!>

RWORK

!>          RWORK es un array de DOUBLE PRECISION, dimension (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U'*U - A) / ( N * norm(A) * EPS )
!>
