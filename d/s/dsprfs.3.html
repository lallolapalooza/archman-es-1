subroutine dsprfs (character uplo, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( * ) afp, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DSPRFS

Propósito:

!>
!> DSPRFS mejora la solución calculada de un sistema de ecuaciones
!> lineales cuando la matriz de coeficientes es simétrica indefinida
!> y empaquetada, y proporciona cotas de error y estimaciones de error
!> hacia atrás para la solución.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices B y X.  NRHS >= 0.
!>

AP

!>          AP es un array DOUBLE PRECISION, de dimensión (N*(N+1)/2)
!>          El triángulo superior o inferior de la matriz simétrica A,
!>          empaquetado por columnas en un array lineal. La j-ésima
!>          columna de A se almacena en el array AP de la siguiente
!>          manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) para j<=i<=n.
!>

AFP

!>          AFP es un array DOUBLE PRECISION, de dimensión (N*(N+1)/2)
!>          La forma factorizada de la matriz A. AFP contiene la matriz
!>          diagonal de bloques D y los multiplicadores utilizados para
!>          obtener U o L de la factorización A = U*D*U**T o
!>          A = L*D*L**T calculada por DSPTRF, almacenada como una
!>          matriz triangular empaquetada.
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          determinada por DSPTRF.
!>

B

!>          B es un array DOUBLE PRECISION, de dimensión (LDB,NRHS)
!>          La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,N).
!>

X

!>          X es un array DOUBLE PRECISION, de dimensión (LDX,NRHS)
!>          En la entrada, la matriz solución X, calculada por DSPTRS.
!>          En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X. LDX >= max(1,N).
!>

FERR

!>          FERR es un array DOUBLE PRECISION, de dimensión (NRHS)
!>          La estimación de la cota de error hacia adelante para cada
!>          vector solución X(j) (la j-ésima columna de la matriz
!>          solución X).
!>          Si XTRUE es la verdadera solución correspondiente a X(j),
!>          FERR(j) es una cota superior estimada para la magnitud del
!>          elemento más grande en (X(j) - XTRUE) dividido por la
!>          magnitud del elemento más grande en X(j). La estimación es
!>          tan fiable como la estimación de RCOND, y es casi siempre
!>          una ligera sobreestimación del error real.
!>

BERR

!>          BERR es un array DOUBLE PRECISION, de dimensión (NRHS)
!>          El error residual relativo componente a componente de cada
!>          vector solución X(j) (es decir, el cambio relativo más
!>          pequeño en cualquier elemento de A o B que hace que X(j)
!>          sea una solución exacta).
!>

WORK

!>          WORK es un array DOUBLE PRECISION, de dimensión (3*N)
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor
!>                 ilegal
!>

Parámetros Internos:

!>  ITMAX es el número máximo de pasos de refinamiento iterativo.
!>
