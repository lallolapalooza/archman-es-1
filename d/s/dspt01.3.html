subroutine dspt01 (character uplo, integer n, double precision, dimension( * ) a, double precision, dimension( * ) afac, integer, dimension( * ) ipiv, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

DSPT01

Propósito:

!>
!> DSPT01 reconstruye una matriz indefinida simétrica empaquetada A a partir de su
!> factorización de bloque L*D*L' o U*D*U' y calcula la residual
!>      norm( C - A ) / ( N * norm(A) * EPS ),
!> donde C es la matriz reconstruida y EPS es la epsilon de máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A se almacena:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          La matriz simétrica original A, almacenada como una matriz
!>          triangular empaquetada.
!>

AFAC

!>          AFAC es un array DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          La forma factorizada de la matriz A, almacenada como una matriz
!>          triangular empaquetada. AFAC contiene la matriz diagonal de bloque D
!>          y los multiplicadores utilizados para obtener el factor L o U de la
!>          factorización de bloque L*D*L' o U*D*U' según lo calculado por DSPTRF.
!>

IPIV

!>          IPIV es un array INTEGER, dimension (N)
!>          Los índices de pivote de DSPTRF.
!>

C

!>          C es un array DOUBLE PRECISION, dimension (LDC,N)
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
