subroutine dsytri (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer info)

DSYTRI

Propósito:

!>
!> DSYTRI calcula la inversa de una matriz real simétrica indefinida
!> A usando la factorización A = U*D*U**T o A = L*D*L**T calculada por
!> DSYTRF.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, la forma es A = U*D*U**T;
!>          = 'L':  Triangular inferior, la forma es A = L*D*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, dimensión (LDA,N)
!>          En la entrada, la matriz diagonal por bloques D y los multiplicadores
!>          utilizados para obtener el factor U o L como se calculó por DSYTRF.
!>
!>          En la salida, si INFO = 0, la inversa (simétrica) de la matriz
!>          original.  Si UPLO = 'U', la parte triangular superior de la
!>          inversa se forma y la parte de A debajo de la diagonal no
!>          se referencia; si UPLO = 'L', la parte triangular inferior de la
!>          inversa se forma y la parte de A encima de la diagonal no
!>          se referencia.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          tal como lo determinó DSYTRF.
!>

WORK

!>          WORK es un array DOUBLE PRECISION, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0: si INFO = i, D(i,i) = 0; la matriz es singular y su
!>               inversa no pudo calcularse.
!>
