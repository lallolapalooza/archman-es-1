subroutine dsyrfs (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DSYRFS

Propósito:

!>
!> DSYRFS mejora la solución calculada de un sistema de ecuaciones
!> lineales cuando la matriz de coeficientes es simétrica indefinida,
!> y proporciona límites de error y estimaciones de error hacia atrás
!> para la solución.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices B y X.  NRHS >= 0.
!>

A

!>          A es un array de DOUBLE PRECISION, dimensión (LDA,N)
!>          La matriz simétrica A.  Si UPLO = 'U', la parte superior
!>          N por N de A contiene la parte triangular superior de
!>          la matriz A, y la parte estrictamente triangular inferior de A
!>          no se referencia.  Si UPLO = 'L', la parte inferior
!>          N por N de A contiene la parte triangular inferior de
!>          la matriz A, y la parte estrictamente triangular superior de A
!>          no se referencia.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

AF

!>          AF es un array de DOUBLE PRECISION, dimensión (LDAF,N)
!>          La forma factorizada de la matriz A.  AF contiene la matriz
!>          diagonal por bloques D y los multiplicadores usados para obtener
!>          U o L de la factorización A = U*D*U**T o
!>          A = L*D*L**T como se calculó por DSYTRF.
!>

LDAF

!>          LDAF es INTEGER
!>          La dimensión principal del array AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          determinados por DSYTRF.
!>

B

!>          B es un array de DOUBLE PRECISION, dimensión (LDB,NRHS)
!>          La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es un array de DOUBLE PRECISION, dimensión (LDX,NRHS)
!>          En la entrada, la matriz solución X, calculada por DSYTRS.
!>          En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

FERR

!>          FERR es un array de DOUBLE PRECISION, dimensión (NRHS)
!>          El límite de error hacia adelante estimado para cada vector solución
!>          X(j) (la j-ésima columna de la matriz solución X).
!>          Si XTRUE es la solución verdadera correspondiente a X(j), FERR(j)
!>          es un límite superior estimado para la magnitud del elemento más
!>          grande en (X(j) - XTRUE) dividido por la magnitud del elemento más
!>          grande en X(j). La estimación es tan fiable como
!>          la estimación para RCOND, y es casi siempre una ligera
!>          sobreestimación del error real.
!>

BERR

!>          BERR es un array de DOUBLE PRECISION, dimensión (NRHS)
!>          El error relativo por componentes hacia atrás de cada vector solución
!>          X(j) (es decir, el cambio relativo más pequeño en
!>          cualquier elemento de A o B que haga que X(j) sea una solución exacta).
!>

WORK

!>          WORK es un array de DOUBLE PRECISION, dimensión (3*N)
!>

IWORK

!>          IWORK es un array INTEGER, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Parámetros Internos:

!>  ITMAX es el número máximo de pasos de refinamiento iterativo.
!>
