subroutine dsyt01_3 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, double precision, dimension( * ) e, integer, dimension( * ) ipiv, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

DSYT01_3

Propósito:

!>
!> DSYT01_3 reconstruye una matriz simétrica indefinida A a partir de su
!> factorización de bloque L*D*L' o U*D*U' calculada por DSYTRF_RK
!> (o DSYTRF_BK) y calcula el residuo
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> donde C es la matriz reconstruida y EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si se almacena la parte triangular superior o inferior de la
!>          matriz simétrica A:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, de dimensión (LDA,N)
!>          La matriz simétrica original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC es un array DOUBLE PRECISION, de dimensión (LDAFAC,N)
!>          Diagonal de la matriz diagonal por bloques D y factores U o L
!>          calculados por DSYTRF_RK y DSYTRF_BK:
!>            a) SOLO elementos diagonales de la matriz diagonal por bloques
!>               simétrica D en la diagonal de A, es decir, D(k,k) = A(k,k);
!>               (los elementos de la supradiagonal (o subdiagonal) de D
!>                deben proporcionarse en la entrada en el array E), y
!>            b) Si UPLO = 'U': factor U en la supradiagonal de A.
!>               Si UPLO = 'L': factor L en la subdiagonal de A.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E es un array DOUBLE PRECISION, de dimensión (N)
!>          En la entrada, contiene los elementos de la supradiagonal (o subdiagonal)
!>          de la matriz diagonal por bloques simétrica
!>          con bloques diagonales de 1x1 o 2x2, donde
!>          Si UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) no referenciado;
!>          Si UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) no referenciado.
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>          Los índices de pivote de DSYTRF_RK (o DSYTRF_BK).
!>

C

!>          C es un array DOUBLE PRECISION, de dimensión (LDC,N)
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, de dimensión (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
