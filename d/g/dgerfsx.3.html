subroutine dgerfsx (character trans, character equed, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx , * ) x, integer ldx, double precision rcond, double precision, dimension( * ) berr, integer n_err_bnds, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, integer nparams, double precision, dimension( * ) params, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGERFSX

Propósito:

!>
!>    DGERFSX mejora la solución calculada de un sistema de ecuaciones
!>    lineales y proporciona límites de error y estimaciones de error
!>    hacia atrás para la solución. Además del límite de error normwise,
!>    el código proporciona un límite de error máximo componente a
!>    componente si es posible. Ver los comentarios de
!>    ERR_BNDS_NORM y ERR_BNDS_COMP para detalles de los límites de error.
!>
!>    El sistema original de ecuaciones lineales puede haber sido
!>    equilibrado antes de llamar a esta rutina, como se describe por
!>    los argumentos EQUED, R y C a continuación. En este caso, la
!>    solución y los límites de error devueltos son para el sistema
!>    original no equilibrado.
!> 
!>     Algunos parámetros opcionales están agrupados en el array PARAMS.
!>     Estas configuraciones determinan cómo se realiza el refinamiento,
!>     pero a menudo los valores predeterminados son aceptables. Si los
!>     valores predeterminados son aceptables, los usuarios pueden pasar
!>     NPARAMS = 0 lo que evita que el código fuente acceda al argumento
!>     PARAMS.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>     Especifica la forma del sistema de ecuaciones:
!>       = 'N':  A * X = B     (Sin transposición)
!>       = 'T':  A**T * X = B  (Transpuesta)
!>       = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

EQUED

!>          EQUED es CHARACTER*1
!>     Especifica la forma de equilibrado que se realizó en A
!>     antes de llamar a esta rutina. Esto es necesario para calcular
!>     correctamente la solución y los límites de error.
!>       = 'N':  Sin equilibrado
!>       = 'R':  Equilibrado de filas, es decir, A ha sido premultiplicada
!>               por diag(R).
!>       = 'C':  Equilibrado de columnas, es decir, A ha sido postmultiplicada
!>               por diag(C).
!>       = 'B':  Equilibrado tanto de filas como de columnas, es decir, A ha
!>               sido reemplazada por diag(R) * A * diag(C).
!>               El lado derecho B se ha modificado en consecuencia.
!>

N

!>          N es INTEGER
!>     El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>     El número de lados derechos, es decir, el número de columnas
!>     de las matrices B y X.  NRHS >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, de dimensión (LDA,N)
!>     La matriz original A de N por N.
!>

LDA

!>          LDA es INTEGER
!>     La dimensión principal del array A.  LDA >= max(1,N).
!>

AF

!>          AF es un array DOUBLE PRECISION, de dimensión (LDAF,N)
!>     Los factores L y U de la factorización A = P*L*U
!>     tal como la calcula DGETRF.
!>

LDAF

!>          LDAF es INTEGER
!>     La dimensión principal del array AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>     Los índices de pivote de DGETRF; para 1<=i<=N, la fila i de la
!>     matriz se intercambió con la fila IPIV(i).
!>

R

!>          R es un array DOUBLE PRECISION, de dimensión (N)
!>     Los factores de escala de fila para A.  Si EQUED = 'R' o 'B', A es
!>     multiplicada a la izquierda por diag(R); si EQUED = 'N' o 'C', R
!>     no se accede.
!>     Si se accede a R, cada elemento de R debe ser una potencia de la base
!>     para asegurar una solución y estimaciones de error fiables.
!>     El escalado por potencias de la base no causa errores de redondeo
!>     a menos que el resultado sea subdesbordamiento o sobredesbordamiento.
!>     Los errores de redondeo durante el escalado conducen a refinar con una
!>     matriz que no es equivalente a la matriz de entrada, produciendo
!>     estimaciones de error que pueden no ser fiables.
!>

C

!>          C es un array DOUBLE PRECISION, de dimensión (N)
!>     Los factores de escala de columna para A.  Si EQUED = 'C' o 'B', A es
!>     multiplicada a la derecha por diag(C); si EQUED = 'N' o 'R', C
!>     no se accede.
!>     Si se accede a C, cada elemento de C debe ser una potencia de la base
!>     para asegurar una solución y estimaciones de error fiables.
!>     El escalado por potencias de la base no causa errores de redondeo
!>     a menos que el resultado sea subdesbordamiento o sobredesbordamiento.
!>     Los errores de redondeo durante el escalado conducen a refinar con una
!>     matriz que no es equivalente a la matriz de entrada, produciendo
!>     estimaciones de error que pueden no ser fiables.
!>

B

!>          B es un array DOUBLE PRECISION, de dimensión (LDB,NRHS)
!>     La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>     La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es un array DOUBLE PRECISION, de dimensión (LDX,NRHS)
!>     En la entrada, la matriz solución X, tal como la calcula DGETRS.
!>     En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>     La dimensión principal del array X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>     Condición recíproca escalada. Esta es una estimación de la
!>     condición recíproca de Skeel de la matriz A después del
!>     equilibrado (si se realizó). Si esto es menor que la precisión
!>     de la máquina (en particular, si es cero), la matriz es singular
!>     a la precisión de trabajo. Tenga en cuenta que el error aún puede
!>     ser pequeño incluso si este número es muy pequeño y la matriz
!>     parece mal condicionada.
!>

BERR

!>          BERR es un array DOUBLE PRECISION, de dimensión (NRHS)
!>     Error relativo de retroceso componente a componente. Este es el
!>     error relativo de retroceso componente a componente de cada vector
!>     solución X(j) (es decir, el cambio relativo más pequeño en cualquier
!>     elemento de A o B que haga que X(j) sea una solución exacta).
!>

N_ERR_BNDS

!>          N_ERR_BNDS es INTEGER
!>     Número de límites de error a devolver para cada lado derecho
!>     y cada tipo (normwise o componente a componente). Ver ERR_BNDS_NORM y
!>     ERR_BNDS_COMP a continuación.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM es
M es un array DOUBLE PRECISION, dimensión (NRHS, N_ERR_BNDS)
!>     Para cada segundo miembro, este array contiene información sobre
!>     varios límites de error y números de condición correspondientes al
!>     error relativo en norma, que se define de la siguiente manera:
!>
!>     Error relativo en norma en el i-ésimo vector solución:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     El array se indexa por el tipo de información de error descrita
!>     a continuación. Actualmente se devuelven hasta tres piezas de
!>     información.
!>
!>     El primer índice en ERR_BNDS_NORM(i,:) corresponde al i-ésimo
!>     segundo miembro.
!>
!>     El segundo índice en ERR_BNDS_NORM(:,err) contiene los siguientes
!>     tres campos:
!>     err = 1  booleano. Confiar en la respuesta si el
!>              número de condición recíproco es menor que el umbral
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  límite de error: El error hacia adelante estimado,
!>              casi con certeza dentro de un factor de 10 del error real
!>              siempre que la siguiente entrada sea mayor que el umbral
!>              sqrt(n) * dlamch('Epsilon'). Este límite de error solo
!>              debe confiarse si el booleano anterior es verdadero.
!>
!>     err = 3  Número de condición recíproco: Número de condición
!>              recíproco en norma estimado. Se compara con el umbral
!>              sqrt(n) * dlamch('Epsilon') para determinar si la
!>              estimación de error es . Estos números de condición
!>              recíprocos son 1 / (norm(Z^{-1},inf) * norm(Z,inf)) para
!>              alguna matriz Z escalada apropiadamente.
!>              Sea Z = S*A, donde S escala cada fila por una potencia de
!>              la base para que todas las sumas absolutas de filas de Z
!>              sean aproximadamente 1.
!>
!>     Vea la Nota de Trabajo de Lapack 165 para más detalles y
!>     precauciones adicionales.
!>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP es un array DOUBLE PRECISION, dimensión (NRHS, N_ERR_BNDS)
!>     Para cada segundo miembro, este array contiene información sobre
!>     varios límites de error y números de condición correspondientes al
!>     error relativo por componentes, que se define de la siguiente
!>     manera:
!>
!>     Error relativo por componentes en el i-ésimo vector solución:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     El array se indexa por el segundo miembro i (del cual depende el
!>     error relativo por componentes), y el tipo de información de
!>     error descrita a continuación. Actualmente se devuelven hasta tres
!>     piezas de información para cada segundo miembro. Si no se solicita
!>     precisión por componentes (PARAMS(3) = 0.0), entonces no se accede
!>     a ERR_BNDS_COMP. Si N_ERR_BNDS < 3, entonces se devuelven como
!>     máximo las primeras entradas (:,N_ERR_BNDS).
!>
!>     El primer índice en ERR_BNDS_COMP(i,:) corresponde al i-ésimo
!>     segundo miembro.
!>
!>     El segundo índice en ERR_BNDS_COMP(:,err) contiene los siguientes
!>     tres campos:
!>     err = 1  booleano. Confiar en la respuesta si el
!>              número de condición recíproco es menor que el umbral
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  límite de error: El error hacia adelante estimado,
!>              casi con certeza dentro de un factor de 10 del error real
!>              siempre que la siguiente entrada sea mayor que el umbral
!>              sqrt(n) * dlamch('Epsilon'). Este límite de error solo
!>              debe confiarse si el booleano anterior es verdadero.
!>
!>     err = 3  Número de condición recíproco: Número de condición
!>              recíproco por componentes estimado. Se compara con el
!>              umbral sqrt(n) * dlamch('Epsilon') para determinar si la
!>              estimación de error es . Estos números de condición
!>              recíprocos son 1 / (norm(Z^{-1},inf) * norm(Z,inf)) para
!>              alguna matriz Z escalada apropiadamente.
!>              Sea Z = S*(A*diag(x)), donde x es la solución para el
!>              segundo miembro actual y S escala cada fila de
!>              A*diag(x) por una potencia de la base para que todas las
!>              sumas absolutas de filas de Z sean aproximadamente 1.
!>
!>     Vea la Nota de Trabajo de Lapack 165 para más detalles y
!>     precauciones adicionales.
!>

NPARAMS

!>          NPARAMS es INTEGER
!>     Especifica el número de parámetros establecidos en PARAMS. Si <= 0,
!>     el array PARAMS nunca se referencia y se usan valores
!>     predeterminados.
!>

PARAMS

!>          PARAMS es un array DOUBLE PRECISION, dimensión (NPARAMS)
!>     Especifica los parámetros del algoritmo. Si una entrada es < 0.0,
!>     entonces esa entrada se rellenará con el valor predeterminado
!>     usado para ese parámetro. Solo se accede a las posiciones hasta
!>     NPARAMS; se usan valores predeterminados para los parámetros de
!>     número más alto.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : Si realizar refinamiento
!>            iterativo o no.
!>         Predeterminado: 1.0D+0
!>            = 0.0:  No se realiza refinamiento y no se calculan
!>                    límites de error.
!>            = 1.0:  Usar el algoritmo de refinamiento de doble
!>                    precisión, posiblemente con cálculos
!>                    doble-simple si el entorno de compilación no
!>                    soporta DOUBLE PRECISION.
!>              (otros valores están reservados para uso futuro)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : Número máximo de cálculos de
!>            residuo permitidos para el refinamiento.
!>         Predeterminado: 10
!>         Agresivo: Establecer a 100 para permitir la convergencia
!>                     usando factorizaciones aproximadas o
!>                     factorizaciones distintas de LU. Si la
!>                     factorización usa una técnica distinta de la
!>                     eliminación gaussiana, las garantías en
!>                     err_bnds_norm y err_bnds_comp pueden dejar de ser
!>                     confiables.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : Bandera que determina si el
!>            código intentará encontrar una solución con pequeño error
!>            relativo por componentes en el algoritmo de doble
!>            precisión. Positivo es verdadero, 0.0 es falso.
!>         Predeterminado: 1.0 (intentar convergencia por componentes)
!>

WORK

!>          WORK es un array DOUBLE PRECISION, dimensión (4*N)
!>

IWORK

!>          IWORK es un array INTEGER, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>       = 0:  Salida exitosa. La solución para cada segundo miembro está
!>         garantizada.
!>       < 0:  Si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>       > 0 y <= N:  U(INFO,INFO) es exactamente cero. La factorización
!>         se ha completado, pero el factor U es exactamente singular, por
!>         lo que no se pudieron calcular la solución y los límites de
!>         error. Se devuelve RCOND = 0.
!>       = N+J: La solución correspondiente al J-ésimo segundo miembro no
!>         está garantizada. Las soluciones correspondientes a otros
!>         segundos miembros K con K > J pueden no estar garantizadas
!>         también, pero solo se informa del primero de ellos. Si no se
!>         solicita un pequeño error por componentes (PARAMS(3) = 0.0)
!>         entonces el J-ésimo segundo miembro es el primero con un
!>         límite de error en norma que no está garantizado (el menor J
!>         tal que ERR_BNDS_NORM(J,1) = 0.0). Por defecto (PARAMS(3) =
!>         1.0) el J-ésimo segundo miembro es el primero con un límite de
!>         error en norma o por componentes que no está garantizado (el
!>         menor J tal que ERR_BNDS_NORM(J,1) = 0.0 o ERR_BNDS_COMP(J,1)
!>         = 0.0). Vea la definición de ERR_BNDS_NORM(:,1) y
!>         ERR_BNDS_COMP(:,1). Para obtener información sobre todos los
!>         segundos miembros, verifique ERR_BNDS_NORM o ERR_BNDS_COMP.
!>
