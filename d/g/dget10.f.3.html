subroutine dget10 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision result)

DGET10

Propósito:

!>
!> DGET10 compara dos matrices A y B y calcula la relación
!> RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de las matrices A y B.
!>

N

!>          N es ENTERO
!>          El número de columnas de las matrices A y B.
!>

A

!>          A es un array de DOBLE PRECISIÓN, de dimensión (LDA,N)
!>          La matriz A de m por n.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

B

!>          B es un array de DOBLE PRECISIÓN, de dimensión (LDB,N)
!>          La matriz B de m por n.
!>

LDB

!>          LDB es ENTERO
!>          La dimensión principal del array B.  LDB >= max(1,M).
!>

WORK

!>          WORK es un array de DOBLE PRECISIÓN, de dimensión (M)
!>

RESULT

!>          RESULT es DOBLE PRECISIÓN
!>          RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>
