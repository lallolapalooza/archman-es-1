subroutine dgetsqrhrt (integer m, integer n, integer mb1, integer nb1, integer nb2, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) work, integer lwork, integer info)

DGETSQRHRT

Propósito:

!>
!> DGETSQRHRT calcula una factorización QR bloqueada de columnas de tamaño NB2
!> de una matriz real M por N, A, con M >= N,
!>
!>    A = Q * R.
!>
!> La rutina utiliza internamente una factorización TSQR bloqueada de columnas
!> de tamaño NB1 y bloqueada de filas de tamaño MB1, y realiza la reconstrucción
!> de los vectores de Householder a partir de la salida de TSQR. La rutina también
!> convierte el factor R_tsqr de la salida de la factorización TSQR en
!> el factor R que corresponde a la factorización QR de Householder,
!>
!>    A = Q_tsqr * R_tsqr = Q * R.
!>
!> Los factores de salida Q y R se almacenan en el mismo formato que en DGEQRT
!> (Q está en representación WY compacta y bloqueada). Consulte la documentación
!> de DGEQRT para más detalles sobre el formato.
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es ENTERO
!>          El número de columnas de la matriz A. M >= N >= 0.
!>

MB1

!>          MB1 es ENTERO
!>          El tamaño del bloque de filas que se utilizará en la TSQR bloqueada.
!>          MB1 > N.
!>

NB1

!>          NB1 es ENTERO
!>          El tamaño del bloque de columnas que se utilizará en la TSQR bloqueada.
!>          N >= NB1 >= 1.
!>

NB2

!>          NB2 es ENTERO
!>          El tamaño del bloque que se utilizará en la QR bloqueada que se
!>          produce. NB2 >= 1.
!>

A

!>          A es un array de DOBLE PRECISIÓN, de dimensión (LDA,N)
!>
!>          En la entrada: una matriz A de M por N.
!>
!>          En la salida:
!>           a) los elementos en y por encima de la diagonal
!>              del array contienen la matriz triangular superior N por N
!>              R correspondiente a la QR de Householder;
!>           b) los elementos por debajo de la diagonal representan Q mediante
!>              las columnas de V bloqueado (representación WY compacta).
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

T

!>          T es un array de DOBLE PRECISIÓN, de dimensión (LDT,N))
!>          Los reflectores de bloque triangulares superiores almacenados en forma compacta
!>          como una secuencia de bloques triangulares superiores.
!>

LDT

!>          LDT es ENTERO
!>          La dimensión principal del array T.  LDT >= NB2.
!>

WORK

!>          (espacio de trabajo) array de DOBLE PRECISIÓN, de dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es ENTERO
!>          La dimensión del array WORK.
!>          Si MIN(M,N) = 0, LWORK >= 1, en caso contrario
!>          LWORK >= MAX( 1, LWT + LW1, MAX( LWT+N*N+LW2, LWT+N*N+N ) ),
!>          donde
!>             NUM_BLOQUES_FILA_TOTALES = CEIL((M-N)/(MB1-N)),
!>             NB1LOCAL = MIN(NB1,N).
!>             LWT = NUM_BLOQUES_FILA_TOTALES * N * NB1LOCAL,
!>             LW1 = NB1LOCAL * N,
!>             LW2 = NB1LOCAL * MAX( NB1LOCAL, ( N - NB1LOCAL ) ).
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo.
!>          La rutina solo calcula el tamaño óptimo del array WORK, devuelve este valor
!>          como la primera entrada del array WORK, y XERBLA no emite ningún mensaje
!>          de error relacionado con LWORK.
!>

INFO

!>          INFO es ENTERO
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
