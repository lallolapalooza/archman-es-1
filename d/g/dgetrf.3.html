subroutine dgetrf (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

DGETRF

Propósito:

!>
!> DGETRF calcula una factorización LU de una matriz general M por N A
!> utilizando pivoteo parcial con intercambios de filas.
!>
!> La factorización tiene la forma
!>    A = P * L * U
!> donde P es una matriz de permutación, L es triangular inferior con unidad
!> elementos diagonales (trapezoidal inferior si m > n), y U es superior
!> triangular (trapezoidal superior si m < n).
!>
!> Esta es la versión de Nivel 3 BLAS de mirada derecha del algoritmo.
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es ENTERO
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array de DOBLE PRECISIÓN, de dimensión (LDA,N)
!>          En la entrada, la matriz M por N a factorizar.
!>          En la salida, los factores L y U de la factorización
!>          A = P*L*U; los elementos diagonales unitarios de L no se almacenan.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV es un array de ENTEROS, de dimensión (min(M,N))
!>          Los índices de pivote; para 1 <= i <= min(M,N), la fila i de la
!>          matriz fue intercambiada con la fila IPIV(i).
!>

INFO

!>          INFO es ENTERO
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, U(i,i) es exactamente cero. La factorización
!>                se ha completado, pero el factor U es exactamente
!>                singular, y ocurrirá una división por cero si se utiliza
!>                para resolver un sistema de ecuaciones.
!>
