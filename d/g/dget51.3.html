Documentación de Función/Subrutina

subroutine dget51 (integer itype, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( * ) work, double precision result)

DGET51

Propósito:

!>
!>      DGET51  generalmente verifica una descomposición de la forma
!>
!>              A = U B V'
!>
!>      donde ' significa transpuesta y U y V son ortogonales.
!>
!>      Específicamente, si ITYPE=1
!>
!>              RESULTADO = | A - U B V' | / ( |A| n ulp )
!>
!>      Si ITYPE=2, entonces:
!>
!>              RESULTADO = | A - B | / ( |A| n ulp )
!>
!>      Si ITYPE=3, entonces:
!>
!>              RESULTADO = | I - UU' | / ( n ulp )
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          Especifica el tipo de pruebas a realizar.
!>          =1: RESULTADO = | A - U B V' | / ( |A| n ulp )
!>          =2: RESULTADO = | A - B | / ( |A| n ulp )
!>          =3: RESULTADO = | I - UU' | / ( n ulp )
!>

N

!>          N es INTEGER
!>          El tamaño de la matriz. Si es cero, DGET51 no hace nada.
!>          Debe ser al menos cero.
!>

A

!>          A es array DOUBLE PRECISION, dimension (LDA, N)
!>          La matriz original (no factorizada).
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de A. Debe ser al menos 1
!>          y al menos N.
!>

B

!>          B es array DOUBLE PRECISION, dimension (LDB, N)
!>          La matriz factorizada.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de B. Debe ser al menos 1
!>          y al menos N.
!>

U

!>          U es array DOUBLE PRECISION, dimension (LDU, N)
!>          La matriz ortogonal en el lado izquierdo de la
!>          descomposición.
!>          No referenciada si ITYPE=2
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal de U. LDU debe ser al menos N y
!>          al menos 1.
!>

V

!>          V es array DOUBLE PRECISION, dimension (LDV, N)
!>          La matriz ortogonal en el lado izquierdo de la
!>          descomposición.
!>          No referenciada si ITYPE=2
!>

LDV

!>          LDV es INTEGER
!>          La dimensión principal de V. LDV debe ser al menos N y
!>          al menos 1.
!>

WORK

!>          WORK es array DOUBLE PRECISION, dimension (2*N**2)
!>

RESULTADO

!>          RESULTADO es DOUBLE PRECISION
!>          Los valores calculados por la prueba especificada por ITYPE.
!>          El valor está actualmente limitado a 1/ulp, para evitar
!>          desbordamiento.
!>          Los errores se marcan con RESULTADO=10/ulp.
!>
