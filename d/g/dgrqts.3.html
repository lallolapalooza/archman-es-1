subroutine dgrqts (integer m, integer p, integer n, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) q, double precision, dimension( lda, * ) r, integer lda, double precision, dimension( * ) taua, double precision, dimension( ldb, * ) b, double precision, dimension( ldb, * ) bf, double precision, dimension( ldb, * ) z, double precision, dimension( ldb, * ) t, double precision, dimension( ldb, * ) bwk, integer ldb, double precision, dimension( * ) taub, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

DGRQTS

Propósito:

!>
!> DGRQTS prueba DGGRQF, que calcula la factorización GRQ de una
!> matriz A de M por N y una matriz B de P por N: A = R*Q y B = Z*T*Q.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz B.  P >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B.  N >= 0.
!>

A

!>          A es array DOUBLE PRECISION, dimension (LDA,N)
!>          La matriz A de M por N.
!>

AF

!>          AF es array DOUBLE PRECISION, dimension (LDA,N)
!>          Detalles de la factorización GRQ de A y B, según lo devuelto
!>          por DGGRQF, ver SGGRQF para más detalles.
!>

Q

!>          Q es array DOUBLE PRECISION, dimension (LDA,N)
!>          La matriz ortogonal Q de N por N.
!>

R

!>          R es array DOUBLE PRECISION, dimension (LDA,MAX(M,N))
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, R y Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA es array DOUBLE PRECISION, dimension (min(M,N))
!>          Los factores escalares de los reflectores elementales, según lo devuelto
!>          por DGGQRC.
!>

B

!>          B es array DOUBLE PRECISION, dimension (LDB,N)
!>          En la entrada, la matriz A de P por N.
!>

BF

!>          BF es array DOUBLE PRECISION, dimension (LDB,N)
!>          Detalles de la factorización GQR de A y B, según lo devuelto
!>          por DGGRQF, ver SGGRQF para más detalles.
!>

Z

!>          Z es array DOUBLE PRECISION, dimension (LDB,P)
!>          La matriz ortogonal Z de P por P.
!>

T

!>          T es array DOUBLE PRECISION, dimension (LDB,max(P,N))
!>

BWK

!>          BWK es array DOUBLE PRECISION, dimension (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de los arrays B, BF, Z y T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB es array DOUBLE PRECISION, dimension (min(P,N))
!>          Los factores escalares de los reflectores elementales, según lo devuelto
!>          por DGGRQF.
!>

WORK

!>          WORK es array DOUBLE PRECISION, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK, LWORK >= max(M,P,N)**2.
!>

RWORK

!>          RWORK es array DOUBLE PRECISION, dimension (M)
!>

RESULT

!>          RESULT es array DOUBLE PRECISION, dimension (4)
!>          Los ratios de prueba:
!>            RESULT(1) = norm( R - A*Q' ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Q - Z'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( N*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
