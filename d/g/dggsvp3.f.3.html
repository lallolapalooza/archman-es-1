subroutine dggsvp3 (character jobu, character jobv, character jobq, integer m, integer p, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision tola, double precision tolb, integer k, integer l, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) iwork, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DGGSVP3

Propósito:

!>
!> DGGSVP3 calcula las matrices ortogonales U, V y Q tales que
!>
!>                    N-K-L  K    L
!>  U**T*A*Q =     K ( 0    A12  A13 )  si M-K-L >= 0;
!>                 L ( 0     0   A23 )
!>             M-K-L ( 0     0    0  )
!>
!>                  N-K-L  K    L
!>         =     K ( 0    A12  A13 )  si M-K-L < 0;
!>             M-K ( 0     0   A23 )
!>
!>                  N-K-L  K    L
!>  V**T*B*Q =   L ( 0     0   B13 )
!>             P-L ( 0     0    0  )
!>
!> donde la matriz A12 de KxK y la matriz B13 de LxL son no singulares
!> triangulares superiores; A23 es triangular superior LxL si M-K-L >= 0,
!> de lo contrario A23 es trapezoidal superior (M-K)xL. K+L = el rango numérico efectivo
!> de la matriz (M+P)xN (A**T,B**T)**T.
!>
!> Esta descomposición es el paso de preprocesamiento para calcular la
!> Descomposición de Valores Singulares Generalizada (GSVD), véase la subrutina
!> DGGSVD3.
!>

Parámetros

JOBU 
!>          JOBU es CHARACTER*1
!>          = 'U':  Se calcula la matriz ortogonal U;
!>          = 'N':  U no se calcula.
!>

JOBV

!>          JOBV es CHARACTER*1
!>          = 'V':  Se calcula la matriz ortogonal V;
!>          = 'N':  V no se calcula.
!>

JOBQ

!>          JOBQ es CHARACTER*1
!>          = 'Q':  Se calcula la matriz ortogonal Q;
!>          = 'N':  Q no se calcula.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz B.  P >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B.  N >= 0.
!>

A

!>          A es array DOUBLE PRECISION, dimension (LDA,N)
!>          En la entrada, A contiene la matriz A de MxN.
!>          En la salida, A contiene la matriz triangular (o trapezoidal)
!>          descrita en la sección Propósito.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,M).
!>

B

!>          B es array DOUBLE PRECISION, dimension (LDB,N)
!>          En la entrada, B contiene la matriz PxN.
!>          En la salida, B contiene la matriz triangular descrita en
!>          la sección Propósito.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,P).
!>

TOLA

!>          TOLA es DOUBLE PRECISION
!>

TOLB

!>          TOLB es DOUBLE PRECISION
!>
!>          TOLA y TOLB son los umbrales para determinar el rango numérico efectivo
!>          de la matriz B y un subbloque de A. Generalmente,
!>          se establecen como
!>             TOLA = MAX(M,N)*norm(A)*MACHEPS,
!>             TOLB = MAX(P,N)*norm(B)*MACHEPS.
!>          El tamaño de TOLA y TOLB puede afectar los errores de retroceso
!>          de la descomposición.
!>

K

!>          K es INTEGER
!>

L

!>          L es INTEGER
!>
!>          En la salida, K y L especifican la dimensión de los subbloques
!>          descritos en la sección Propósito.
!>          K + L = rango numérico efectivo de (A**T,B**T)**T.
!>

U

!>          U es array DOUBLE PRECISION, dimension (LDU,M)
!>          Si JOBU = 'U', U contiene la matriz ortogonal U.
!>          Si JOBU = 'N', U no se referencia.
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal del array U. LDU >= max(1,M) si
!>          JOBU = 'U'; LDU >= 1 en caso contrario.
!>

V

!>          V es array DOUBLE PRECISION, dimension (LDV,P)
!>          Si JOBV = 'V', V contiene la matriz ortogonal V.
!>          Si JOBV = 'N', V no se referencia.
!>

LDV

!>          LDV es INTEGER
!>          La dimensión principal del array V. LDV >= max(1,P) si
!>          JOBV = 'V'; LDV >= 1 en caso contrario.
!>

Q

!>          Q es array DOUBLE PRECISION, dimension (LDQ,N)
!>          Si JOBQ = 'Q', Q contiene la matriz ortogonal Q.
!>          Si JOBQ = 'N', Q no se referencia.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q. LDQ >= max(1,N) si
!>          JOBQ = 'Q'; LDQ >= 1 en caso contrario.
!>

IWORK

!>          IWORK es array INTEGER, dimension (N)
!>

TAU

!>          TAU es array DOUBLE PRECISION, dimension (N)
!>

WORK

!>          WORK es array DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK. LWORK >= 1.
!>
!>          Si LWORK = -1, se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no emite
!>          ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
