DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine dget03 (integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldainv, * ) ainv, integer ldainv, double precision, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

DGET03

Propósito:

!>
!> DGET03 calcula el residuo de una matriz general multiplicada por su inversa:
!>    norm( I - AINV*A ) / ( N * norm(A) * norm(AINV) * EPS ),
!> donde EPS es el épsilon de la máquina.
!>

Parámetros

N 
!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array DOUBLE PRECISION, dimension (LDA,N)
!>          La matriz original N x N A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

AINV

!>          AINV es un array DOUBLE PRECISION, dimension (LDAINV,N)
!>          La inversa de la matriz A.
!>

LDAINV

!>          LDAINV es INTEGER
!>          La dimensión principal del array AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK es un array DOUBLE PRECISION, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK es INTEGER
!>          La dimensión principal del array WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          El recíproco del número de condición de A, calculado como
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          norm(I - AINV*A) / ( N * norm(A) * norm(AINV) * EPS )
!>
