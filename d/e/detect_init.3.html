detect_init(3)

Paquete: libchardet

Licencias: MPL

detect_init(3) Manual de funciones de biblioteca detect_init(3)

NOMBRE

detect_init - inicializa la estructura Detect

SINOPSIS

#include <chardet.h>
Detect * chardet_init (void);

DESCRIPCIÓN

inicializa la estructura Detect.

VALOR DEVUELTO

Devuelve NULL si falla y devuelve la estructura Detect en caso de éxito.

typedef struct Detect_t {
	Detector *detect;
} Detect;

EJEMPLO

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "falló la inicialización del manejador chardet\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	detect_reset (&d);
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "Asignación de memoria fallida\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect_handledata (&d, "¾È³çÇÏ¼¼¿ä", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "Al procesar el manejador, ocurrió falta de memoria\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"El 2do argumento de chardet() debe ser asignado con memoria "
					"mediante la API detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# comprobar soporte de obj->bom con la constante CHARDET_BOM_CHECK
	printf ("codificación: %s, confianza: %f, existe bom: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
	detect_destroy (&d);
    return 0;
}
