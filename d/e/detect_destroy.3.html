detect_destroy(3)

Paquete: libchardet

Licencias: MPL

detect_destroy(3) Manual de Funciones de Biblioteca detect_destroy(3)

NOMBRE

detect_destroy - liberar estructura Detector

SINOPSIS

#include <chardet.h>

void detect_destroy (Detect ** handle);

DESCRIPCIÓN

Libera los recursos asignados por la API detect_init.

VALOR DE RETORNO

void

EJEMPLO

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "falló inicialización del manejador chardet\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	detect_reset (&d);
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "Error de asignación de memoria\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect_handledata (&d, "안녕하세요", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "Al procesar, ocurrió falta de memoria\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"El 3er argumento de chardet_handledata() debe ser asignado con memoria"
					" con la API detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# comprobar el obj->bom soportado con la constante CHARDET_BOM_CHECK
	printf ("codificación: %s, confianza: %f, existe bom: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
	detect_destroy (&d);
    return 0;
}
