detect_obj_init(3) Manual de funciones de la biblioteca detect_obj_init(3)

NOMBRE

detect_obj_init - inicializar la estructura DetectObject

SINOPSIS

#include <chardet.h>
DetectObj chardet_obj_init (void);

DESCRIPCIÓN

inicializa la estructura DetectObj que almacena información de chardet.

VALOR DE RETORNO

Devuelve la estructura DetectObject. El valor devuelto debe ser liberado por la API chardet_obj_free.

typedef struct DetectObject {
	char * encoding;
	float confidence;
	short bom;
} DetectObj;

EJEMPLO

#include <chardet.h>
int main (void) {
	DetectObj *obj;
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "Fallo de asignación de memoria\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect ("¾È³çÇÏ¼¼¿ä", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "En el procesamiento del manejo, ocurrió falta de memoria\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"El segundo argumento de chardet() debe ser una asignación de memoria "
					"con la API detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# comprueba el soporte obj->bom con la constante CHARDET_BOM_CHECK
	printf ("codificación: %s, confianza: %f, existe bom: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
    return 0;
}
