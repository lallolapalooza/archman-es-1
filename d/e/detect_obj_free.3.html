detect_obj_free(3) Manual de Funciones de Biblioteca detect_obj_free(3)

NOMBRE

chardet_obj_free - liberar la estructura DetectObject

SINOPSIS

#include <chardet.h>
void chardet_obj_free (DetectObj **);

DESCRIPCIÓN

Libera los recursos de la estructura DetectObject que fueron asignados por la API chardet_obj_init.

VALOR DE RETORNO

void

EJEMPLO

#include <chardet.h>
int main (void) {
	DetectObj *obj;
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "Fallo en la asignación de memoria\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect ("¾È³çÇÏ¼¼¿ä", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "Al procesar el manejador, ocurrió falta de memoria\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"El segundo argumento de chardet() debe ser asignado en memoria "
					"con la API detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# verifica el soporte de obj->bom con la constante CHARDET_BOM_CHECK
	printf ("codificación: %s, confianza: %f, existe bom: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
    return 0;
}
