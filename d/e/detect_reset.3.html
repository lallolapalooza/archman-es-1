detect_reset(3)

Paquete: libchardet

Licencias: MPL

detect_reset(3) Manual de Funciones de Biblioteca detect_reset(3)

NOMBRE

detect_reset - reinicia la estructura Detect

SINOPSIS

#include <chardet.h>
void chardet_reset (Detect **handle);

DESCRIPCIÓN

Reinicia la estructura Detect

VALOR DE RETORNO

void

EJEMPLO

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	int i, arrayNum;
	char *str[] = {
		"this is ascii",
		"ÀÌ°Ç euc-kr ÀÔ´Ï´Ù."
	};
	arrayNum = sizeof (str) / sizeof (str[0]);
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "falló la inicialización del manejador de chardet\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	for ( i=0; i<arrayNum; i++ ) {
		detect_reset (&d);
		if ( (obj = detect_obj_init ()) == NULL ) {
			fprintf (stderr, "Error de asignación de memoria\n");
			return CHARDET_MEM_ALLOCATED_FAIL;
		}
		switch (detect_handledata (&d, "¾È³çÇÏ¼¼¿ä", &obj)) {
			case CHARDET_OUT_OF_MEMORY :
				fprintf (stderr, "En el procesamiento del manejador, ocurrió falta de memoria\n");
				detect_obj_free (&obj);
				return CHARDET_OUT_OF_MEMORY;
			case CHARDET_NULL_OBJECT :
				fprintf (stderr,
						"El segundo argumento de chardet() debe ser asignación de memoria "
						"con la API detect_obj_init\n");
				return CHARDET_NULL_OBJECT;
		}
		# comprueba el objeto soportado obj->bom con la constante CHARDET_BOM_CHECK
		printf ("codificación: %s, confianza: %f, existe bom: %d\n", obj->encoding, obj->confidence, obj->bom);
		detect_obj_free (&obj);
	}
	detect_destroy (&d);
    return 0;
}
