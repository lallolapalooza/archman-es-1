ALASMG

Propósito:

!>
!> ALASMG imprime un resumen de los resultados de una de las rutinas -CHK-.
!>

Parámetros

TYPE 
!>          TYPE es CHARACTER*3
!>          El nombre de la ruta LAPACK.
!>

NOUT

!>          NOUT es INTEGER
!>          La unidad en la que se deben imprimir los resultados.
!>          NOUT >= 0.
!>

NFAIL

!>          NFAIL es INTEGER
!>          El número de pruebas que no superaron la relación umbral.
!>

NRUN

!>          NRUN es INTEGER
!>          El número total de pruebas.
!>

NERRS

!>          NERRS es INTEGER
!>          El número de mensajes de error registrados.
!>
