DESCRIPCIÓ

N

La interfaz de E/S asíncrona POSIX (AIO) permite a las aplicaciones iniciar una o más operaciones de E/S que se realizan de forma asíncrona (es decir, en segundo plano). La aplicación puede optar por ser notificada de la finalización de la operación de E/S de diversas maneras: mediante la entrega de una señal, mediante la instanciación de un hilo, o sin notificación

alguna.

La interfaz POSIX AIO consta de las siguientes

funciones:
aio_read(
3)
Encola una solicitud de lectura. Este es el análogo asíncrono de read(
2).
aio_write(
3)
Encola una solicitud de escritura. Este es el análogo asíncrono de write(
2).
aio_fsync(
3)
Encola una solicitud de sincronización para las operaciones de E/S en un descriptor de archivo. Este es el análogo asíncrono de fsync(2) y fdatasync
(2).
aio_error(
3)
Obtiene el estado de error de una solicitud de E/S
encolada.
aio_return(
3)
Obtiene el estado de retorno de una solicitud de E/S
completada.
aio_suspend(
3)
Suspende al llamante hasta que una o más de un conjunto especificado de solicitudes de E/S se
completen.
aio_cancel(
3)
Intenta cancelar solicitudes de E/S pendientes en un descriptor de archivo
especificado.

El aiocb ("bloque de control de E/S asíncrona") define parámetros que controlan una operación de E/S. Se emplea un argumento de este tipo con todas las funciones listadas anteriormente. Esta estructura tiene la siguiente

forma: 


#include <aiocb.h>
struct aiocb {
    /* El orden de estos campos depende de la implementación */
    int             aio_fildes;     /* Descriptor de archivo */
    off_t           aio_offset;     /* Desplazamiento del archivo */
    volatile void  *aio_buf;        /* Ubicación del búfer */
    size_t          aio_nbytes;     /* Longitud de la transferencia */
    int             aio_reqprio;    /* Prioridad de la solicitud */
    struct sigevent aio_sigevent;   /* Método de notificación */
    int             aio_lio_opcode; /* Operación a realizar;
                                       solo lio_listio() */
    /* Varios campos internos de implementación no mostrados */
};
/* Códigos de operación para 'aio_lio_opcode': */
enum { LIO_READ, LIO_WRITE, LIO_NOP }
;

NOTA

S

Es una buena idea poner a cero el búfer del bloque de control antes de usarlo (ver memset(3)). El búfer del bloque de control y el búfer al que apunta por aio_buf no deben modificarse mientras la operación de E/S está en curso. Estos búferes deben permanecer válidos hasta que la operación de E/S se

complete.

Las operaciones de lectura o escritura asíncronas simultáneas que utilizan el mismo estructura aiocb producen resultados

indefinidos.

La implementación actual de POSIX AIO en Linux es proporcionada en espacio de usuario por glibc. Esta tiene una serie de limitaciones, siendo la más notable que mantener múltiples hilos para realizar operaciones de E/S es costoso y escala mal. El trabajo ha estado en progreso durante algún tiempo en una implementación de máquina de estados del kernel de E/S asíncrona (ver io_submit(2), io_setup(2), io_cancel(2), io_destroy(2), io_getevents(2)), pero esta implementación aún no ha madurado hasta el punto en que la implementac

ión POSIX AIO pueda ser completamente reimplemen
tada utilizando las llamadas al sistema del kernel.

EJEMPLO

S

El programa siguiente abre cada uno de los archivos nombrados en sus argumentos de línea de comandos y encola una solicitud en el descriptor de archivo resultante utilizando aio_read(3). El programa luego entra en un bucle, monitoreando periódicamente cada una de las operaciones de E/S que aún están en curso utilizando aio_error(3). Cada una de las solicitudes de E/S está configurada para proporcionar notificación mediante la entrega de una señal. Después de que todas las solicitudes de E/S se hayan completado, el programa recupera su estado utilizando aio_retur

n(3).

La señal SIGQUIT (generada al escribir control-\) hace que el programa solicite la cancelación de cada una de las solicitudes pendientes utilizando aio_cance

l(3).

"x".

Programa

fuente 
#include <err.h>
#include <fcntl.h>
#include <stdlib.h>
#include <unistd.h>
#include <stdio.h>
#include <errno.h>
#include <aio.h>
#include <signal.h>
#define BUF_SIZE 20     /* Tamaño de los búferes para las operaciones de lectura */
struct ioRequest {      /* Estructura definida por la aplicación para rastrear
                           las solicitudes de E/S */
    int           reqNum;
    int           status;
    struct aiocb *aiocbp;
};
static volatile sig_atomic_t gotSIGQUIT = 0;
                        /* Al recibir SIGQUIT, intentamos
                           cancelar todas las solicitudes de E/S pendientes */
static void             /* Manejador para SIGQUIT */
quitHandler(int sig)
{
    gotSIGQUIT = 1;
}
#define IO_SIGNAL SIGUSR1   /* Señal utilizada para notificar la finalización de la E/S */
static void                 /* Manejador para la señal de finalización de la E/S */
aioSigHandler(int sig, siginfo_t *si, void *ucontext)
{
    if (si->si_code == SI_ASYNCIO) {
        write(STDOUT_FILENO, "I/O completion signal received\n", 31);
        /* La estructura ioRequest correspondiente estaría disponible como
               struct ioRequest *ioReq = si->si_value.sival_ptr;
           y el descriptor de archivo estaría disponible a través de
               ioReq->aiocbp->aio_fildes */
    }
}
int
main(int argc, char *argv[])
{
    struct sigaction sa;
    int s;
    int numReqs;        /* Número total de solicitudes de E/S en cola */
    int openReqs;       /* Número de solicitudes de E/S aún en curso */
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <pathname> <pathname>...\n",
                argv[0]);
        exit(EXIT_FAILURE);
    }
    numReqs = argc - 1;
    /* Asignamos nuestros arrays.  */
    struct ioRequest *ioList = calloc(numReqs, sizeof(*ioList));
    if (ioList == NULL)
        err(EXIT_FAILURE, "calloc");
    struct aiocb *aiocbList = calloc(numReqs, sizeof(*aiocbList));
    if (aiocbList == NULL)
        err(EXIT_FAILURE, "calloc");
    /* Establecemos los manejadores para SIGQUIT y la señal de finalización de la E/S.  */
    sa.sa_flags = SA_RESTART;
    sigemptyset(&sa.sa_mask);
    sa.sa_handler = quitHandler;
    if (sigaction(SIGQUIT, &sa, NULL) == -1)
        err(EXIT_FAILURE, "sigaction");
    sa.sa_flags = SA_RESTART | SA_SIGINFO;
    sa.sa_sigaction = aioSigHandler;
    if (sigaction(IO_SIGNAL, &sa, NULL) == -1)
        err(EXIT_FAILURE, "sigaction");
    /* Abrimos cada archivo especificado en la línea de comandos y ponemos en cola
       una solicitud de lectura en el descriptor de archivo resultante.  */
    for (size_t j = 0; j < numReqs; j++) {
        ioList[j].reqNum = j;
        ioList[j].status = EINPROGRESS;
        ioList[j].aiocbp = &aiocbList[j];
        ioList[j].aiocbp->aio_fildes = open(argv[j + 1], O_RDONLY);
        if (ioList[j].aiocbp->aio_fildes == -1)
            err(EXIT_FAILURE, "open");
        printf("opened %s on descriptor %d\n", argv[j + 1],
                ioList[j].aiocbp->aio_fildes);
        ioList[j].aiocbp->aio_buf = malloc(BUF_SIZE);
        if (ioList[j].aiocbp->aio_buf == NULL)
            err(EXIT_FAILURE, "malloc");
        ioList[j].aiocbp->aio_nbytes = BUF_SIZE;
        ioList[j].aiocbp->aio_reqprio = 0;
        ioList[j].aiocbp->aio_offset = 0;
        ioList[j].aiocbp->aio_sigevent.sigev_notify = SIGEV_SIGNAL;
        ioList[j].aiocbp->aio_sigevent.sigev_signo = IO_SIGNAL;
        ioList[j].aiocbp->aio_sigevent.sigev_value.sival_ptr =
                                &ioList[j];
        s = aio_read(ioList[j].aiocbp);
        if (s == -1)
            err(EXIT_FAILURE, "aio_read");
    }
    openReqs = numReqs;
    /* Bucle, monitorizando el estado de las solicitudes de E/S.  */
    while (openReqs > 0) {
        sleep(3);       /* Retraso entre cada paso de monitorización */
        if (gotSIGQUIT) {
            /* Al recibir SIGQUIT, intentamos cancelar cada una de las
               solicitudes de E/S pendientes y mostramos el estado devuelto
               por las solicitudes de cancelación.  */
            printf("got SIGQUIT; canceling I/O requests: \n");
            for (size_t j = 0; j < numReqs; j++) {
                if (ioList[j].status == EINPROGRESS) {
                    printf("    Request %zu on descriptor %d:", j,
                            ioList[j].aiocbp->aio_fildes);
                    s = aio_cancel(ioList[j].aiocbp->aio_fildes,
                            ioList[j].aiocbp);
                    if (s == AIO_CANCELED)
                        printf("I/O canceled\n");
                    else if (s == AIO_NOTCANCELED)
                        printf("I/O not canceled\n");
                    else if (s == AIO_ALLDONE)
                        printf("I/O all done\n");
                    else
                        perror("aio_cancel");
                }
            }
            gotSIGQUIT = 0;
        }
        /* Comprobamos el estado de cada solicitud de E/S que aún está
           en curso.  */
        printf("aio_error():\n");
        for (size_t j = 0; j < numReqs; j++) {
            if (ioList[j].status == EINPROGRESS) {
                printf("    for request %zu (descriptor %d): ",
                        j, ioList[j].aiocbp->aio_fildes);
                ioList[j].status = aio_error(ioList[j].aiocbp);
                switch (ioList[j].status) {
                case 0:
                    printf("I/O succeeded\n");
                    break;
                case EINPROGRESS:
                    printf("In progress\n");
                    break;
                case ECANCELED:
                    printf("Canceled\n");
                    break;
                default:
                    perror("aio_error");
                    break;
                }
                if (ioList[j].status != EINPROGRESS)
                    openReqs--;
            }
        }
    }
    printf("All I/O requests completed\n");
    /* Comprobamos el estado devuelto de todas las solicitudes de E/S.  */
    printf("aio_return():\n");
    for (size_t j = 0; j < numReqs; j++) {
        ssize_t s;
        s = aio_return(ioList[j].aiocbp);
        printf("    for request %zu (descriptor %d): %zd\n",
                j, ioList[j].aiocbp->aio_fildes, s);
    }
    exit(EXIT_SUCCESS)
;
}
