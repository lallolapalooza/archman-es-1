#include <stdlib.h>
#include <string.h>
#include <sys/apparmor.h>
#include <sys/types.h>
#include <sys/stat.h>

#include <stdio.h>
#include <unistd.h>
int main(int argc, char *argv[]) {
       int fd;
       unsigned long tok;
       char buf[10];
       /* random() es una mala elección */
       tok = random();
       /* abrir /etc/passwd fuera de cualquier hat */
       if ((fd=open("/etc/passwd", O_RDONLY)) < 0)
               perror("Fallo al abrir /etc/passwd");
       /* confirmar por nosotros mismos que podemos leer realmente /etc/passwd */
       memset(&buf, 0, 10);
       if (read(fd, &buf, 10) == -1) {
               perror("Fallo al leer /etc/passwd pre-hat");
               _exit(1);
       }
       buf[9] = '\0';
       printf("/etc/passwd: %s\n", buf);
       /* cambiar al subperfil "hat", que no debería tener
        * acceso de lectura a /etc/passwd -- aunque tenemos un
        * descriptor de archivo válido en el momento de la llamada a aa_change_hat(). */
       if (aa_change_hat("hat", tok)) {
               perror("Fallo al cambiar de hat -- abortando");
               _exit(1);
       }
       /* confirmar que no podemos leer /etc/passwd */
       lseek(fd,0,SEEK_SET);
       memset(&buf, 0, 10);
       if (read(fd, &buf, 10) == -1)
               perror("Fallo al leer /etc/passwd post-hat");
       buf[9] = '\0';
       printf("/etc/passwd: %s\n", buf);
       return 0;
}
