EJEMPLO

S

En el siguiente ejemplo, un objeto 'MyObject' se firma usando la clave contenida en un EVP_MD_CTX. La firma se escribe en MyObject.signature. Luego, el objeto se emite en formato DER y después se carga de nuevo y se

verifica. 
 #include <openssl/x509.h>
 #include <openssl/asn1t.h>
 /* Un objeto utilizado para almacenar los campos de datos ASN1 que serán firmados */
 typedef struct MySignInfoObject_st
 {
     ASN1_INTEGER *version;
     X509_ALGOR sig_alg;
 } MySignInfoObject;
 DECLARE_ASN1_FUNCTIONS(MySignInfoObject)
 /*
  * Un objeto de nivel superior que contiene los campos ASN1, el algoritmo de firma y
  * la firma de salida.
  */
 typedef struct MyObject_st
 {
     MySignInfoObject info;
     X509_ALGOR sig_alg;
     ASN1_BIT_STRING *signature;
 } MyObject;
 DECLARE_ASN1_FUNCTIONS(MyObject)
 /* La definición ASN1 de MySignInfoObject */
 ASN1_SEQUENCE_cb(MySignInfoObject, NULL) = {
     ASN1_SIMPLE(MySignInfoObject, version, ASN1_INTEGER)
     ASN1_EMBED(MySignInfoObject, sig_alg, X509_ALGOR),
 } ASN1_SEQUENCE_END_cb(MySignInfoObject, MySignInfoObject)
 /* Funciones new, free, d2i & i2d para MySignInfoObject */
 IMPLEMENT_ASN1_FUNCTIONS(MySignInfoObject)
 /* La definición ASN1 de MyObject */
 ASN1_SEQUENCE_cb(MyObject, NULL) = {
     ASN1_EMBED(MyObject, info, MySignInfoObject),
     ASN1_EMBED(MyObject, sig_alg, X509_ALGOR),
     ASN1_SIMPLE(MyObject, signature, ASN1_BIT_STRING)
 } ASN1_SEQUENCE_END_cb(MyObject, MyObject)
 /* Funciones new, free, d2i & i2d para MyObject */
 IMPLEMENT_ASN1_FUNCTIONS(MyObject)
 int test_asn1_item_sign_verify(const char *mdname, EVP_PKEY *pkey, long version)
 {
    int ret = 0;
    unsigned char *obj_der = NULL;
    const unsigned char *p = NULL;
    MyObject *obj = NULL, *loaded_obj = NULL;
    const ASN1_ITEM *it = ASN1_ITEM_rptr(MySignInfoObject);
    EVP_MD_CTX *sctx = NULL, *vctx = NULL;
    int len;
    /* Crear MyObject y establecer su versión */
    obj = MyObject_new();
    if (obj == NULL)
        goto err;
    if (!ASN1_INTEGER_set(obj->info.version, version))
        goto err;
    /* Establecer la clave y el resumen utilizados para firmar */
    sctx = EVP_MD_CTX_new();
    if (sctx == NULL
        || !EVP_DigestSignInit_ex(sctx, NULL, mdname, NULL, NULL, pkey))
        goto err;
    /*
     * it contiene el mapeo entre los datos ASN.1 y el objeto MySignInfoObject
     * obj->info es el objeto 'MySignInfoObject' que será
     *   convertido en datos DER y luego firmado.
     * obj->signature contendrá la firma de salida.
     * obj->sig_alg se rellena con el id del algoritmo de firma de la clave privada.
     * obj->info.sig_alg es otra copia del id del algoritmo de firma que se encuentra
     * dentro de MyObject.
     */
    len = ASN1_item_sign_ctx(it, &obj->sig_alg, &obj->info.sig_alg,
                             obj->signature, &obj->info, sctx);
    if (len <= 0
        || X509_ALGOR_cmp(&obj->sig_alg, &obj->info.sig_alg) != 0)
        goto err;
    /* Emitir MyObject en forma der */
    len = i2d_MyObject(obj, &obj_der);
    if (len <= 0)
        goto err;
    /* Establecer la clave y el resumen utilizados para verificar */
    vctx = EVP_MD_CTX_new();
    if (vctx == NULL
        || !EVP_DigestVerifyInit_ex(vctx, NULL, mdname, NULL, NULL, pkey))
        goto err;
    /* Cargar los datos der de nuevo en un objeto */
    p = obj_der;
    loaded_obj = d2i_MyObject(NULL, &p, len);
    if (loaded_obj == NULL)
        goto err;
    /* Verificar el objeto cargado */
    ret = ASN1_item_verify_ctx(it, &loaded_obj->sig_alg, loaded_obj->signature,
                               &loaded_obj->info, vctx);
err:
    OPENSSL_free(obj_der);
    MyObject_free(loaded_obj);
    MyObject_free(obj);
    EVP_MD_CTX_free(sctx);
    EVP_MD_CTX_free(vctx);
    return ret;
 }
