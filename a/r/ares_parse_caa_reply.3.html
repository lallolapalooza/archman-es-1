ares_parse_caa_reply(3)

Paquete: c-ares

Licencias: MIT

ARES_PARSE_CAA_REPLY(3) Manual de funciones de biblioteca ARES_PARSE_CAA_REPLY(3)

NOMBRE

ares_parse_caa_reply - Analizar una respuesta a una consulta DNS de tipo CAA

SINOPSIS

#include <ares.h>
int ares_parse_caa_reply(const unsigned char* abuf, int alen,
                         struct ares_caa_reply **caa_out);

DESCRIPCIÓN

La función ares_parse_caa_reply analiza la respuesta a una consulta de tipo CAA en una lista enlazada (un elemento por subcadena) de estructuras struct ares_caa_reply. Los parámetros abuf y alen proporcionan el contenido de la respuesta. El resultado se almacena en memoria asignada y un puntero a ella se almacena en la variable apuntada por caa_out. Es responsabilidad del llamador liberar la estructura resultante caa_out cuando ya no sea necesaria, utilizando la función ares_free_data(3).

La estructura ares_caa_reply(3) contiene los siguientes campos:


struct ares_caa_reply {
  struct ares_caa_reply *next;
  int                    critical;
  unsigned char         *property;
  size_t                 plength; /* plength excluye el nulo */
  unsigned char         *value;
  size_t                 length;  /* length excluye el nulo */
};

VALORES DE RETORNO

ares_parse_caa_reply puede devolver cualquiera de los siguientes valores:

ARES_SUCCESS
La respuesta se analizó correctamente.
ARES_EBADRESP
La respuesta estaba malformada.
ARES_ENODATA
La respuesta no contenía una respuesta a la consulta.
ARES_ENOMEM
Se agotó la memoria.

EJEMPLO

#include <arpa/inet.h>
#include <time.h>
#include <sys/time.h>
#include <netdb.h>
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>
#include "ares.h"
static void dns_callback(void *arg,
                         int status,
                         int timeouts,
                         unsigned char *abuf,
                         int alen)
  {
    struct ares_caa_reply *caa_out;
    int err;
    err = ares_parse_caa_reply (abuf, alen, &caa_out);
    if (err == ARES_SUCCESS)
      {
        struct ares_caa_reply *caa_curr;
        for (caa_curr=caa_out; caa_curr; caa_curr=caa_curr->next)
          printf ("%s. CAA %i %s \"%s\"\n", arg,
                                            caa_curr->critical,
                                            caa_curr->property,
                                            caa_curr->value);
      }
    else
      {
        printf ("err=%i\n", err);
      }
    ares_free_data (caa_out);
  }
static void main_loop(ares_channel_t **channel)
  {
    int nfds, count;
    fd_set readers, writers;
    struct timeval tv, *tvp;
    while (1)
      {
        FD_ZERO (&readers);
        FD_ZERO (&writers);
        nfds = ares_fds (*channel, &readers, &writers);
        if (nfds == 0)
          break;
        tvp = ares_timeout (*channel, NULL, &tv);
        count = select (nfds, &readers, &writers, NULL, tvp);
        ares_process (*channel, &readers, &writers);
      }
  }
int main(int argc, char **argv)
  {
    const char *sversion;
    int iversion;
    int err;
    sversion = ares_version (&iversion);
    printf ("c-ares version %s\n", sversion);
    char *domain = "wikipedia.org";
    if (argc > 1)
      domain = argv[1];
    ares_channel_t *channel;
    if ((err = ares_init (&channel)) != ARES_SUCCESS)
      {
        printf ("ares_init() failed (%i)\n", err);
        exit (EXIT_FAILURE);
      }
    ares_query (channel, domain,
                1,   /* ns_c_in */
                257, /* T_CAA */
                dns_callback, domain);
    main_loop (&channel);
    ares_destroy (channel);
    exit (EXIT_SUCCESS);
  }
